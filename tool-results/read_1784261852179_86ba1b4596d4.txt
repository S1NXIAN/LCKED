     3→/**
     4→ * LCKED — VaultsSidebar
     5→ * ---------------------------------------------------------------------------
     6→ * Proton Pass–style vault selector. Renders the fixed rows (All / Favorites /
     7→ * Trash) plus every user-defined vault that hasn't been hidden. Each row is a
     8→ * drag-and-drop drop target so an item dragged from the list can be moved
     9→ * between vaults.
    10→ *
    11→ * Layout (post VAULTSIDEBAR-REWRITE):
    12→ *   • Every row uses a two-line layout — name on top, "N items" below in
    13→ *     smaller muted text. The 3-vertical-dots ⋮ menu button is ALWAYS visible
    14→ *     on rows that have a menu (no longer hover-gated) and lives in a reserved
    15→ *     7×7 slot on the right edge so the right edges line up across rows.
    16→ *
    17→ * Menus:
    18→ *   • All Items  ⮕  Organize vaults   (opens AlertDialog with hide/unhide checkboxes)
    19→ *   • Trash      ⮕  Restore all items / Empty trash
    20→ *   • Custom     ⮕  Edit / Move all items (submenu, only when vault has >1 item)
    21→ *                   / Hide vault / Delete vault
    22→ *
    23→ * Pattern notes:
    24→ *   • Rows that host a ⋮ dropdown are rendered as <div role="button"> (NOT
    25→ *     <button>) so the trigger button doesn't nest inside another button —
    26→ *     nested buttons are invalid HTML and break the accessibility tree.
    27→ *   • Hidden vaults (ids in `hiddenVaultIds`) are filtered out of the sidebar
    28→ *     list but remain valid move targets — the "Move all items" submenu lists
    29→ *     every other custom vault, hidden or not.
    30→ *   • "Edit" (was "Rename") opens the vault editor sheet via
    31→ *     `setVaultEditorOpen(true, id)`; "Delete vault" calls `deleteVault` (which
    32→ *     orphan-rescues any items inside).
    33→ */

   165→/**
   166→ * Render a vault's icon inside a colored rounded swatch. The fill is the
   167→ * vault color at ~16% opacity; the icon glyph uses the full hex for contrast.
   168→ * Used by VaultsSidebar rows AND the create-vault-dialog picker preview.
   169→ */
   170→export function VaultIcon({ icon, color, size = 28, className, bare = false }: VaultIconProps & { bare?: boolean }) {
   171→  const hex = vaultColorHex(color);
   172→  const Resolved = LUCIDE_BY_ID[icon] ?? Home;
   173→  const px = `${size}px`;
   174→  const glyph = Math.round(size * 0.55);
   175→  if (bare) {
   176→    return (
   177→      <Resolved
   178→        size={glyph}
   179→        strokeWidth={2}
   180→        style={{ color: hex }}
   181→        className={cn("shrink-0", className)}
   182→        aria-hidden="true"
   183→      />
   184→    );
   185→  }
   186→  return (
   187→    <span
   188→      className={cn("inline-flex shrink-0 items-center justify-center rounded-lg", className)}
   189→      style={{
   190→        width: px,
   191→        height: px,
   192→        backgroundColor: `${hex}29`, // ~16% opacity
   193→        color: hex,
   194→      }}
   195→      aria-hidden="true"
   196→    >
   197→      <Resolved size={glyph} strokeWidth={2} />
   198→    </span>
   199→  );
   200→}

   229→interface VaultRowProps {
   230→  /** Glyph rendered in the swatch. Pass a Lucide component for fixed rows. */
   231→  icon: React.ReactNode;
   232→  /** Row label. */
   233→  label: string;
   234→  /** Item count shown beneath the label. */
   235→  count: number;
   236→  /** Whether this row is the active filter. */
   237→  active: boolean;
   238→  /** Click handler — sets the active vault filter. */
   239→  onSelect: () => void;
   240→  /** Optional drag-and-drop target vault id (undefined = no drop). */
   241→  dropVaultId?: string | null;
   242→  /** Optional amber-tinted style (used by Trash when non-empty). */
   243→  warn?: boolean;
   244→  /** Optional ⋮ menu node (rendered in the reserved right-edge slot). */
   245→  menu?: React.ReactNode;
   246→  /** Whether the row is being dragged over. */
   247→  dragOver?: boolean;
   248→}
   249→
   250→/**
   251→ * A single sidebar row. Two-line layout: name on top, "N items" beneath in
   252→ * smaller muted text. When `menu` is provided we render a div instead of a
   253→ * button to avoid invalid nested-button HTML. The reserved `menuSlot` keeps
   254→ * every row's right edge aligned whether or not a menu is present.
   255→ */
   256→function VaultRow({
   257→  icon,
   258→  label,
   259→  count,
   260→  active,
   261→  onSelect,
   262→  dropVaultId,
   263→  warn,
   264→  menu,
   265→  dragOver,
   266→}: VaultRowProps) {
   267→  const Tag = menu ? "div" : "button";
   268→  const tagProps = menu
   269→    ? { role: "button", tabIndex: 0 }
   270→    : { type: "button" as const };
   271→
   272→  const handleDrop = (e: React.DragEvent) => {
   273→    if (dropVaultId === undefined) return;
   274→    e.preventDefault();
   275→    // Handle multi-item drag (from multi-select mode).
   276→    const multiData = e.dataTransfer.getData("text/lcked-items");
   277→    if (multiData) {
   278→      try {
   279→        const ids = JSON.parse(multiData) as string[];
   280→        for (const id of ids) {
   281→          useVault.getState().moveItemToVault(id, dropVaultId ?? null);
   282→        }
   283→        toast.success(`Moved ${ids.length} item${ids.length === 1 ? "" : "s"} to ${label}`);
   284→        return;
   285→      } catch { /* fall through to single */ }
   286→    }
   287→    const id = e.dataTransfer.getData("text/lcked-item");
   288→    if (!id) return;
   289→    // dropVaultId === null means "default vault" (no vault).
   290→    useVault.getState().moveItemToVault(id, dropVaultId ?? null);
   291→    toast.success(`Moved to ${label}`);
   292→  };
   293→  const handleDragOver = (e: React.DragEvent) => {
   294→    if (dropVaultId === undefined) return;
   295→    // Allow drop so the browser fires onDrop.
   296→    e.preventDefault();
   297→    e.dataTransfer.dropEffect = "move";
   298→  };
   299→  const handleKeyDown = (e: React.KeyboardEvent) => {
   300→    if (!menu) return;
   301→    if (e.key === "Enter" || e.key === " ") {
   302→      e.preventDefault();
   303→      onSelect();
   304→    }
   305→  };
   306→
   307→  const countLabel = `${count} ${count === 1 ? "item" : "items"}`;
   308→
   309→  return (
   310→    <Tag
   311→      {...tagProps}
   312→      onClick={(e: React.MouseEvent) => {
   313→        // If the click landed on the menu slot, don't toggle the row.
   314→        const target = e.target as HTMLElement;
   315→        if (target.closest("[data-menu-slot]")) return;
   316→        onSelect();
   317→      }}
   318→      onKeyDown={menu ? handleKeyDown : undefined}
   319→      onDrop={handleDrop}
   320→      onDragOver={handleDragOver}
   321→      aria-current={active ? "true" : undefined}
   322→      className={cn(
   323→        "group relative flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition-colors",
   324→        "focus:outline-none focus-visible:ring-2 focus-visible:ring-ring/60",
   325→        active
   326→          ? "bg-accent text-accent-foreground"
   327→          : warn
   328→            ? "text-foreground hover:bg-amber-500/10"
   329→            : "text-foreground/80 hover:bg-muted/60 hover:text-foreground",
   330→        dragOver && "ring-2 ring-primary/60 ring-offset-1 ring-offset-background",
   331→      )}
   332→    >
   333→      <span className="flex h-6 w-6 shrink-0 items-center justify-center">{icon}</span>
   334→      <span className="min-w-0 flex-1">
   335→        <span className="block truncate text-sm font-medium leading-tight">
   336→          {label}
   337→        </span>
   338→        <span
   339→          className={cn(
   340→            "block text-xs leading-tight tabular-nums",
   341→            warn && !active
   342→              ? "text-amber-400/80"
   343→              : active
   344→                ? "text-accent-foreground/70"
   345→                : "text-muted-foreground",
   346→          )}
   347→        >
   348→          {countLabel}
   349→        </span>
   350→      </span>
   351→      {/* Reserved menu slot — same width on every row so right edges align. */}
   352→      <span
   353→        data-menu-slot
   354→        className="relative z-10 flex h-6 w-6 shrink-0 items-center justify-center"
   355→      >
   356→        {menu}
   357→      </span>
   358→    </Tag>
   359→  );
   360→}

   389→  active,
   390→  count,

   392→  otherVaults,
   393→  onSelect,
   394→  onEdit,

   400→  onDropClear,

   402→  const [confirmOpen, setConfirmOpen] = React.useState(false);
   403→  // "Move all items" only makes sense when there's more than one to move.
   404→  const showMoveAll = count > 1;
   405→
   406→  const handleMoveAll = (toVaultId: string | null) => {
   407→    // Fire-and-forget — the parent handler shows the toast + switches view.

   409→  };

   411→  return (
   412→    <div
   413→      onDragEnter={onDragEnterRow}

   429→                <MoreVertical className="h-3.5 w-3.5" />

   434→                <Pencil className="h-3.5 w-3.5" />

   448→                          className="inline-flex h-4 w-4 shrink-0 items-center justify-center rounded"
   449→                          style={{ backgroundColor: "#7777F829", color: "#7777F8" }}

   458→                        <DropdownMenuItem
   459→                          key={v.id}
   460→                          onSelect={() => handleMoveAll(v.id)}

   476→              <DropdownMenuItem
   477→                variant="destructive"
   478→                onSelect={(e) => {
   479→                  e.preventDefault();

   495→              The vault will be removed. Items inside it will be moved to your

   498→          </AlertDialogHeader>
   499→          <AlertDialogFooter>
   500→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   501→            <AlertDialogAction
   502→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"

   505→                onDelete();
   506→              }}
   507→            >
   508→              Delete vault
   509→            </AlertDialogAction>
   510→          </AlertDialogFooter>
   511→        </AlertDialogContent>
   512→      </AlertDialog>

   519→export function VaultsSidebar() {
   520→  const items = useVault((s) => s.items);
   521→  const vaults = useVault((s) => s.vaults);
   522→  const hiddenVaultIds = useVault((s) => s.hiddenVaultIds);
   523→  const activeVault = useVault((s) => s.activeVault);
   524→  const setActiveVault = useVault((s) => s.setActiveVault);
   525→  const setCreateVaultDialogOpen = useVault((s) => s.setCreateVaultDialogOpen);
   526→  const setVaultEditorOpen = useVault((s) => s.setVaultEditorOpen);
   527→  const deleteVault = useVault((s) => s.deleteVault);
   528→  const toggleVaultHidden = useVault((s) => s.toggleVaultHidden);
   529→  const moveAllItems = useVault((s) => s.moveAllItems);
   530→  const reorderVaults = useVault((s) => s.reorderVaults);
   531→  const emptyTrash = useVault((s) => s.emptyTrash);
   532→  const restoreItem = useVault((s) => s.restoreItem);
   533→
   534→  // Track which row is currently being dragged-over so we can ring-highlight.
   535→  const [overKey, setOverKey] = React.useState<string | null>(null);
   536→  // "Organize vaults" dialog (triggered from All Items ⋮ menu).
   537→  const [organizeOpen, setOrganizeOpen] = React.useState(false);
   538→  // "Empty trash" confirmation (triggered from Trash ⋮ menu).
   539→  const [emptyTrashOpen, setEmptyTrashOpen] = React.useState(false);
   540→
   541→  // Helper: extract item IDs from a drop event (handles single + multi drag).
   542→  const extractDroppedIds = (e: React.DragEvent): string[] => {
   543→    const multi = e.dataTransfer.getData("text/lcked-items");
   544→    if (multi) {
   545→      try { return JSON.parse(multi); } catch { /* fall through */ }
   546→    }
   547→    const single = e.dataTransfer.getData("text/lcked-item");
   548→    return single ? [single] : [];
   549→  };
   550→
   551→  // Hidden vaults are excluded from the sidebar list AND from search.
   552→  const visibleVaults = vaults.filter((v) => !hiddenVaultIds.includes(v.id));
   553→
   554→  // Live counts.
   555→  const allCount = items.filter((i) => !i.trashed).length;
   556→  const favCount = items.filter((i) => !i.trashed && i.favorite).length;
   557→  const trashCount = items.filter((i) => i.trashed).length;
   558→  const vaultCount = (id: string) =>
   559→    items.filter((i) => !i.trashed && i.vaultId === id).length;
   560→
   561→  const handleDelete = async (v: VaultDef) => {
   562→    try {
   563→      await deleteVault(v.id);
   564→      toast.success(`Deleted vault “${v.name}”`);
   565→    } catch {
   566→      toast.error("Could not delete vault");
   567→    }
   568→  };
   569→
   570→  const handleHide = (v: VaultDef) => {
   571→    toggleVaultHidden(v.id);
   572→    // If the user just hid the active vault, jump back to All Items so they
   573→    // don't end up looking at an empty list filter.
   574→    if (activeVault === v.id) setActiveVault("all");
   575→    toast.success(`Hid “${v.name}” — reveal it from Organize vaults`);
   576→  };
   577→
   578→  const handleMoveAll = async (
   579→    fromVault: VaultDef,
   580→    toVaultId: string | null,
   581→  ) => {
   582→    const toLabel =
   583→      toVaultId === null
   584→        ? "All Items"
   585→        : (vaults.find((x) => x.id === toVaultId)?.name ?? "vault");
   586→    try {
   587→      await moveAllItems(fromVault.id, toVaultId);
   588→      toast.success(`Moved all items from “${fromVault.name}” to ${toLabel}`);
   589→      // Switch to the destination so the user sees the moved items.
   590→      setActiveVault(toVaultId ?? "all");
   591→    } catch {
   592→      toast.error("Could not move items");
   593→    }
   594→  };
   595→
   596→  const handleRestoreAll = async () => {
   597→    const trashed = items.filter((i) => i.trashed);
   598→    if (trashed.length === 0) return;
   599→    try {
   600→      await Promise.all(trashed.map((t) => restoreItem(t.id)));
   601→      toast.success(
   602→        `Restored ${trashed.length} ${trashed.length === 1 ? "item" : "items"}`,
   603→      );
   604→    } catch {
   605→      toast.error("Could not restore all items");
   606→    }
   607→  };
   608→
   609→  const handleEmptyTrash = async () => {
   610→    try {
   611→      await emptyTrash();
   612→      toast.success("Trash emptied");
   613→    } catch {
   614→      toast.error("Could not empty trash");
   615→    }
   616→  };
   617→
   618→  // Helper to render a single custom-vault row (with menu).
   619→  const renderCustomVault = (v: VaultDef) => (
   620→    <CustomVaultRow
   621→      key={v.id}
   622→      vault={v}
   623→      active={activeVault === v.id}
   624→      count={vaultCount(v.id)}
   625→      dragOver={overKey === `vault-${v.id}`}
   626→      otherVaults={vaults.filter((o) => o.id !== v.id)}
   627→      onSelect={() => setActiveVault(v.id)}
   628→      onEdit={() => setVaultEditorOpen(true, v.id)}
   629→      onDelete={() => handleDelete(v)}
   630→      onHide={() => handleHide(v)}
   631→      onMoveAll={(toVaultId) => handleMoveAll(v, toVaultId)}
   632→      onDragEnterRow={() => setOverKey(`vault-${v.id}`)}
   633→      onDragLeaveRow={(e) => {
   634→        const related = e.relatedTarget as Node | null;
   635→        if (related && (e.currentTarget as Node).contains(related)) return;
   636→        setOverKey((c) => (c === `vault-${v.id}` ? null : c));
   637→      }}
   638→      onDropClear={() => setOverKey(null)}
   639→    />
   640→  );
   641→
   642→  return (
   643→    <div className="lcked-scroll flex h-full flex-col gap-1 overflow-y-auto px-2 py-2">
   644→      {/* Header */}
   645→      <div className="flex items-center justify-between px-3 py-2">
   646→        <span className="text-[10px] font-semibold uppercase tracking-[0.18em] text-muted-foreground">
   647→          Vaults
   648→        </span>
   649→        <Button
   650→          size="icon"
   651→          variant="ghost"
   652→          className="h-6 w-6 text-muted-foreground hover:text-foreground"
   653→          onClick={() => setCreateVaultDialogOpen(true)}
   654→          aria-label="Create vault"
   655→        >
   656→          <Plus className="h-4 w-4" />
   657→        </Button>
   658→      </div>
   659→
   660→      {/* All items — drop = move to default vault (vaultId = null). */}
   661→      <div
   662→        onDragEnter={() => setOverKey("all")}
   663→        onDragLeave={(e) => {
   664→          const related = e.relatedTarget as Node | null;
   665→          if (related && (e.currentTarget as Node).contains(related)) return;
   666→          setOverKey((c) => (c === "all" ? null : c));
   667→        }}
   668→        onDrop={async (e) => {
   669→          setOverKey(null);
   670→          const ids = extractDroppedIds(e);
   671→          if (ids.length === 0) return;
   672→          try {
   673→            for (const id of ids) await useVault.getState().moveItemToVault(id, null);
   674→            toast.success(`Moved ${ids.length} item${ids.length === 1 ? "" : "s"} to All Items`);
   675→          } catch {
   676→            toast.error("Could not move item");
   677→          }
   678→        }}
   679→        onDragOver={(e) => {
   680→          e.preventDefault();
   681→          e.dataTransfer.dropEffect = "move";
   682→        }}
   683→      >
   684→        <VaultRow
   685→          icon={
   686→            <span
   687→              className="inline-flex h-7 w-7 items-center justify-center rounded-lg"
   688→              style={{ backgroundColor: "#7777F829", color: "#7777F8" }}
   689→              aria-hidden="true"
   690→            >
   691→              <ShieldCheck className="h-4 w-4" />
   692→            </span>
   693→          }
   694→          label="All Items"
   695→          count={allCount}
   696→          active={activeVault === "all"}
   697→          onSelect={() => setActiveVault("all")}
   698→          dragOver={overKey === "all"}
   699→          menu={
   700→            <DropdownMenu>
   701→              <DropdownMenuTrigger asChild>
   702→                <button type="button" className="flex h-6 w-6 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-muted hover:text-foreground" aria-label="All items options">
   703→                  <MoreVertical className="h-3.5 w-3.5" />
   704→                </button>
   705→              </DropdownMenuTrigger>
   706→              <DropdownMenuContent align="end" className="w-44">
   707→                <DropdownMenuItem onSelect={() => setOrganizeOpen(true)}>
   708→                  <SlidersHorizontal className="h-3.5 w-3.5" />
   709→                  Organize vaults
   710→                </DropdownMenuItem>
   711→              </DropdownMenuContent>
   712→            </DropdownMenu>
   713→          }
   714→        />
   715→      </div>
   716→
   717→      {/* Favorites — drop = toggle favorite on. No menu (info-only row). */}
   718→      <div
   719→        onDragEnter={() => setOverKey("favorites")}
   720→        onDragLeave={(e) => {
   721→          const related = e.relatedTarget as Node | null;
   722→          if (related && (e.currentTarget as Node).contains(related)) return;
   723→          setOverKey((c) => (c === "favorites" ? null : c));
   724→        }}
   725→        onDrop={async (e) => {
   726→          setOverKey(null);
   727→          const ids = extractDroppedIds(e);
   728→          if (ids.length === 0) return;
   729→          let count = 0;
   730→          for (const id of ids) {
   731→            const it = useVault.getState().items.find((i) => i.id === id);
   732→            if (!it) continue;
   733→            // Restore from trash if needed.
   734→            if (it.trashed) await useVault.getState().restoreItem(id);
   735→            // Favorite if not already.
   736→            if (!it.favorite) {
   737→              await useVault.getState().toggleFavorite(id);
   738→              count++;
   739→            }
   740→          }
   741→          if (count > 0) toast.success(`Added ${count} to Favorites`);
   742→          else toast.success("Items restored to Favorites");
   743→        }}
   744→        onDragOver={(e) => {
   745→          e.preventDefault();
   746→          e.dataTransfer.dropEffect = "move";
   747→        }}
   748→      >
   749→        <VaultRow
   750→          icon={
   751→            <span
   752→              className="inline-flex h-7 w-7 items-center justify-center rounded-lg"
   753→              style={{ backgroundColor: "#FFB84D29", color: "#FFB84D" }}
   754→              aria-hidden="true"
   755→            >
   756→              <Star className="h-4 w-4" />
   757→            </span>
   758→          }
   759→          label="Favorites"
   760→          count={favCount}
   761→          active={activeVault === "favorites"}
   762→          onSelect={() => setActiveVault("favorites")}
   763→          dragOver={overKey === "favorites"}
   764→        />
   765→      </div>
   766→
   767→      {/* Custom vaults (hidden ones filtered out) */}
   768→      {visibleVaults.length > 0 && (
   769→        <div className="mt-1 flex flex-col gap-0.5">
   770→          {visibleVaults.map(renderCustomVault)}
   771→        </div>
   772→      )}
   773→
   774→      {/* Trash — drop = soft-delete (move to Trash). */}
   775→      <div
   776→        className="mt-auto"
   777→        onDragEnter={() => setOverKey("trash")}
   778→        onDragLeave={(e) => {
   779→          const related = e.relatedTarget as Node | null;
   780→          if (related && (e.currentTarget as Node).contains(related)) return;
   781→          setOverKey((c) => (c === "trash" ? null : c));
   782→        }}
   783→        onDrop={async (e) => {
   784→          setOverKey(null);
   785→          const ids = extractDroppedIds(e);
   786→          if (ids.length === 0) return;
   787→          try {
   788→            for (const id of ids) await useVault.getState().trashItem(id);
   789→            toast.success(`Moved ${ids.length} item${ids.length === 1 ? "" : "s"} to Trash`);
   790→          } catch {
   791→            toast.error("Could not move to Trash");
   792→          }
   793→        }}
   794→        onDragOver={(e) => {
   795→          e.preventDefault();
   796→          e.dataTransfer.dropEffect = "move";
   797→        }}
   798→      >
   799→        <VaultRow
   800→          icon={
   801→            <span
   802→              className={cn(
   803→                "inline-flex h-7 w-7 items-center justify-center rounded-lg",
   804→                trashCount > 0
   805→                  ? "bg-amber-500/15 text-amber-400"
   806→                  : "bg-muted text-muted-foreground",
   807→              )}
   808→              aria-hidden="true"
   809→            >
   810→              <Trash2 className="h-4 w-4" />
   811→            </span>
   812→          }
   813→          label="Trash"
   814→          count={trashCount}
   815→          active={activeVault === "trash"}
   816→          warn={trashCount > 0}
   817→          onSelect={() => setActiveVault("trash")}
   818→          dragOver={overKey === "trash"}
   819→          menu={
   820→            <DropdownMenu>
   821→              <DropdownMenuTrigger asChild>
   822→                <button type="button" className="flex h-6 w-6 items-center justify-center rounded text-muted-foreground transition-colors hover:bg-muted hover:text-foreground" aria-label="Trash options">
   823→                  <MoreVertical className="h-3.5 w-3.5" />
   824→                </button>
   825→              </DropdownMenuTrigger>
   826→              <DropdownMenuContent align="end" className="w-44">
   827→                <DropdownMenuItem
   828→                  disabled={trashCount === 0}
   829→                  onSelect={() => {
   830→                    void handleRestoreAll();
   831→                  }}
   832→                >
   833→                  <RotateCcw className="h-3.5 w-3.5" />
   834→                  Restore all items
   835→                </DropdownMenuItem>
   836→                <DropdownMenuSeparator />
   837→                <DropdownMenuItem
   838→                  variant="destructive"
   839→                  disabled={trashCount === 0}
   840→                  onSelect={(e) => {
   841→                    e.preventDefault();
   842→                    setEmptyTrashOpen(true);
   843→                  }}
   844→                >
   845→                  <Trash2 className="h-3.5 w-3.5" />
   846→                  Empty trash
   847→                </DropdownMenuItem>
   848→              </DropdownMenuContent>
   849→            </DropdownMenu>
   850→          }
   851→        />
   852→      </div>
   853→
   854→      {/* ---------------------------- Dialogs ---------------------------- */}
   855→
   856→      {/* Organize vaults — show/hide + reorder custom vaults. */}
   857→      <AlertDialog open={organizeOpen} onOpenChange={setOrganizeOpen}>
   858→        <AlertDialogContent>
   859→          <AlertDialogHeader>
   860→            <AlertDialogTitle>Organize vaults</AlertDialogTitle>
   861→            <AlertDialogDescription>
   862→              Toggle visibility and reorder vaults. Hidden vaults keep all their items but won&apos;t appear in the sidebar or search.
   863→            </AlertDialogDescription>
   864→          </AlertDialogHeader>
   865→          <div className="lcked-scroll max-h-80 overflow-y-auto pr-1">
   866→            {vaults.length === 0 ? (
   867→              <p className="px-1 py-6 text-center text-sm text-muted-foreground">
   868→                No custom vaults yet. Use the + button above to create one.
   869→              </p>
   870→            ) : (
   871→              <ul className="flex flex-col gap-0.5">
   872→                {vaults.map((v, idx) => {
   873→                  const isHidden = hiddenVaultIds.includes(v.id);
   874→                  const c = vaultCount(v.id);
   875→                  return (
   876→                    <li key={v.id} className="flex items-center gap-2">
   877→                      {/* Reorder buttons */}
   878→                      <div className="flex flex-col">
   879→                        <button
   880→                          type="button"
   881→                          disabled={idx === 0}
   882→                          className="flex h-4 w-5 items-center justify-center text-muted-foreground disabled:opacity-30 hover:text-foreground"
   883→                          onClick={() => {
   884→                            const newOrder = vaults.map((v) => v.id);
   885→                            [newOrder[idx - 1], newOrder[idx]] = [newOrder[idx], newOrder[idx - 1]];
   886→                            reorderVaults(newOrder);
   887→                          }}
   888→                          aria-label="Move up"
   889→                        >
   890→                          <ChevronUp className="h-3.5 w-3.5" />
   891→                        </button>
   892→                        <button
   893→                          type="button"
   894→                          disabled={idx === vaults.length - 1}
   895→                          className="flex h-4 w-5 items-center justify-center text-muted-foreground disabled:opacity-30 hover:text-foreground"
   896→                          onClick={() => {
   897→                            const newOrder = vaults.map((v) => v.id);
   898→                            [newOrder[idx + 1], newOrder[idx]] = [newOrder[idx], newOrder[idx + 1]];
   899→                            reorderVaults(newOrder);
   900→                          }}
   901→                          aria-label="Move down"
   902→                        >
   903→                          <ChevronDown className="h-3.5 w-3.5" />
   904→                        </button>
   905→                      </div>
   906→                      <label
   907→                        className={cn(
   908→                          "flex flex-1 cursor-pointer items-center gap-3 rounded-md px-2 py-2 transition-colors hover:bg-muted/60",
   909→                          isHidden && "opacity-60",
   910→                        )}
   911→                      >
   912→                        <Checkbox
   913→                          checked={!isHidden}
   914→                          onCheckedChange={() => toggleVaultHidden(v.id)}
   915→                          aria-label={`Show vault ${v.name} in sidebar`}
   916→                        />
   917→                        <VaultIcon icon={v.icon} color={v.color} size={24} />
   918→                        <div className="min-w-0 flex-1">
   919→                          <div className="truncate text-sm font-medium">
   920→                            {v.name}
   921→                          </div>
   922→                          <div className="text-xs text-muted-foreground">
   923→                            {c} {c === 1 ? "item" : "items"}
   924→                          </div>
   925→                        </div>
   926→                        {isHidden && (
   927→                          <span className="flex items-center gap-1 text-xs text-muted-foreground">
   928→                            <EyeOff className="h-3.5 w-3.5" />
   929→                            Hidden
   930→                          </span>
   931→                        )}
   932→                      </label>
   933→                    </li>
   934→                  );
   935→                })}
   936→              </ul>
   937→            )}
   938→          </div>
   939→          <AlertDialogFooter>
   940→            <AlertDialogAction>Done</AlertDialogAction>
   941→          </AlertDialogFooter>
   942→        </AlertDialogContent>
   943→      </AlertDialog>
   944→
   945→      {/* Empty trash — destructive confirm. */}
   946→      <AlertDialog open={emptyTrashOpen} onOpenChange={setEmptyTrashOpen}>
   947→        <AlertDialogContent>
   948→          <AlertDialogHeader>
   949→            <AlertDialogTitle>Empty trash?</AlertDialogTitle>
   950→            <AlertDialogDescription>
   951→              This will permanently delete {trashCount}{" "}
   952→              {trashCount === 1 ? "item" : "items"} from your trash. This action
   953→              cannot be undone.
   954→            </AlertDialogDescription>
   955→          </AlertDialogHeader>
   956→          <AlertDialogFooter>
   957→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   958→            <AlertDialogAction
   959→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   960→              onClick={() => {
   961→                setEmptyTrashOpen(false);
   962→                void handleEmptyTrash();
   963→              }}
   964→            >
   965→              Empty trash
   966→            </AlertDialogAction>
   967→          </AlertDialogFooter>
   968→        </AlertDialogContent>
   969→      </AlertDialog>
   970→    </div>
   971→  );
   972→}
