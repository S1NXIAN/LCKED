   140→function download(filename: string, content: string, mime: string) {
   141→  const blob = new Blob([content], { type: mime });
   142→  const url = URL.createObjectURL(blob);
   143→  const a = document.createElement("a");
   144→  a.href = url;
   145→  a.download = filename;
   146→  a.click();
   147→  URL.revokeObjectURL(url);
   148→}

   150→function formatBytes(bytes: number) {
   151→  if (!bytes) return "0 B";
   152→  if (bytes < 1024) return `${bytes} B`;
   153→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   154→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
   155→}

   486→                  "flex flex-col items-start gap-2 rounded-xl border p-3 text-left transition-all",
   487→                  active
   488→                    ? "border-primary bg-primary/5 ring-1 ring-primary/30"

   794→
   795→  const triggerFilePicker = (src: ImportSource) => {
   796→    setPendingSource(src);

   801→    setFile(f);
   802→    try {
   803→      const text = await f.text();
   804→      setDetectedFmt(detectFormat(f.name, text));
   805→    } catch {
   806→      setDetectedFmt("");

   809→
   810→  const handleImport = async () => {
   811→    if (!file) return;

   820→            : undefined,

   823→      setPendingSource(null);

  1172→/* ============================== Keyboard section ============================== */
  1173→
  1174→/**
  1175→ * Keyboard settings — WCAG 2.1.4 compliance. Three layers:
  1176→ *   1. Global disable toggle for all single-key shortcuts
  1177→ *   2. Per-shortcut remapping (capture-next-key)
  1178→ *   3. (input-scoped disabling is handled in useVaultKeybinds)
  1179→ */
  1180→function KeyboardSection() {
  1181→  const singleKeyDisabled = useKeyboardSettings((s) => s.singleKeyDisabled);
  1182→  const setSingleKeyDisabled = useKeyboardSettings((s) => s.setSingleKeyDisabled);
  1183→  const rebind = useKeyboardSettings((s) => s.rebind);
  1184→  const resetBindings = useKeyboardSettings((s) => s.resetBindings);
  1185→  const bindings = useKeyboardSettings((s) => s.bindings);
  1186→  const [capturing, setCapturing] = React.useState<string | null>(null);
  1187→
  1188→  React.useEffect(() => {
  1189→    if (!capturing) return;
  1190→    const onKey = (e: KeyboardEvent) => {
  1191→      e.preventDefault();
  1192→      e.stopPropagation();
  1193→      if (["Meta", "Control", "Shift", "Alt"].includes(e.key)) return;
  1194→      const parts: string[] = [];
  1195→      if (e.metaKey) parts.push("Meta");
  1196→      if (e.ctrlKey) parts.push("Control");
  1197→      if (e.shiftKey) parts.push("Shift");
  1198→      if (e.altKey) parts.push("Alt");
  1199→      parts.push(e.key.length === 1 ? e.key.toLowerCase() : e.key);
  1200→      rebind(capturing, parts.join("+"));
  1201→      setCapturing(null);
  1202→    };
  1203→    window.addEventListener("keydown", onKey, { capture: true });
  1204→    return () =>
  1205→      window.removeEventListener("keydown", onKey, { capture: true } as EventListenerOptions);
  1206→  }, [capturing, rebind]);
  1207→
  1208→  const remappable = SHORTCUTS.filter(
  1209→    (s) => (s.category === "list" || s.category === "detail") && s.default.length === 1,
  1210→  );
  1211→
  1212→  return (
  1213→    <>
  1214→      <Separator />
  1215→      <section className="space-y-3">
  1216→        <div className="flex items-center gap-2 text-sm font-semibold">
  1217→          <SettingsIcon className="h-4 w-4 text-muted-foreground" />
  1218→          Keyboard
  1219→        </div>
  1220→
  1221→        <label className="flex cursor-pointer items-center justify-between rounded-lg border border-border bg-muted/20 px-3 py-2.5">
  1222→          <span className="flex flex-col">
  1223→            <span className="text-sm">Disable single-key shortcuts</span>
  1224→            <span className="text-[11px] text-muted-foreground">
  1225→              Keeps modifier shortcuts (⌘K, ⌘⇧L…). Helps screen-reader users.
  1226→            </span>
  1227→          </span>
  1228→          <Switch checked={singleKeyDisabled} onCheckedChange={setSingleKeyDisabled} />
  1229→        </label>
  1230→
  1231→        <div className="space-y-1">
  1232→          <div className="flex items-center justify-between">
  1233→            <Label className="text-xs text-muted-foreground">Remap single-key shortcuts</Label>
  1234→            <button
  1235→              onClick={() => resetBindings()}
  1236→              className="text-[11px] text-muted-foreground underline-offset-2 hover:text-foreground hover:underline"
  1237→            >
  1238→              Reset to defaults
  1239→            </button>
  1240→          </div>
  1241→          <div className="lcked-scroll max-h-44 space-y-0.5 overflow-y-auto pr-1">
  1242→            {remappable.map((s) => (
  1243→              <div
  1244→                key={s.id}
  1245→                className="flex items-center justify-between gap-2 rounded-md px-2 py-1 hover:bg-muted/40"
  1246→              >
  1247→                <span className="text-xs text-foreground/90">{s.label}</span>
  1248→                <button
  1249→                  onClick={() => setCapturing(capturing === s.id ? null : s.id)}
  1250→                  className={cn(
  1251→                    "rounded-md border px-2 py-0.5 font-mono text-[11px] transition-colors",
  1252→                    capturing === s.id
  1253→                      ? "border-primary bg-primary/10 text-primary animate-pulse"
  1254→                      : "border-border bg-muted text-muted-foreground hover:text-foreground",
  1255→                  )}
  1256→                >
  1257→                  {capturing === s.id ? "press keys…" : formatBinding(bindings[s.id] ?? s.default)}
  1258→                </button>
  1259→              </div>
  1260→            ))}
  1261→          </div>
  1262→        </div>
  1263→      </section>
  1264→    </>
  1265→  );
  1266→}
