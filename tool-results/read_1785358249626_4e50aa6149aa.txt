    78→    78→function blankItem(type: ItemType): NewItemInput {
    79→    79→  const base = {
    80→    80→    name: "",
    81→    81→    favorite: false,
    82→    82→    pinned: false,
    83→    83→    folder: "",
    84→    84→    customFields: [],
    85→    85→    createdAt: Date.now(),
    86→    86→    updatedAt: Date.now(),
    87→    87→    vaultId: null as string | null,
    88→    88→    trashed: false,
    89→    89→    trashedAt: null as number | null,
    90→    90→  };
    91→    91→  if (type === "login") {
    92→    92→    return { ...base, type, details: { username: "", password: "", urls: [""], totp: "", notes: "" } };
    93→    93→  }
    94→    94→  if (type === "note") {
    95→    95→    return { ...base, type, details: { content: "" } };
    96→    96→  }
    97→    97→  if (type === "card") {
    98→    98→    return {
    99→    99→      ...base,
   100→   100→      type,
   101→   101→      details: { cardholder: "", number: "", brand: "", cvv: "", expiry: "", pin: "", notes: "" },
   102→   102→    };
   103→   103→  }
   104→   104→  return {
   105→   105→    ...base,
   106→   106→    type,
   107→   107→    details: {
   108→   108→      firstName: "", lastName: "", email: "", phone: "", company: "",
   109→   109→      address1: "", address2: "", city: "", state: "", zip: "", country: "", notes: "",
   110→   110→    },
   111→   111→  };
   112→   112→}

   198→   198→export function ItemEditor() {
   199→   199→  const open = useVault((s) => s.editorOpen);
   200→   200→  const editorItemId = useVault((s) => s.editorItemId);
   201→   201→  const vaults = useVault((s) => s.vaults);
   202→   202→  const settings = useVault((s) => s.settings);
   203→   203→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   204→   204→  const saveItem = useVault((s) => s.saveItem);
   205→   205→
   206→   206→  const [form, setForm] = React.useState<NewItemInput | null>(null);
   207→   207→  const [busy, setBusy] = React.useState(false);
   208→   208→  // Stable React keys for URL + custom-field rows (IE-2). Index-based keys
   209→   209→  // cause focus/selection to jump to the wrong field when a middle row is
   210→   210→  // removed. These arrays stay in lockstep with urls / customFields.
   211→   211→  const [urlKeys, setUrlKeys] = React.useState<string[]>([]);
   212→   212→  const [cfKeys, setCfKeys] = React.useState<string[]>([]);
   213→   213→  const keyCounter = React.useRef(0);
   214→   214→  const nextKey = React.useCallback(() => `k${++keyCounter.current}`, []);
   215→   215→  // Track the initial form snapshot for dirty-checking (IE-4).
   216→   216→  const [initialForm, setInitialForm] = React.useState<string>("");
   217→   217→  const [confirmDiscardOpen, setConfirmDiscardOpen] = React.useState(false);
   218→   218→
   219→   219→  // Initialise form when the sheet opens. Depends ONLY on [open, editorItemId]
   220→   220→  // (IE-1) — the `items` array changes on every unrelated mutation (auto-purge,
   221→   221→  // duplicate, drag-drop), which would overwrite the user's unsaved edits.
   222→   222→  React.useEffect(() => {
   223→   223→    if (!open) {
   224→   224→      setForm(null);
   225→   225→      setUrlKeys([]);
   226→   226→      setCfKeys([]);
   227→   227→      setInitialForm("");
   228→   228→      return;
   229→   229→    }
   230→   230→    if (editorItemId) {
   231→   231→      // Read via getState() so we don't depend on the items array.
   232→   232→      const existing = useVault.getState().items.find((i) => i.id === editorItemId);
   233→   233→      if (existing) {
   234→   234→        const input = itemToInput(existing);
   235→   235→        setForm(input);
   236→   236→        setUrlKeys((existing.details as { urls?: string[] }).urls?.map(() => nextKey()) ?? []);
   237→   237→        setCfKeys(existing.customFields.map(() => nextKey()));
   238→   238→        setInitialForm(JSON.stringify(input));
   239→   239→        return;
   240→   240→      }
   241→   241→    }
   242→   242→    // Honour a type pre-selected via ⌘K / icon rail, then clear the stash.
   243→   243→    const initialType = consumeNewItemType() ?? "login";
   244→   244→    const blank = blankItem(initialType);
   245→   245→    setForm(blank);
   246→   246→    setUrlKeys([""].map(() => nextKey()));
   247→   247→    setCfKeys([]);
   248→   248→    setInitialForm(JSON.stringify(blank));
   249→   249→  }, [open, editorItemId, nextKey]);
   250→   250→
   251→   251→  const isEditing = Boolean(editorItemId);
   252→   252→
   253→   253→  if (!open || !form) return null;
   254→   254→
   255→   255→  const isDirty = JSON.stringify(form) !== initialForm;
   256→   256→
   257→   257→  const update = (patch: Partial<NewItemInput>) => setForm((f) => (f ? { ...f, ...patch } : f));
   258→   258→  const updateDetails = (patch: Record<string, unknown>) =>
   259→   259→    setForm((f) => (f ? { ...f, details: { ...(f.details as object), ...patch } } : f));
   260→   260→
   261→   261→  const handleSave = async () => {
   262→   262→    if (!form) return;
   263→   263→    const trimmedName = form.name.trim();
   264→   264→    if (!trimmedName) {
   265→   265→      toast.error("Please give the item a name");
   266→   266→      return;
   267→   267→    }
   268→   268→    // Trim the name before saving so "  GitHub  " doesn't break fuzzy-search.
   269→   269→    const toSave = { ...form, name: trimmedName };
   270→   270→    setBusy(true);
   271→   271→    try {
   272→   272→      await saveItem(toSave, editorItemId ?? undefined);
   273→   273→      toast.success(isEditing ? "Item updated" : "Item created");
   274→   274→      setEditorOpen(false);
   275→   275→    } catch (err) {
   276→   276→      console.error(err);
   277→   277→      toast.error("Could not save item");
   278→   278→    } finally {
   279→   279→      setBusy(false);
   280→   280→    }
   281→   281→  };
   282→   282→
   283→   283→  // Intercept Sheet close: if there are unsaved edits, ask first (IE-4).
   284→   284→  const handleOpenChange = (o: boolean) => {
   285→   285→    if (!o && isDirty && !busy) {
   286→   286→      setConfirmDiscardOpen(true);
   287→   287→      return;
   288→   288→    }
   289→   289→    setEditorOpen(o);
   290→   290→  };
   291→   291→
   292→   292→  const addCustomField = () => {
   293→   293→    update({ customFields: [...form.customFields, { name: "", value: "", type: "text" }] });
   294→   294→    setCfKeys((k) => [...k, nextKey()]);
   295→   295→  };
   296→   296→  const updateCustomField = (idx: number, patch: Partial<CustomField>) =>
   297→   297→    update({
   298→   298→      customFields: form.customFields.map((cf, i) => (i === idx ? { ...cf, ...patch } : cf)),
   299→   299→    });
   300→   300→  const removeCustomField = (idx: number) => {
   301→   301→    update({ customFields: form.customFields.filter((_, i) => i !== idx) });
   302→   302→    setCfKeys((k) => k.filter((_, i) => i !== idx));
   303→   303→  };
   304→   304→
   305→   305→  // Login URL list helpers.
   306→   306→  const urls = form.type === "login" ? form.details.urls : [];
   307→   307→  const setUrl = (idx: number, val: string) => {
   308→   308→    const next = urls.map((u, i) => (i === idx ? val : u));
   309→   309→    updateDetails({ urls: next });
   310→   310→  };
   311→   311→  const addUrl = () => {
   312→   312→    updateDetails({ urls: [...urls, ""] });
   313→   313→    setUrlKeys((k) => [...k, nextKey()]);
   314→   314→  };
   315→   315→  const removeUrl = (idx: number) => {
   316→   316→    updateDetails({ urls: urls.filter((_, i) => i !== idx) });
   317→   317→    setUrlKeys((k) => k.filter((_, i) => i !== idx));
   318→   318→  };
   319→   319→
   320→   320→  // Switch item type while preserving overlapping fields (IE-3).
   321→   321→  const switchType = (t: ItemType) => {
   322→   322→    setForm((f) => {
   323→   323→      if (!f) return f;
   324→   324→      const blank = blankItem(t);
   325→   325→      return { ...blank, name: f.name, favorite: f.favorite, pinned: f.pinned, folder: f.folder, customFields: f.customFields, vaultId: f.vaultId };
   326→   326→    });
   327→   327→  };
   328→   328→
   329→   329→  // Selected vault (for the header dropdown).
   330→   330→  const selectedVault = form.vaultId ? vaults.find((v) => v.id === form.vaultId) ?? null : null;
   331→   331→
   332→   332→  return (
   333→   333→    <Sheet open={open} onOpenChange={handleOpenChange}>
   334→   334→      <SheetContent
   335→   335→        side="right"
   336→   336→        className="w-full gap-0 overflow-hidden border-l border-border bg-background p-0 sm:max-w-[480px] [&>button:last-child]:hidden"
   337→   337→      >
   338→   338→        {/* Header — type icon + title (left); vault selector + Save (right) */}
   339→   339→        <SheetHeader className="flex-row items-center justify-between gap-2 border-b border-border px-4 py-3">
   340→   340→          <SheetTitle className="flex min-w-0 items-center gap-2.5 text-base font-semibold">
   341→   341→            <ItemTypeIcon type={form.type} size="sm" />
   342→   342→            <span className="truncate">{isEditing ? "Edit item" : "New item"}</span>
   343→   343→          </SheetTitle>
   344→   344→          <div className="flex shrink-0 items-center gap-1.5">
   345→   345→            {/* Vault selector dropdown */}
   346→   346→            <DropdownMenu>
   347→   347→              <DropdownMenuTrigger asChild>
   348→   348→                <button
   349→   349→                  type="button"
   350→   350→                  className="flex h-8 items-center gap-1.5 rounded-md border border-border bg-background px-2 text-xs text-foreground transition-colors hover:bg-muted/60"
   351→   351→                  aria-label="Select vault"
   352→   352→                  title={selectedVault ? selectedVault.name : "No vault"}
   353→   353→                >
   354→   354→                  {selectedVault ? (
   355→   355→                    <VaultIcon icon={selectedVault.icon} color={selectedVault.color} size={18} />
   356→   356→                  ) : (
   357→   357→                    <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   358→   358→                      —
   359→   359→                    </span>
   360→   360→                  )}
   361→   361→                  <span className="max-w-[80px] truncate">{selectedVault ? selectedVault.name : "No vault"}</span>
   362→   362→                  <ChevronsUpDown className="h-3 w-3 text-muted-foreground" />
   363→   363→                </button>
   364→   364→              </DropdownMenuTrigger>
   365→   365→              <DropdownMenuContent align="end" className="min-w-[180px]">
   366→   366→                <DropdownMenuItem
   367→   367→                  onSelect={() => update({ vaultId: null })}
   368→   368→                  className={cn("gap-2", !selectedVault && "bg-muted/60")}
   369→   369→                >
   370→   370→                  <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   371→   371→                    —
   372→   372→                  </span>
   373→   373→                  No vault
   374→   374→                </DropdownMenuItem>
   375→   375→                {vaults.map((v) => (
   376→   376→                  <DropdownMenuItem
   377→   377→                    key={v.id}
   378→   378→                    onSelect={() => update({ vaultId: v.id })}
   379→   379→                    className={cn("gap-2", selectedVault?.id === v.id && "bg-muted/60")}
   380→   380→                  >
   381→   381→                    <VaultIcon icon={v.icon} color={v.color} size={18} />
   382→   382→                    <span className="truncate">{v.name}</span>
   383→   383→                    {selectedVault?.id === v.id && <Check className="ml-auto h-3.5 w-3.5" />}
   384→   384→                  </DropdownMenuItem>
   385→   385→                ))}
   386→   386→              </DropdownMenuContent>
   387→   387→            </DropdownMenu>
   388→   388→
   389→   389→            {/* Save / Create — type="submit" so Enter inside any input saves. */}
   390→   390→            <Button
   391→   391→              type="submit"
   392→   392→              size="sm"
   393→   393→              disabled={busy}
   394→   394→              className="min-w-[72px]"
   395→   395→              form="lcked-item-editor-form"
   396→   396→            >
   397→   397→              {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : isEditing ? "Save" : "Create"}
   398→   398→            </Button>
   399→   399→          </div>
   400→   400→        </SheetHeader>
   401→   401→        <SheetDescription className="sr-only">
   402→   402→          {isEditing ? "Edit the selected vault item." : "Create a new vault item."}
   403→   403→        </SheetDescription>
   404→   404→
   405→   405→        {/* Body — flat scrollable form. The <form> wraps the body + footer so
   406→   406→            Enter inside any input triggers handleSave (type="submit" button
   407→   407→            in the header references this form via the `form` attribute). */}
   408→   408→        <form id="lcked-item-editor-form" onSubmit={(e) => { e.preventDefault(); handleSave(); }} className="flex min-h-0 flex-1 flex-col">
   409→   409→        <div className="lcked-scroll flex-1 overflow-y-auto p-4">
   410→   410→          {/* Type selector — only when creating */}
   411→   411→          {!isEditing && (
   412→   412→            <div className="mb-4 grid grid-cols-4 gap-2">
   413→   413→              {(Object.keys(ITEM_TYPE_LABELS) as ItemType[]).map((t) => (
   414→   414→                <button
   415→   415→                  key={t}
   416→   416→                  type="button"
   417→   417→                  onClick={() => switchType(t)}
   418→   418→                  className={cn(
   419→   419→                    "flex flex-col items-center gap-1.5 rounded-lg border p-3 text-xs transition-colors",
   420→   420→                    form.type === t
   421→   421→                      ? "border-primary bg-primary/10 text-foreground"
   422→   422→                      : "border-border text-muted-foreground hover:bg-muted/50",
   423→   423→                  )}
   424→   424→                >
   425→   425→                  <ItemTypeIcon type={t} size="sm" />
   426→   426→                  {ITEM_TYPE_LABELS[t]}
   427→   427→                </button>
   428→   428→              ))}
   429→   429→            </div>
   430→   430→          )}
   431→   431→
   432→   432→          {/* Name + favorite star — inline to save space */}
   433→   433→          <div className="mb-4">
   434→   434→            <div className="mb-1 flex items-center justify-between">
   435→   435→              <div className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   436→   436→                Name
   437→   437→              </div>
   438→   438→              <button
