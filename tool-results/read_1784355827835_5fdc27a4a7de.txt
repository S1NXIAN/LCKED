    75→function subtitle(item: VaultItem): string {
    76→  switch (item.type) {
    77→    case "login":
    78→      return item.details.username || item.details.urls[0] || "—";
    79→    case "note":
    80→      return (item.details.content || "Empty note").replace(/\n/g, " ").trim();
    81→    case "card":
    82→      return (
    83→        item.details.brand ||
    84→        item.details.number.slice(-4).padStart(item.details.number.length, "•") ||
    85→        "—"
    86→      );
    87→    case "identity":
    88→      return [item.details.firstName, item.details.lastName].filter(Boolean).join(" ") || item.details.email || "—";
    89→  }

    92→/* ----------------------- ActiveRowHighlight (rAF spring) --------------- */
    93→
    94→/**
    95→ * Single persistent highlight element rendered ONCE inside the <ul>.
    96→ *
    97→ * Slides toward the active item's bounding box via a requestAnimationFrame
    98→ * spring (exponential lerp, factor 0.22). Uses the `animateRef` pattern —
    99→ * the latest step function is stashed in a ref so external listeners
   100→ * (scroll, MutationObserver) can always kick the loop without re-binding.
   101→ *
   102→ * On `activeId` change: measure + start the animation. If the item isn't
   103→ * found in the DOM, `setVisible(false)` and bail. A MutationObserver re-
   104→ * checks when the list DOM changes; a scroll listener re-measures on
   105→ * scroll so the highlight stays glued to the active row.
   106→ */
   107→function ActiveRowHighlight({
   108→  listRef,
   109→  activeId,
   110→}: {
   111→  listRef: React.RefObject<HTMLUListElement | null>;
   112→  activeId: string | null;
   113→}) {
   114→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   115→  const rafRef = React.useRef<number | null>(null);
   116→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   117→  const wasVisibleRef = React.useRef(false);
   118→  const [visible, setVisible] = React.useState(false);
   119→  // animateRef holds the latest step function so external listeners can
   120→  // kick it without re-binding their own dependencies.
   121→  const animateRef = React.useRef<(() => void) | null>(null);
   122→
   123→  // Measure the active item's rect relative to the <ul>. If not found,
   124→  // hide the highlight.
   125→  const measure = React.useCallback(() => {
   126→    const list = listRef.current;
   127→    if (!activeId || !list) {
   128→      targetRef.current = null;
   129→      setVisible(false);
   130→      return;
   131→    }
   132→    const el = list.querySelector<HTMLElement>(
   133→      `[data-item-id="${CSS.escape(activeId)}"]`,
   134→    );
   135→    if (!el) {
   136→      targetRef.current = null;
   137→      setVisible(false);
   138→      return;
   139→    }
   140→    const r = el.getBoundingClientRect();
   141→    const lr = list.getBoundingClientRect();
   142→    const next = {
   143→      x: r.left - lr.left,
   144→      y: r.top - lr.top,
   145→      w: r.width,
   146→      h: r.height,
   147→    };
   148→    // First activation (or returning from invisible): snap to target so
   149→    // we don't slide in from (0, 0).
   150→    if (!wasVisibleRef.current) {
   151→      const hl = highlightRef.current;
   152→      if (hl) {
   153→        hl.style.transform = `translate(${next.x}px, ${next.y}px)`;
   154→        hl.style.width = `${next.w}px`;
   155→        hl.style.height = `${next.h}px`;
   156→      }
   157→    }
   158→    targetRef.current = next;
   159→    setVisible(true);
   160→  }, [activeId, listRef]);
   161→
   162→  // (Re)define the animation step on every render so it captures the
   163→  // latest refs. Store it in animateRef for external listeners.
   164→  React.useEffect(() => {
   165→    const step = () => {
   166→      const hl = highlightRef.current;
   167→      const target = targetRef.current;
   168→      const list = listRef.current;
   169→      if (!hl || !target || !list) {
   170→        rafRef.current = null;
   171→        return;
   172→      }
   173→      const cur = hl.getBoundingClientRect();
   174→      const lr = list.getBoundingClientRect();
   175→      const cx = cur.left - lr.left;
   176→      const cy = cur.top - lr.top;
   177→      const FACTOR = 0.22;
   178→      const nx = cx + (target.x - cx) * FACTOR;
   179→      const ny = cy + (target.y - cy) * FACTOR;
   180→      const nw = cur.width + (target.w - cur.width) * FACTOR;
   181→      const nh = cur.height + (target.h - cur.height) * FACTOR;
   182→      const settled =
   183→        Math.abs(target.x - nx) < 0.5 &&
   184→        Math.abs(target.y - ny) < 0.5 &&
   185→        Math.abs(target.w - nw) < 0.5 &&
   186→        Math.abs(target.h - nh) < 0.5;
   187→      const fx = settled ? target.x : nx;
   188→      const fy = settled ? target.y : ny;
   189→      const fw = settled ? target.w : nw;
   190→      const fh = settled ? target.h : nh;
   191→      hl.style.transform = `translate(${fx}px, ${fy}px)`;
   192→      hl.style.width = `${fw}px`;
   193→      hl.style.height = `${fh}px`;
   194→      if (settled) {
   195→        rafRef.current = null;
   196→        return;
   197→      }
   198→      rafRef.current = requestAnimationFrame(step);
   199→    };
   200→    animateRef.current = step;
   201→    return () => {
   202→      animateRef.current = null;
   203→    };
   204→  }, [listRef]);
   205→
   206→  // Track visibility transitions for the snap-on-first-show behavior.
   207→  React.useEffect(() => {
   208→    wasVisibleRef.current = visible;
   209→  }, [visible]);
   210→
   211→  // Re-measure + kick the animation if it isn't already running.
   212→  const kick = React.useCallback(() => {
   213→    measure();
   214→    if (!rafRef.current && animateRef.current) {
   215→      rafRef.current = requestAnimationFrame(animateRef.current);
   216→    }
   217→  }, [measure]);
   218→
   219→  // On activeId change: measure + start the animation. Cancel any in-
   220→  // flight loop on cleanup.
   221→  React.useEffect(() => {
   222→    kick();
   223→    return () => {
   224→      if (rafRef.current) {
   225→        cancelAnimationFrame(rafRef.current);
   226→        rafRef.current = null;
   227→      }
   228→    };
   229→  }, [activeId, kick]);
   230→
   231→  // MutationObserver — re-check when the list DOM changes (items added,
   232→  // removed, reordered by sort, attributes toggled). Uses a microtask delay
   233→  // so the DOM has time to settle after AnimatePresence exit animations.
   234→  React.useEffect(() => {
   235→    const list = listRef.current;
   236→    if (!list) return;
   237→    let pending = false;
   238→    const observer = new MutationObserver(() => {
   239→      if (pending) return;
   240→      pending = true;
   241→      requestAnimationFrame(() => {
   242→        pending = false;
   243→        kick();
   244→      });
   245→    });
   246→    observer.observe(list, {
   247→      childList: true,
   248→      subtree: true,
   249→      attributes: true,
   250→      attributeFilter: ["data-item-id"],
   251→    });
   252→    return () => observer.disconnect();
   253→  }, [listRef, kick]);
   254→
   255→  // Scroll listener — re-measure + re-animate on scroll so the highlight
   256→  // stays glued to the active row (both inside the scroll viewport and
   257→  // when the page itself scrolls).
   258→  React.useEffect(() => {
   259→    const list = listRef.current;
   260→    if (!list) return;
   261→    const container =
   262→      list.closest('[data-radix-scroll-area-viewport]') ??
   263→      list.parentElement ??
   264→      null;
   265→    const handler = () => kick();
   266→    container?.addEventListener("scroll", handler, { passive: true });
   267→    window.addEventListener("scroll", handler, { passive: true });
   268→    return () => {
   269→      container?.removeEventListener("scroll", handler);
   270→      window.removeEventListener("scroll", handler);
   271→    };
   272→  }, [listRef, kick]);
   273→
   274→  if (!visible) return null;
   275→  return (
   276→    <div
   277→      ref={highlightRef}
   278→      aria-hidden
   279→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   280→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   281→    />
   282→  );
   283→}

   287→export function ItemList({
   288→  filter,
   289→  setFilter,
   290→  activeVault: activeVaultProp,
   291→  onMobileBack,
   292→}: {
   293→  /** Type filter (All / Login / Note / Card / Identity). */
   294→  filter: FilterType;
   295→  setFilter: (f: FilterType) => void;
   296→  /** Active vault filter from the store — passed explicitly so the list re-
   297→   *  renders when the sidebar switches between All/Favorites/Trash/vault. */
   298→  activeVault?: string;
   299→  /** Mobile-only callback when the user taps Back from a Trash detail. */
   300→  onMobileBack?: () => void;
   301→}) {
   302→  // Store hooks — we read activeVault directly here as well so the list
   303→  // always re-renders even if a parent forgets to thread the prop through.
   304→  const items = useVault((s) => s.items);
   305→  const vaults = useVault((s) => s.vaults);
   306→  const selectedId = useVault((s) => s.selectedId);
   307→  const setSelected = useVault((s) => s.setSelected);
   308→  const searchQuery = useVault((s) => s.searchQuery);
   309→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   310→  const trashItem = useVault((s) => s.trashItem);
   311→  const restoreItem = useVault((s) => s.restoreItem);
   312→  const permanentlyDeleteItem = useVault((s) => s.permanentlyDeleteItem);
   313→  const moveItemToVault = useVault((s) => s.moveItemToVault);
   314→  const toggleFavorite = useVault((s) => s.toggleFavorite);
   315→  const sortFavoritesFirst = useVault((s) => s.settings.sortFavoritesFirst);
   316→
   317→  const activeVault = activeVaultProp ?? useVault.getState().activeVault;
   318→  const isTrashView = activeVault === "trash";
   319→
   320→  // Sort persistence — read the initial value from localStorage so the
   321→  // user's last sort preference survives reloads.
   322→  const [sort, setSortState] = React.useState<SortKey>(() => {
   323→    if (typeof window === "undefined") return "newest";
   324→    const stored = window.localStorage.getItem("lcked-sort") as SortKey | null;
   325→    return stored === "newest" || stored === "oldest" || stored === "alphabetical"
   326→      ? stored
   327→      : "newest";
   328→  });
   329→  const setSort = React.useCallback((next: SortKey) => {
   330→    setSortState(next);
   331→    try {
   332→      window.localStorage.setItem("lcked-sort", next);
   333→    } catch {
   334→      // localStorage may be denied (private mode, etc.) — best-effort.
   335→    }
   336→  }, []);
   337→
   338→  const [multiSelect, setMultiSelect] = React.useState(false);
   339→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
   340→  // Reset multi-select whenever we leave multi-select mode.
   341→  React.useEffect(() => {
   342→    if (!multiSelect) setSelectedIds(new Set());
   343→  }, [multiSelect]);
   344→
   345→  // Deferred search query — keeps typing responsive on large vaults by
   346→  // letting the filter/sort work happen at lower priority.
   347→  const deferredSearch = React.useDeferredValue(searchQuery);
   348→
   349→  // List ref for the ActiveRowHighlight.
   350→  const listRef = React.useRef<HTMLUListElement | null>(null);
   351→
   352→  /* --------------------------- filtering + sorting --------------------------- */
   353→  const filtered = React.useMemo(() => {
   354→    let list = items;
   355→    // Vault-level filter.
   356→    if (activeVault === "trash") list = list.filter((i) => i.trashed);
   357→    else if (activeVault === "favorites") list = list.filter((i) => !i.trashed && i.favorite);
   358→    else if (activeVault && activeVault !== "all")
   359→      list = list.filter((i) => !i.trashed && i.vaultId === activeVault);
   360→    else list = list.filter((i) => !i.trashed);
   361→    // Type filter (secondary, list-header dropdown).
   362→    if (filter !== "all" && typeof filter === "string") {
   363→      list = list.filter((i) => i.type === filter);
   364→    }
   365→    // Fuzzy search.
   366→    list = searchItems(list, deferredSearch);
   367→    // Sort.
   368→    const sorted = [...list];
   369→    sorted.sort((a, b) => {
   370→      // Pin ALWAYS pins to top (unconditional). The schema doesn't have a
   371→      // dedicated `pinned` flag yet — this branch is a placeholder so the
   372→      // behavior is correct the moment one is added.
   373→      const aPin = false;
   374→      const bPin = false;
   375→      // Favorite only when sortFavoritesFirst is on.
   376→      const aFav = sortFavoritesFirst && a.favorite;
   377→      const bFav = sortFavoritesFirst && b.favorite;
   378→      // Favorite takes priority over pin when both active.
   379→      if (aFav !== bFav) return aFav ? -1 : 1;
   380→      if (aPin !== bPin) return aPin ? -1 : 1;
   381→      // Primary sort.
   382→      if (sort === "newest") return b.updatedAt - a.updatedAt;
   383→      if (sort === "oldest") return a.updatedAt - b.updatedAt;
   384→      return a.name.localeCompare(b.name);
   385→    });
   386→    return sorted;
   387→  }, [items, activeVault, filter, deferredSearch, sort, sortFavoritesFirst]);
   388→
   389→  // Scroll-into-view when selection changes via keyboard.
   390→  React.useEffect(() => {
   391→    if (!selectedId) return;
   392→    const el = document.querySelector(`[data-item-id="${CSS.escape(selectedId)}"]`);
   393→    el?.scrollIntoView({ block: "nearest", behavior: "smooth" });
   394→  }, [selectedId]);
   395→
   396→  /* --------------------------- multi-select actions --------------------------- */
   397→  const toggleSelected = (id: string) => {
   398→    setSelectedIds((prev) => {
   399→      const next = new Set(prev);
   400→      if (next.has(id)) next.delete(id);
   401→      else next.add(id);
   402→      return next;
   403→    });
   404→  };
   405→  const selectAll = () => setSelectedIds(new Set(filtered.map((i) => i.id)));
   406→  const deselectAll = () => setSelectedIds(new Set());
   407→
   408→  const handleMultiTrash = async () => {
   409→    if (selectedIds.size === 0) return;
   410→    const ids = Array.from(selectedIds);
   411→    try {
   412→      await Promise.all(ids.map((id) => trashItem(id)));
   413→      toast.success(`Moved ${ids.length} to Trash`);
   414→      setMultiSelect(false);
   415→    } catch {
   416→      toast.error("Could not move items");
   417→    }
   418→  };
   419→  const handleMultiMove = async (vaultId: string | null) => {
   420→    if (selectedIds.size === 0) return;
   421→    const ids = Array.from(selectedIds);
   422→    try {
   423→      await Promise.all(ids.map((id) => moveItemToVault(id, vaultId)));
   424→      const label =
   425→        vaultId === null
   426→          ? "All Items"
   427→          : vaults.find((v) => v.id === vaultId)?.name ?? "vault";
   428→      toast.success(`Moved ${ids.length} to ${label}`);
   429→      setMultiSelect(false);
   430→    } catch {
   431→      toast.error("Could not move items");
   432→    }
   433→  };
   434→
   435→  /* ------------------------------- item helpers ------------------------------- */
   436→  // Helper used by the empty-area context menu (and any other caller that
   437→  // wants to spawn the editor pre-seeded with a type).
   438→  const createItem = React.useCallback((type: ItemType) => {
   439→    stashNewItemType(type);
   440→    setEditorOpen(true);
   441→  }, [setEditorOpen]);
   442→
   443→  const copyField = React.useCallback(
   444→    async (value: string | undefined, label: string) => {
   445→      if (!value) return;
   446→      try {
   447→        await copyWithAutoClear(value, label);
   448→        toast.success(`${label} copied`, { description: "Auto-clears in 30s" });
   449→      } catch {
   450→        toast.error("Clipboard access denied");
   451→      }
   452→    },
   453→    [],
   454→  );
   455→
   456→  /* ------------------------------- rendering ------------------------------- */
   457→  return (
   458→    <div className="flex h-full flex-col">
   459→      {/* Filter bar — type + sort + multi-select menu */}
   460→      <div className="border-b border-border bg-background px-3 py-2">
   461→        <div className="flex items-center gap-2">
   462→          <Select
   463→            value={typeof filter === "string" ? filter : "all"}
   464→            onValueChange={(v) => setFilter(v as FilterType)}
   465→          >
   466→            <SelectTrigger size="sm" className="h-8 w-[110px] shrink-0 border-border bg-muted/40 dark:bg-secondary/20">
   467→              <LayoutGrid className="h-3.5 w-3.5 text-muted-foreground" />
   468→              <SelectValue placeholder="Type" />
   469→            </SelectTrigger>
   470→            <SelectContent>
   471→              {TYPE_OPTIONS.map((opt) => (
   472→                <SelectItem key={opt.value} value={opt.value}>
   473→                  {opt.label}
   474→                </SelectItem>
   475→              ))}
   476→            </SelectContent>
   477→          </Select>
   478→          {/* Sort dropdown — outline + muted/40 + border-border to match
   479→              the type Select (fixes the visual mismatch from the old
   480→              `variant="secondary"` button). */}
   481→          <DropdownMenu>
   482→            <DropdownMenuTrigger asChild>
   483→              <Button
   484→                variant="outline"
   485→                size="sm"
   486→                className="h-8 shrink-0 gap-1.5 border-border bg-muted/40 px-2.5 hover:bg-muted/60 dark:bg-secondary/20"
   487→              >
   488→                <ArrowUpDown className="h-3.5 w-3.5 text-muted-foreground" />
   489→                <span className="text-xs">{SORT_OPTIONS.find((o) => o.value === sort)?.label ?? "Sort"}</span>
   490→                <ChevronDown className="h-3 w-3 opacity-50" />
   491→              </Button>
   492→            </DropdownMenuTrigger>
   493→            <DropdownMenuContent align="start" className="w-40">
   494→              {SORT_OPTIONS.map((opt) => (
   495→                <DropdownMenuItem key={opt.value} onClick={() => setSort(opt.value)} className="text-xs">
   496→                  {opt.label}
   497→                </DropdownMenuItem>
   498→              ))}
   499→            </DropdownMenuContent>
   500→          </DropdownMenu>
   501→
   502→          {/* 3-dots multi-select + select-all menu */}
   503→          <DropdownMenu>
   504→            <DropdownMenuTrigger asChild>
   505→              <Button
   506→                variant="ghost"
   507→                size="icon"
   508→                className="ml-auto h-8 w-8 shrink-0 text-muted-foreground hover:text-foreground"
   509→                aria-label="List actions"
   510→                aria-pressed={multiSelect}
   511→              >
   512→                <MoreVertical className="h-4 w-4" />
   513→              </Button>
   514→            </DropdownMenuTrigger>
   515→            <DropdownMenuContent align="end" className="w-44">
   516→              <DropdownMenuItem onClick={() => setMultiSelect((m) => !m)}>
   517→                {multiSelect ? "Exit multi-select" : "Multi-select"}
   518→              </DropdownMenuItem>
   519→              <DropdownMenuSeparator />
   520→              <DropdownMenuItem onClick={selectAll} disabled={filtered.length === 0}>
   521→                Select all
   522→              </DropdownMenuItem>
   523→              <DropdownMenuItem onClick={deselectAll} disabled={selectedIds.size === 0}>
   524→                Deselect all
   525→              </DropdownMenuItem>
   526→            </DropdownMenuContent>
   527→          </DropdownMenu>
   528→        </div>
   529→
   530→      </div>
   531→
   532→      {/* Multi-select action bar */}
   533→      <AnimatePresence>
   534→        {multiSelect && (
   535→          <motion.div
   536→            initial={{ height: 0, opacity: 0 }}
   537→            animate={{ height: "auto", opacity: 1 }}
   538→            exit={{ height: 0, opacity: 0 }}
   539→            className="overflow-hidden border-b border-border bg-muted/40"
   540→          >
   541→            <div className="flex items-center gap-2 px-3 py-2">
   542→              <Button size="sm" variant="ghost" className="h-7 gap-1.5 px-2" onClick={() => setMultiSelect(false)}>
   543→                <X className="h-3.5 w-3.5" />
   544→                Cancel
   545→              </Button>
   546→              {/* "N Item(s) selected" — fixed minWidth prevents layout
   547→                  shift as the count changes between 0/1/many. */}
   548→              <span
   549→                className="text-xs text-muted-foreground"
   550→                style={{ minWidth: 130, display: "inline-block" }}
   551→              >
   552→                {selectedIds.size} Item{selectedIds.size === 1 ? "" : "s"} selected
   553→              </span>

   556→                <DropdownMenu>
   557→                  <DropdownMenuTrigger asChild>
   558→                    <Button size="sm" variant="outline" className="h-7 gap-1.5 px-2" disabled={selectedIds.size === 0}>
   559→                      Actions
   560→                      <ChevronDown className="h-3 w-3 opacity-90" />
   561→                    </Button>
   562→                  </DropdownMenuTrigger>
   563→                  <DropdownMenuContent align="end" className="w-48">
   564→                    {/* Move to vault submenu */}
   565→                    <DropdownMenuSub>
   566→                      <DropdownMenuSubTrigger>
   567→                        <FolderInput className="h-3.5 w-3.5" />
   568→                        Move to vault
   569→                      </DropdownMenuSubTrigger>
   570→                      <DropdownMenuSubContent className="w-44">
   571→                        <DropdownMenuItem onSelect={() => handleMultiMove(null)}>
   572→                          All Items
   573→                        </DropdownMenuItem>
   574→                        {vaults.map((v) => (
   575→                          <DropdownMenuItem key={v.id} onSelect={() => handleMultiMove(v.id)}>
   576→                            {v.name}
   577→                          </DropdownMenuItem>
   578→                        ))}
   579→                      </DropdownMenuSubContent>
   580→                    </DropdownMenuSub>
   581→                    {isTrashView ? (
   582→                      <>
   583→                        <DropdownMenuSeparator />
   584→                        <DropdownMenuItem onSelect={handleMultiRestore}>
   585→                          <RotateCcw className="h-3.5 w-3.5" />
   586→                          Restore all
   587→                        </DropdownMenuItem>
   588→                        <DropdownMenuSeparator />
   589→                        <DropdownMenuItem variant="destructive" onSelect={handleMultiDelete}>
   590→                          <Trash2 className="h-3.5 w-3.5" />
   591→                          Delete permanently
   592→                        </DropdownMenuItem>
   593→                      </>
   594→                    ) : (
   595→                      <>
   596→                        <DropdownMenuSeparator />
   597→                        <DropdownMenuItem variant="destructive" onSelect={handleMultiTrash}>
   598→                          <Trash2 className="h-3.5 w-3.5" />
   599→                          Move to trash
   600→                        </DropdownMenuItem>
   601→                      </>
   602→                    )}
   603→                  </DropdownMenuContent>
   604→                </DropdownMenu>
   605→              </div>
   606→            </div>
   607→          </motion.div>
   608→        )}
   609→      </AnimatePresence>
   610→
   611→      {/* List — wrapped in a ContextMenu so right-clicking empty space
   612→          offers quick "New Login/Note/Card/Identity" actions. */}
   613→      <ContextMenu>
   614→        <ContextMenuTrigger asChild>
   615→          <div className="lcked-scroll min-h-0 flex-1 overflow-y-auto">
   616→            <div className="p-2">
   617→              {filtered.length === 0 ? (
   618→                <EmptyList
   619→                  hasItems={items.length > 0}
   620→                  isTrash={isTrashView}
   621→                  onCreate={() => createItem("login")}
   622→                />
   623→              ) : (
   624→                <LayoutGroup>
   625→                  <ul
   626→                    ref={listRef}
   627→                    className="relative space-y-0.5"
   628→                    role="listbox"
   629→                    aria-label="Vault items"
   630→                  >
   631→                    {/* Single persistent highlight — slides between active
   632→                        items via rAF spring (see ActiveRowHighlight). */}
   633→                    <ActiveRowHighlight
   634→                      listRef={listRef}
   635→                      activeId={multiSelect ? null : selectedId}
   636→                    />
   637→                    <AnimatePresence initial={false}>
   638→                      {filtered.map((item) => {
   639→                        const active = item.id === selectedId && !multiSelect;
   640→                        const checked = selectedIds.has(item.id);
   641→                        return (
   642→                          <motion.li
   643→                            key={item.id}
   644→                            layout
   645→                            initial={{ opacity: 0 }}
   646→                            animate={{ opacity: 1 }}
   647→                            exit={{ opacity: 0 }}
   648→                            transition={{ duration: 0.12 }}
   649→                            data-item-id={item.id}
   650→                            className="relative"
   651→                          >
   652→                            <ItemRow
   653→                              item={item}
   654→                              active={active}
   655→                              checked={checked}
   656→                              multiSelect={multiSelect}
   657→                              isTrashView={isTrashView}
   658→                              onPick={() =>
   659→                                multiSelect ? toggleSelected(item.id) : setSelected(item.id)
   660→                              }
   661→                              onToggleSelected={() => toggleSelected(item.id)}
   662→                              onRestore={() =>
   663→                                restoreItem(item.id).then(() =>
   664→                                  toast.success("Item restored"),
   665→                                )
   666→                              }
   667→                              onPermanentDelete={() =>
   668→                                permanentlyDeleteItem(item.id).then(() =>
   669→                                  toast.success("Permanently deleted"),
   670→                                )
   671→                              }
   672→                              onCopyField={copyField}
   673→                              onToggleFavorite={() => toggleFavorite(item.id)}
   674→                              onEdit={() => setEditorOpen(true, item.id)}
   675→                              onTrash={() =>
   676→                                trashItem(item.id).then(() =>
   677→                                  toast.success("Moved to Trash"),
   678→                                )
   679→                              }
   680→                            />
   681→                          </motion.li>
   682→                        );
   683→                      })}
   684→                    </AnimatePresence>
   685→                  </ul>
   686→                </LayoutGroup>
   687→              )}
   688→            </div>
   689→          </div>
   690→        </ContextMenuTrigger>
   691→        <ContextMenuContent>
   692→          <ContextMenuItem onSelect={() => createItem("login")}>
   693→            <KeyRound className="h-4 w-4 text-violet-400" />
   694→            New Login
   695→          </ContextMenuItem>
   696→          <ContextMenuItem onSelect={() => createItem("note")}>
   697→            <StickyNote className="h-4 w-4 text-amber-400" />
   698→            New Note
   699→          </ContextMenuItem>
   700→          <ContextMenuItem onSelect={() => createItem("card")}>
   701→            <CreditCard className="h-4 w-4 text-emerald-400" />
   702→            New Card
   703→          </ContextMenuItem>
   704→          <ContextMenuItem onSelect={() => createItem("identity")}>
   705→            <UserRound className="h-4 w-4 text-sky-400" />
   706→            New Identity
   707→          </ContextMenuItem>
   708→        </ContextMenuContent>
   709→      </ContextMenu>
   710→
   711→      {/* Mobile back hint (only in trash view, for screen-reader friendliness) */}
   712→      {isTrashView && onMobileBack && (
   713→        <button
   714→          onClick={onMobileBack}
   715→          className="sr-only focus:not-sr-only"
   716→        >
   717→          Back to All Items
   718→        </button>
   719→      )}
   720→    </div>
   721→  );

   726→/**
   727→ * The per-item row, extracted so it can carry its own ContextMenu without
   728→ * forcing the parent map to re-render every row when one is right-clicked.
   729→ */
   730→function ItemRow({
   731→  item,
   732→  active,
   733→  checked,
   734→  multiSelect,
   735→  isTrashView,
   736→  onPick,
   737→  onToggleSelected,
   738→  onRestore,
   739→  onPermanentDelete,
   740→  onCopyField,
   741→  onToggleFavorite,
   742→  onEdit,
   743→  onTrash,
   744→}: {
   745→  item: VaultItem;
   746→  active: boolean;
   747→  checked: boolean;
   748→  multiSelect: boolean;
   749→  isTrashView: boolean;
   750→  onPick: () => void;
   751→  onToggleSelected: () => void;
   752→  onRestore: () => void;
   753→  onPermanentDelete: () => void;
   754→  onCopyField: (value: string | undefined, label: string) => void;
   755→  onToggleFavorite: () => void;
   756→  onEdit: () => void;
   757→  onTrash: () => void;
   758→}) {
   759→  // Per-item context menu — wraps the button so left-click still picks the
   760→  // item, and right-click opens the context menu with copy/edit/etc.
   761→  return (
   762→    <ContextMenu>
   763→      <ContextMenuTrigger asChild>
   764→        <div
   765→          onClick={onPick}
   766→          role="option"
   767→          aria-selected={active || checked}
   768→          draggable={!multiSelect}
   769→          onDragStart={(e) => {
   770→            e.dataTransfer.setData("text/lcked-item", item.id);
   771→            e.dataTransfer.effectAllowed = "move";
   772→          }}
   773→          className={cn(
   774→            "group relative flex w-full cursor-pointer items-center gap-3 rounded-lg px-2.5 py-2 text-left transition-colors",
   775→            active ? "text-accent-foreground" : "hover:bg-muted/60",
   776→            checked && "bg-accent/60",
   777→          )}
   778→        >
   779→          {/* Multi-select checkbox OR favicon/type-icon */}
   780→          {multiSelect ? (
   781→            <span className="relative z-10 flex h-7 w-7 shrink-0 items-center justify-center">
   782→              {checked ? (
   783→                <CheckSquare className="h-5 w-5 text-primary" />
   784→              ) : (
   785→                <Square className="h-5 w-5 text-muted-foreground" />
   786→              )}
   787→            </span>
   788→          ) : item.type === "login" && item.details.urls[0] ? (
   789→            <FaviconIcon
   790→              url={item.details.urls[0]}
   791→              size={28}
   792→              className="relative z-10"
   793→              fallback={<ItemTypeIcon type={item.type} size="sm" className="relative z-10" />}
   794→            />
   795→          ) : (
   796→            <ItemTypeIcon type={item.type} size="sm" className="relative z-10" />
   797→          )}
   798→
   799→          <div className="relative z-10 min-w-0 flex-1">
   800→            <div className="flex items-center gap-1.5">
   801→              <span className={cn("truncate text-sm font-medium", item.trashed && "line-through opacity-60")}>
   802→                {item.name}
   803→              </span>
   804→              {item.favorite && !item.trashed && (
   805→                <Star className="h-3 w-3 shrink-0 fill-amber-400 text-amber-400" />
   806→              )}
   807→            </div>
   808→            <div className="truncate text-xs text-muted-foreground">
   809→              {subtitle(item)}
   810→            </div>
   811→          </div>
   812→
   813→          {/* Type badge OR trash hover actions */}
   814→          {isTrashView ? (
   815→            <div className="relative z-10 flex shrink-0 items-center gap-0.5 opacity-0 transition-opacity group-hover:opacity-100 focus-within:opacity-100">
   816→              <Button
   817→                size="icon"
   818→                variant="ghost"
   819→                className="h-7 w-7 text-muted-foreground hover:text-foreground"
   820→                onClick={(e) => {
   821→                  e.stopPropagation();
   822→                  onRestore();
   823→                }}
   824→                aria-label="Restore"
   825→                title="Restore"
   826→              >
   827→                <RotateCcw className="h-3.5 w-3.5" />
   828→              </Button>
   829→              <Button
   830→                size="icon"
   831→                variant="ghost"
   832→                className="h-7 w-7 text-muted-foreground hover:text-red-400"
   833→                onClick={(e) => {
   834→                  e.stopPropagation();
   835→                  onPermanentDelete();
   836→                }}
   837→                aria-label="Delete permanently"
   838→                title="Delete permanently"
   839→              >
   840→                <Trash2 className="h-3.5 w-3.5" />
   841→              </Button>
   842→            </div>
   843→          ) : null}
   844→        </div>
   845→      </ContextMenuTrigger>
   846→      <ContextMenuContent>
   847→        {item.type === "login" && (
   848→          <>
   849→            <ContextMenuItem
   850→              onSelect={() => onCopyField(item.details.username, isEmail(item.details.username) ? "Email" : "Username")}
   851→              disabled={!item.details.username}
   852→            >
   853→              {isEmail(item.details.username) ? "Copy email" : "Copy username"}
   854→            </ContextMenuItem>
   855→            {item.details.password && (
   856→              <ContextMenuItem onSelect={() => onCopyField(item.details.password, "Password")}>
   857→                Copy password
   858→              </ContextMenuItem>
   859→            )}
   860→            {item.details.urls[0] && (
   861→              <ContextMenuItem onSelect={() => onCopyField(item.details.urls[0], "URL")}>
   862→                Copy URL
   863→              </ContextMenuItem>
   864→            )}
   865→            <ContextMenuSeparator />
   866→          </>
   867→        )}
   868→        {!isTrashView && (
   869→          <ContextMenuItem onSelect={onToggleFavorite}>
   870→            {item.favorite ? "Unfavorite" : "Favorite"}
   871→          </ContextMenuItem>
   872→        )}
   873→        <ContextMenuItem onSelect={onEdit}>Edit</ContextMenuItem>
   874→        {!isTrashView && (
   875→          <>
   876→            <ContextMenuSeparator />
   877→            <ContextMenuItem variant="destructive" onSelect={onTrash}>
   878→              Move to trash
   879→            </ContextMenuItem>
   880→          </>
   881→        )}
   882→      </ContextMenuContent>
   883→    </ContextMenu>
   884→  );
   885→}
