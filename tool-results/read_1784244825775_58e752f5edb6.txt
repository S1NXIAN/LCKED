     1→/**
     2→ * LCKED — Vault store (Zustand)
     3→ * ---------------------------------------------------------------------------
     4→ * Owns the auth state machine + every vault mutation. CryptoKey objects live
     5→ * only in memory here — they are NEVER persisted. All async actions are
     6→ * fault-tolerant: a failed IndexedDB write reverts the optimistic UI change.
     7→ */
     8→
     9→import { create } from "zustand";
    10→import { persist } from "zustand/middleware";
    11→import {
    12→  buildVerifier,
    13→  bytesToBase64,
    14→  checkVerifier,
    15→  decryptJson,
    16→  deriveMasterKey,
    17→  encryptJson,
    18→  generateVaultKey,
    19→  PBKDF2_ITERATIONS,
    20→  randomBytes,
    21→  randomId,
    22→  unwrapVaultKey,
    23→  wrapVaultKey,
    24→} from "@/lib/crypto";
    25→import {
    26→  deleteStoredItem,
    27→  loadAllStoredItems,
    28→  loadVaultMeta,
    29→  putStoredItem,
    30→  saveVaultMeta,
    31→  vaultExists,
    32→  wipeVault,
    33→} from "@/lib/vault-db";
    34→import {
    35→  DEFAULT_VAULT_SETTINGS,
    36→  type GeneratorOptions,
    37→  type NewItemInput,
    38→  type VaultDef,
    39→  type VaultItem,
    40→  type VaultMeta,
    41→  type VaultSettings,
    42→} from "@/lib/types";
    43→import {
    44→  exportToCsv,
    45→  importFromText,
    46→  type ImportResult,
    47→  type LckedExport,
    48→} from "@/lib/import-export";
    49→
    50→export type VaultStatus = "loading" | "setup" | "locked" | "unlocked";
    51→
    52→interface VaultState {
    53→  status: VaultStatus;
    54→  items: VaultItem[];
    55→  /** User-defined vaults (colored containers, Proton Pass–style). */
    56→  vaults: VaultDef[];
    57→  /** Active vault filter: "all" | "trash" | vault id. */
    58→  activeVault: string;
    59→  /** Vault editor (create/rename/delete vault) dialog state. */
    60→  vaultEditorOpen: boolean;
    61→  editingVaultId: string | null;
    62→  createVaultDialogOpen: boolean;
    63→  settings: VaultSettings;
    64→  /** UI-only selection / filter state (not persisted). */
    65→  selectedId: string | null;
    66→  searchQuery: string;
    67→  /** CryptoKey handles — held in memory only while unlocked. */
    68→  masterKey: CryptoKey | null;
    69→  vaultKey: CryptoKey | null;
    70→  /** Whether the active editor is open (and which item, if editing). */
    71→  editorOpen: boolean;
    72→  editorItemId: string | null;
    73→  generatorOpen: boolean;
    74→  importExportOpen: boolean;
    75→  settingsOpen: boolean;
    76→  commandOpen: boolean;
    77→
    78→  // lifecycle
    79→  init: () => Promise<void>;
    80→  setupVault: (masterPassword: string) => Promise<void>;
    81→  unlock: (masterPassword: string) => Promise<boolean>;
    82→  lock: () => void;
    83→  resetVault: () => Promise<void>;
    84→
    85→  // item CRUD
    86→  saveItem: (input: NewItemInput, existingId?: string) => Promise<VaultItem>;
    87→  /** Soft-delete — moves the item to Trash with a 30-day TTL. */
    88→  trashItem: (id: string) => Promise<void>;
    89→  restoreItem: (id: string) => Promise<void>;
    90→  permanentlyDeleteItem: (id: string) => Promise<void>;
    91→  emptyTrash: () => Promise<void>;
    92→  moveItemToVault: (itemId: string, vaultId: string | null) => Promise<void>;
    93→  toggleFavorite: (id: string) => Promise<void>;
    94→  duplicateItem: (id: string) => Promise<void>;
    95→  importItems: (filename: string, text: string) => Promise<ImportResult>;
    96→
    97→  // vault (custom containers) CRUD
    98→  createVault: (name: string, color: string, icon: string) => Promise<VaultDef>;
    99→  deleteVault: (id: string) => Promise<void>;
   100→  renameVault: (id: string, name: string) => Promise<void>;
   101→  updateVault: (id: string, patch: Partial<Omit<VaultDef, "id" | "createdAt">>) => Promise<void>;
   102→  setActiveVault: (v: string) => void;
   103→  setVaultEditorOpen: (open: boolean, vaultId?: string | null) => void;
   104→  setCreateVaultDialogOpen: (open: boolean) => void;
   105→
   106→  // settings
   107→  updateSettings: (patch: Partial<VaultSettings>) => Promise<void>;
   108→  updateGenerator: (patch: Partial<GeneratorOptions>) => void;
   109→  changeMasterPassword: (current: string, next: string) => Promise<boolean>;
   110→
   111→  // export
   112→  exportEncrypted: (password: string) => Promise<string>;
   113→  exportCsv: () => string;
   114→
   115→  // UI helpers
   116→  setSelected: (id: string | null) => void;
   117→  setSearch: (q: string) => void;
   118→  setEditorOpen: (open: boolean, itemId?: string | null) => void;
   119→  setGeneratorOpen: (open: boolean) => void;
   120→  setImportExportOpen: (open: boolean) => void;
   121→  setSettingsOpen: (open: boolean) => void;
   122→  setCommandOpen: (open: boolean) => void;
   123→}
   124→
   125→export const useVault = create<VaultState>()(
   126→  persist(
   127→    (set, get) => ({
   128→      status: "loading",
   129→      items: [],
   130→      vaults: [],
   131→      activeVault: "all",
   132→      vaultEditorOpen: false,
   133→      editingVaultId: null,
   134→      createVaultDialogOpen: false,
   135→      settings: DEFAULT_VAULT_SETTINGS,
   136→      selectedId: null,
   137→      searchQuery: "",
   138→      masterKey: null,
   139→      vaultKey: null,
   140→      editorOpen: false,
   141→      editorItemId: null,
   142→      generatorOpen: false,
   143→      importExportOpen: false,
   144→      settingsOpen: false,
   145→      commandOpen: false,
   146→
   147→      /* ------------------------------ lifecycle ------------------------------ */
   148→
   149→      init: async () => {
   150→        try {
   151→          if (typeof window === "undefined") return;
   152→          const exists = await vaultExists();
   153→          set({ status: exists ? "locked" : "setup" });
   154→        } catch (err) {
   155→          console.error("init failed", err);
   156→          set({ status: "setup" });
   157→        }
   158→      },
   159→
   160→      setupVault: async (masterPassword) => {
   161→        const salt = bytesToBase64(randomBytes(16));
   162→        const masterKey = await deriveMasterKey(masterPassword, salt, PBKDF2_ITERATIONS);
   163→        const vaultKey = await generateVaultKey();
   164→        const { ciphertext, iv } = await wrapVaultKey(vaultKey, masterKey);
   165→        const verifier = await buildVerifier(masterKey);
   166→
   167→        const meta: VaultMeta = {
   168→          id: "singleton",
   169→          salt,
   170→          iterations: PBKDF2_ITERATIONS,
   171→          encryptedVaultKey: ciphertext,
   172→          vaultKeyIv: iv,
   173→          verifier: verifier.verifier,
   174→          verifierIv: verifier.verifierIv,
   175→          verifierToken: verifier.verifierToken,
   176→          createdAt: Date.now(),
   177→          settings: DEFAULT_VAULT_SETTINGS,
   178→          vaults: [],
   179→        };
   180→        await saveVaultMeta(meta);
   181→
   182→        set({
   183→          status: "unlocked",
   184→          masterKey,
   185→          vaultKey,
   186→          settings: DEFAULT_VAULT_SETTINGS,
   187→          items: [],
   188→          vaults: [],
   189→          activeVault: "all",
   190→          selectedId: null,
   191→        });
   192→      },
   193→
   194→      unlock: async (masterPassword) => {
   195→        const meta = await loadVaultMeta();
   196→        if (!meta) {
   197→          set({ status: "setup" });
   198→          return false;
   199→        }
   200→        const masterKey = await deriveMasterKey(
   201→          masterPassword,
   202→          meta.salt,
   203→          meta.iterations,
   204→        );
   205→        const ok = await checkVerifier(
   206→          masterKey,
   207→          meta.verifier,
   208→          meta.verifierIv,
   209→          meta.verifierToken,
   210→        );
   211→        if (!ok) return false;
   212→
   213→        const vaultKey = await unwrapVaultKey(
   214→          meta.encryptedVaultKey,
   215→          meta.vaultKeyIv,
   216→          masterKey,
   217→        );
   218→
   219→        // Decrypt every stored item.
   220→        const stored = await loadAllStoredItems();
   221→        const items: VaultItem[] = [];
   222→        const toReencrypt: VaultItem[] = [];
   223→        let needsMetaSave = false;
   224→        const TRASH_TTL_MS = 30 * 24 * 60 * 60 * 1000; // 30 days
   225→        const now = Date.now();
   226→        for (const s of stored) {
   227→          try {
   228→            const item = await decryptJson<VaultItem>(s.ciphertext, s.iv, vaultKey);
   229→            // Migration: ensure the new soft-delete + vaultId fields exist on
   230→            // items written by older LCKED versions. Mutated records are queued
   231→            // for a transparent re-encrypt + persist.
   232→            let migrated: VaultItem | null = null;
   233→            if (item.vaultId === undefined) {
   234→              item.vaultId = null;
   235→              migrated = item;
   236→            }
   237→            if (item.trashed === undefined) {
   238→              item.trashed = false;
   239→              migrated = item;
   240→            }
   241→            if (item.trashedAt === undefined) {
   242→              item.trashedAt = null;
   243→              migrated = item;
   244→            }
   245→            // 30-day auto-delete: drop trashed items whose TTL has expired.
   246→            if (item.trashed && item.trashedAt && now - item.trashedAt > TRASH_TTL_MS) {
   247→              try {
   248→                await deleteStoredItem(item.id);
   249→              } catch {
   250→                // best-effort; keep going
   251→              }
   252→              continue;
   253→            }
   254→            if (migrated) toReencrypt.push(migrated);
   255→            items.push(item);
   256→          } catch {
   257→            // skip corrupt record but keep the vault usable
   258→            console.warn("Failed to decrypt item", s.id);
   259→          }
   260→        }
   261→        // Persist any migrated records so we never re-migrate them again.
   262→        if (toReencrypt.length > 0) {
   263→          await Promise.all(
   264→            toReencrypt.map(async (it) => {
   265→              const { ciphertext, iv } = await encryptJson(it, vaultKey);
   266→              await putStoredItem({
   267→                id: it.id,
   268→                type: it.type,
   269→                ciphertext,
   270→                iv,
   271→                createdAt: it.createdAt,
   272→                updatedAt: it.updatedAt,
   273→              });
   274→            }),
   275→          );
   276→        }
   277→        items.sort((a, b) => b.updatedAt - a.updatedAt);
   278→
   279→        // Hydrate vaults (custom containers) — migrate older meta that lacks
   280→        // the field by persisting an empty array.
   281→        const vaults: VaultDef[] = Array.isArray(meta.vaults) ? meta.vaults : [];
   282→        if (!Array.isArray(meta.vaults)) {
   283→          needsMetaSave = true;
   284→        }
   285→        if (needsMetaSave) {
   286→          await saveVaultMeta({ ...meta, vaults });
   287→        }
   288→
   289→        set({
   290→          status: "unlocked",
   291→          masterKey,
   292→          vaultKey,
   293→          items,
   294→          vaults,
   295→          activeVault: "all",
   296→          settings: meta.settings,
   297→        });
   298→        return true;
   299→      },
   300→
   301→      lock: () => {
   302→        set({
   303→          status: "locked",
   304→          masterKey: null,
   305→          vaultKey: null,
   306→          items: [],
   307→          vaults: [],
   308→          activeVault: "all",
   309→          selectedId: null,
   310→          searchQuery: "",
   311→          editorOpen: false,
   312→          generatorOpen: false,
   313→          settingsOpen: false,
   314→          commandOpen: false,
   315→          vaultEditorOpen: false,
   316→          editingVaultId: null,
   317→          createVaultDialogOpen: false,
   318→        });
   319→      },
   320→
   321→      resetVault: async () => {
   322→        await wipeVault();
   323→        set({
   324→          status: "setup",
   325→          items: [],
   326→          vaults: [],
   327→          activeVault: "all",
   328→          masterKey: null,
   329→          vaultKey: null,
   330→          selectedId: null,
   331→          settings: DEFAULT_VAULT_SETTINGS,
   332→          vaultEditorOpen: false,
   333→          editingVaultId: null,
   334→          createVaultDialogOpen: false,
   335→        });
   336→      },
   337→
   338→      /* ------------------------------- CRUD -------------------------------- */
   339→
   340→      saveItem: async (input, existingId) => {
   341→        const { vaultKey } = get();
   342→        if (!vaultKey) throw new Error("Vault is locked");
   343→
   344→        const now = Date.now();
   345→        const id = existingId ?? randomId();
   346→        const existing = existingId ? get().items.find((i) => i.id === existingId) : undefined;
   347→        // Never inherit trashed state from a duplicated item — duplicates land
   348→        // in the active view, not in Trash.
   349→        const baseTrashed = existing ? existing.trashed : false;
   350→        const baseTrashedAt = existing ? existing.trashedAt : null;
   351→        const baseVaultId = existing ? existing.vaultId : (input as VaultItem).vaultId ?? null;
   352→        const item: VaultItem = {
   353→          ...(input as VaultItem),
   354→          id,
   355→          createdAt: existing?.createdAt ?? now,
   356→          updatedAt: now,
   357→          vaultId: baseVaultId,
   358→          trashed: baseTrashed,
   359→          trashedAt: baseTrashedAt,
   360→        } as VaultItem;
   361→
   362→        const { ciphertext, iv } = await encryptJson(item, vaultKey);
   363→        await putStoredItem({ id, type: item.type, ciphertext, iv, createdAt: item.createdAt, updatedAt: now });
   364→
   365→        set((state) => {
   366→          const others = state.items.filter((i) => i.id !== id);
   367→          const next = [item, ...others].sort((a, b) => b.updatedAt - a.updatedAt);
   368→          return { items: next, selectedId: id };
   369→        });
   370→        return item;
   371→      },
   372→
   373→      /**
   374→       * Soft-delete: marks the item as trashed and stamps trashedAt. The record
   375→       * stays in IndexedDB (still encrypted) so it can be restored. Auto-purged
   376→       * 30 days later on the next unlock.
   377→       */
   378→      trashItem: async (id) => {
   379→        await updateItemFlags(id, { trashed: true, trashedAt: Date.now() }, get, set);
   380→      },
   381→
   382→      restoreItem: async (id) => {
   383→        await updateItemFlags(id, { trashed: false, trashedAt: null }, get, set);
   384→      },
   385→
   386→      permanentlyDeleteItem: async (id) => {
   387→        const prev = get().items;
   388→        set((state) => ({
   389→          items: state.items.filter((i) => i.id !== id),
   390→          selectedId: state.selectedId === id ? null : state.selectedId,
   391→        }));
   392→        try {
   393→          await deleteStoredItem(id);
   394→        } catch (err) {
   395→          console.error("permanent delete failed", err);
   396→          set({ items: prev });
   397→          throw err;
   398→        }
   399→      },
   400→
   401→      emptyTrash: async () => {
   402→        const trashed = get().items.filter((i) => i.trashed);
   403→        const prev = get().items;
   404→        // Optimistic: drop all trashed items from UI.
   405→        set((state) => ({
   406→          items: state.items.filter((i) => !i.trashed),
   407→          selectedId:
   408→            state.selectedId && trashed.some((t) => t.id === state.selectedId)
   409→              ? null
   410→              : state.selectedId,
   411→        }));
   412→        try {
   413→          await Promise.all(trashed.map((t) => deleteStoredItem(t.id)));
   414→        } catch (err) {
   415→          console.error("emptyTrash failed", err);
   416→          set({ items: prev });
   417→          throw err;
   418→        }
   419→      },
   420→
   421→      moveItemToVault: async (itemId, vaultId) => {
   422→        await updateItemFlags(itemId, { vaultId, updatedAt: Date.now() }, get, set);
   423→      },
   424→
   425→      toggleFavorite: async (id) => {
   426→        const item = get().items.find((i) => i.id === id);
   427→        if (!item) return;
   428→        const updated: VaultItem = {
   429→          ...item,
   430→          favorite: !item.favorite,
   431→          updatedAt: Date.now(),
   432→        } as VaultItem;
   433→        const { vaultKey } = get();
   434→        if (vaultKey) {
   435→          const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   436→          await putStoredItem({ id, type: updated.type, ciphertext, iv, createdAt: updated.createdAt, updatedAt: updated.updatedAt });
   437→        }
   438→        set((state) => ({
   439→          items: state.items
   440→            .map((i) => (i.id === id ? updated : i))
   441→            .sort((a, b) => b.updatedAt - a.updatedAt),
   442→        }));
   443→      },
   444→
   445→      duplicateItem: async (id) => {
   446→        const item = get().items.find((i) => i.id === id);
   447→        if (!item) return;
   448→        const { id: _id, createdAt: _c, updatedAt: _u, trashed: _t, trashedAt: _ta, ...rest } = item;
   449→        void _id; void _c; void _u; void _t; void _ta;
   450→        await get().saveItem(rest as NewItemInput);
   451→      },
   452→
   453→      importItems: async (filename, text) => {
   454→        const { vaultKey } = get();
   455→        if (!vaultKey) throw new Error("Vault is locked");
   456→        const { result, items } = importFromText(filename, text);
   457→
   458→        // Encrypted LCKED imports require the export's password — handled by a
   459→        // dedicated UI flow. Here we surface a clear hint instead of failing.
   460→        if ((result as any).__raw) {
   461→          return {
   462→            imported: 0,
   463→            skipped: 0,
   464→            warnings: ["Encrypted LCKED files need their own password — use the dedicated import path."],
   465→          };
   466→        }
   467→
   468→        let imported = 0;
   469→        for (const input of items) {
   470→          try {
   471→            await get().saveItem(input);
   472→            imported++;
   473→          } catch {
   474→            result.skipped++;
   475→          }
   476→        }
   477→        result.imported = imported;
   478→        return result;
   479→      },
   480→
   481→      /* --------------------------- vaults (containers) ---------------------- */
   482→
   483→      createVault: async (name, color, icon) => {
   484→        const meta = await loadVaultMeta();
   485→        if (!meta) throw new Error("Vault meta missing");
   486→        const vault: VaultDef = {
   487→          id: randomId(),
   488→          name: name.trim() || "Untitled vault",
   489→          color,
   490→          icon,
   491→          createdAt: Date.now(),
   492→        };
   493→        const vaults = [...(meta.vaults ?? []), vault];
   494→        await saveVaultMeta({ ...meta, vaults });
   495→        set({ vaults, createVaultDialogOpen: false, vaultEditorOpen: false, editingVaultId: null });
   496→        return vault;
   497→      },
   498→
   499→      deleteVault: async (id) => {
   500→        const meta = await loadVaultMeta();
   501→        if (!meta) return;
   502→        const vaults = (meta.vaults ?? []).filter((v) => v.id !== id);
   503→        await saveVaultMeta({ ...meta, vaults });
   504→        // Orphan any items that lived in this vault — they fall back to the
   505→        // default vault (vaultId = null). Re-encrypt + persist each.
   506→        const { vaultKey, items } = get();
   507→        const orphaned = items.filter((i) => i.vaultId === id);
   508→        if (vaultKey && orphaned.length > 0) {
   509→          await Promise.all(
   510→            orphaned.map(async (it) => {
   511→              const next: VaultItem = { ...it, vaultId: null, updatedAt: Date.now() } as VaultItem;
   512→              const { ciphertext, iv } = await encryptJson(next, vaultKey);
   513→              await putStoredItem({
   514→                id: next.id,
   515→                type: next.type,
   516→                ciphertext,
   517→                iv,
   518→                createdAt: next.createdAt,
   519→                updatedAt: next.updatedAt,
   520→              });
   521→            }),
   522→          );
   523→          set((state) => ({
   524→            items: state.items.map((i) =>
   525→              i.vaultId === id ? ({ ...i, vaultId: null, updatedAt: Date.now() } as VaultItem) : i,
   526→            ),
   527→          }));
   528→        }
   529→        set((state) => ({
   530→          vaults,
   531→          activeVault: state.activeVault === id ? "all" : state.activeVault,
   532→          vaultEditorOpen: state.editingVaultId === id ? false : state.vaultEditorOpen,
   533→          editingVaultId: state.editingVaultId === id ? null : state.editingVaultId,
   534→        }));
   535→      },
   536→
   537→      renameVault: async (id, name) => {
   538→        await get().updateVault(id, { name });
   539→      },
   540→
   541→      updateVault: async (id, patch) => {
   542→        const meta = await loadVaultMeta();
   543→        if (!meta) return;
   544→        const vaults = (meta.vaults ?? []).map((v) =>
   545→          v.id === id ? { ...v, ...patch, name: patch.name ?? v.name } : v,
   546→        );
   547→        await saveVaultMeta({ ...meta, vaults });
   548→        set({ vaults, vaultEditorOpen: false, editingVaultId: null });
   549→      },
   550→
   551→      setActiveVault: (v) => set({ activeVault: v }),
   552→      setVaultEditorOpen: (open, vaultId = null) =>
   553→        set({ vaultEditorOpen: open, editingVaultId: vaultId }),
   554→      setCreateVaultDialogOpen: (open) => set({ createVaultDialogOpen: open }),
   555→
   556→      /* ----------------------------- settings ------------------------------- */
   557→
   558→      updateSettings: async (patch) => {
   559→        const next = { ...get().settings, ...patch };
   560→        set({ settings: next });
   561→        const meta = await loadVaultMeta();
   562→        if (meta) {
   563→          await saveVaultMeta({ ...meta, settings: next });
   564→        }
   565→      },
   566→
   567→      updateGenerator: (patch) => {
   568→        set((state) => ({
   569→          settings: { ...state.settings, generator: { ...state.settings.generator, ...patch } },
   570→        }));
   571→      },
   572→
   573→      changeMasterPassword: async (current, next) => {
   574→        const meta = await loadVaultMeta();
   575→        if (!meta) return false;
   576→        const currentKey = await deriveMasterKey(current, meta.salt, meta.iterations);
   577→        const ok = await checkVerifier(currentKey, meta.verifier, meta.verifierIv, meta.verifierToken);
   578→        if (!ok) return false;
   579→
   580→        // New salt + new master key; re-wrap the SAME vault key so no item
   581→        // needs re-encryption.
   582→        const newSalt = bytesToBase64(randomBytes(16));
   583→        const newMasterKey = await deriveMasterKey(next, newSalt, PBKDF2_ITERATIONS);
   584→        const { vaultKey } = get();
   585→        if (!vaultKey) return false;
   586→        const wrapped = await wrapVaultKey(vaultKey, newMasterKey);
   587→        const verifier = await buildVerifier(newMasterKey);
   588→        await saveVaultMeta({

   663→/**
   664→ * Patch one or more flags on an item (vaultId / trashed / trashedAt / favorite)
   665→ * and re-encrypt + persist it. Used by trashItem / restoreItem / moveItemToVault
   666→ * so we have a single fault-tolerant code path.
   667→ */
   668→async function updateItemFlags(
   669→  id: string,
   670→  patch: Partial<Pick<VaultItem, "vaultId" | "trashed" | "trashedAt" | "updatedAt">>,
   671→  get: () => VaultState,
   672→  set: (partial: Partial<VaultState>) => void,
   673→): Promise<void> {
   674→  const item = get().items.find((i) => i.id === id);
   675→  if (!item) return;
   676→  const now = patch.updatedAt ?? Date.now();
   677→  const updated: VaultItem = { ...item, ...patch, updatedAt: now } as VaultItem;
   678→  const { vaultKey } = get();
   679→  if (vaultKey) {
   680→    const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   681→    await putStoredItem({
   682→      id,
   683→      type: updated.type,
   684→      ciphertext,
   685→      iv,
   686→      createdAt: updated.createdAt,
   687→      updatedAt: now,
   688→    });
   689→  }
   690→  set({
   691→    items: get()
   692→      .items.map((i) => (i.id === id ? updated : i))
   693→      .sort((a, b) => b.updatedAt - a.updatedAt),
   694→  });
   695→}

   697→/**
   698→ * Copy-to-clipboard with an automatic 30-second clear. Returns the timeout id
   699→ * so callers can cancel early if needed. Sensitive values never linger.
   700→ */
   701→const clipboardTimers = new Map<string, ReturnType<typeof setTimeout>>();
   702→
   703→export async function copyWithAutoClear(
   704→  value: string,
   705→  key = "default",
   706→  clearMs = 30_000,
   707→): Promise<void> {
   708→  if (typeof navigator === "undefined" || !navigator.clipboard?.writeText) {
   709→    throw new Error("Clipboard API unavailable");
   710→  }
   711→  await navigator.clipboard.writeText(value);
   712→  // Clear any prior timer for this key.
   713→  const prior = clipboardTimers.get(key);
   714→  if (prior) clearTimeout(prior);
   715→  const timer = setTimeout(async () => {
   716→    try {
   717→      const current = await navigator.clipboard.readText().catch(() => "");
   718→      if (current === value) {
   719→        await navigator.clipboard.writeText("");
   720→      }
   721→    } catch {
   722→      // readText may be denied — best-effort clear.
   723→    }
   724→    clipboardTimers.delete(key);
   725→  }, clearMs);
   726→  clipboardTimers.set(key, timer);
   727→}
   728→
   729→export function cancelClipboardClear(key = "default"): void {
   730→  const t = clipboardTimers.get(key);
   731→  if (t) {
   732→    clearTimeout(t);
   733→    clipboardTimers.delete(key);
   734→  }
   735→}
