     3→/**
     4→ * LCKED — VaultsSidebar
     5→ * ---------------------------------------------------------------------------
     6→ * Proton Pass–style vault selector. Renders the fixed rows (All / Favorites /
     7→ * Trash) plus every user-defined vault. Each row is a drag-and-drop drop
     8→ * target so an item dragged from the list can be moved between vaults.
     9→ *
    10→ * Pattern notes:
    11→ *   • Rows that host a ⋮ dropdown are rendered as <div role="button"> (NOT
    12→ *     <button>) so the trigger button doesn't nest inside another button —
    13→ *     nested buttons are invalid HTML and break accessibility tree.
    14→ *   • A fixed-width slot is reserved for the menu button on EVERY row (even
    15→ *     the menu-less ones) so all the right-aligned counts line up.
    16→ *   • "Rename" opens the vault editor sheet via setVaultEditorOpen(true, id);
    17→ *     "Delete" calls deleteVault (which orphan-rescues any items inside).
    18→ *   • Each row wrapper carries a `data-vault-key` attribute consumed by the
    19→ *     sliding VaultActiveHighlight (rAF spring, same pattern as item-list's
    20→ *     ActiveRowHighlight). The highlight lives in the OUTER container so it
    21→ *     can reach Trash (which sits outside the scroll area).
    22→ *   • CustomVaultRow is wrapped in <ContextMenu> with the same items as the
    23→ *     3-dots dropdown: Edit, Move all items (submenu), Hide vault, Delete.
    24→ */

    84→export interface VaultIconProps {
    85→  /** Vault icon id (see VAULT_ICONS). */
    86→  icon: string;
    87→  /** Vault color id (see VAULT_COLORS). */
    88→  color: string;
    89→  /** Pixel size of the square swatch. Default 28. */

    94→/**
    95→ * Render a vault's icon inside a colored rounded swatch. The fill is the
    96→ * vault color at 18% opacity; the icon glyph uses the full hex for contrast.
    97→ * Used by VaultsSidebar rows AND the create-vault-dialog picker preview.
    98→ * The Lucide-component lookup lives in `./vault-lucide-icons` so both this
    99→ * module and the picker share a single source of truth.
   100→ */
   101→export function VaultIcon({ icon, color, size = 28, className, bare = false }: VaultIconProps & { bare?: boolean }) {
   102→  const hex = vaultColorHex(color);
   103→  const Resolved = VAULT_LUCIDE_BY_ID[icon] ?? Home;
   104→  const px = `${size}px`;
   105→  const glyph = Math.round(size * 0.55);
   106→  if (bare) {
   107→    return (
   108→      <Resolved
   109→        size={glyph}
   110→        strokeWidth={2}
   111→        style={{ color: hex }}
   112→        className={cn("shrink-0", className)}
   113→        aria-hidden="true"
   114→      />
   115→    );
   116→  }
   117→  return (
   118→    <span
   119→      className={cn("inline-flex shrink-0 items-center justify-center rounded-lg", className)}
   120→      style={{
   121→        width: px,
   122→        height: px,
   123→        backgroundColor: `${hex}29`, // ~16% opacity
   124→        color: hex,
   125→      }}
   126→      aria-hidden="true"
   127→    >
   128→      <Resolved size={glyph} strokeWidth={2} />
   129→    </span>
   130→  );
   131→}

   138→interface VaultRowProps {
   139→  /** Glyph rendered in the swatch. Pass a Lucide component for fixed rows. */
   140→  icon: React.ReactNode;
   141→  /** Row label. */
   142→  label: string;
   143→  /** Item count shown on the right. */
   144→  count: number;
   145→  /** Whether this row is the active filter. */
   146→  active: boolean;
   147→  /** Click handler — sets the active vault filter. */
   148→  onSelect: () => void;
   149→  /** Optional drag-and-drop target vault id (null = no drop). */
   150→  dropVaultId?: string | null;
   151→  /** Optional amber-tinted style (used by Trash when non-empty). */
   152→  warn?: boolean;
   153→  /** Optional ⋮ menu node (rendered in the reserved slot). */
   154→  menu?: React.ReactNode;
   155→  /** Whether the row is being dragged over. */
   156→  dragOver?: boolean;
   157→}
   158→
   159→/**
   160→ * A single sidebar row. When `menu` is provided we render a div instead of a
   161→ * button to avoid invalid nested-button HTML. The reserved `menuSlot` keeps
   162→ * every row's right edge aligned whether or not a menu is present.
   163→ *
   164→ * The active state keeps `text-accent-foreground` but NO `bg-accent` — the
   165→ * sliding VaultActiveHighlight provides the background, so a static one
   166→ * would visually fight the spring.
   167→ */
   168→function VaultRow({
   169→  icon,
   170→  label,
   171→  count,
   172→  active,
   173→  onSelect,
   174→  dropVaultId,
   175→  warn,
   176→  menu,
   177→  dragOver,
   178→}: VaultRowProps) {
   179→  const Tag = menu ? "div" : "button";
   180→  const tagProps = menu
   181→    ? { role: "button", tabIndex: 0 }
   182→    : { type: "button" as const };
   183→
   184→  const handleDrop = (e: React.DragEvent) => {
   185→    if (dropVaultId === undefined) return;
   186→    e.preventDefault();
   187→    const id = e.dataTransfer.getData("text/lcked-item");
   188→    if (!id) return;
   189→    useVault.getState().moveItemToVault(id, dropVaultId ?? null);
   190→    toast.success(`Moved to ${label}`);
   191→  };
   192→  const handleDragOver = (e: React.DragEvent) => {
   193→    if (dropVaultId === undefined) return;
   194→    e.preventDefault();
   195→    e.dataTransfer.dropEffect = "move";
   196→  };
   197→  const handleKeyDown = (e: React.KeyboardEvent) => {
   198→    if (!menu) return;
   199→    if (e.key === "Enter" || e.key === " ") {
   200→      e.preventDefault();
   201→      onSelect();
   202→    }
   203→  };
   204→
   205→  return (
   206→    <Tag
   207→      {...tagProps}
   208→      onClick={(e: React.MouseEvent) => {
   209→        const target = e.target as HTMLElement;
   210→        if (target.closest("[data-menu-slot]")) return;
   211→        onSelect();
   212→      }}
   213→      onKeyDown={menu ? handleKeyDown : undefined}
   214→      onDrop={handleDrop}
   215→      onDragOver={handleDragOver}
   216→      aria-current={active ? "true" : undefined}
   217→      className={cn(
   218→        "group relative flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition-colors duration-100",
   219→        "focus:outline-none focus-visible:ring-2 focus-visible:ring-ring/60",
   220→        active
   221→          ? "text-accent-foreground"
   222→          : warn
   223→            ? "text-foreground hover:bg-amber-500/10"
   224→            : "text-foreground/80 hover:bg-muted/60 hover:text-foreground",
   225→        dragOver && "ring-2 ring-primary/60 ring-offset-1 ring-offset-background",
   226→      )}
   227→    >
   228→      <span className="flex h-7 w-7 shrink-0 items-center justify-center">{icon}</span>
   229→      <span className="min-w-0 flex-1">
   230→        <span className="block truncate text-sm font-medium leading-tight">{label}</span>
   231→        <span
   232→          className={cn(
   233→            "block text-xs leading-tight tabular-nums",
   234→            warn && !active ? "text-amber-400/80" : active ? "text-accent-foreground/70" : "text-muted-foreground",
   235→          )}
   236→        >
   237→          {count} {count === 1 ? "item" : "items"}
   238→        </span>
   239→      </span>
   240→      <span
   241→        data-menu-slot
   242→        className="flex h-7 w-7 shrink-0 items-center justify-center"
   243→      >
   244→        {menu}
   245→      </span>
   246→    </Tag>
   247→  );
   248→}

   479→/* --------------------- VaultActiveHighlight (rAF spring) --------------------- */
   480→
   481→/**
   482→ * Single persistent highlight rendered ONCE inside the sidebar outer
   483→ * container. Slides toward the active row's bounding box via a
   484→ * requestAnimationFrame spring (exponential lerp, factor 0.22). Uses the
   485→ * same `animateRef` pattern as item-list's ActiveRowHighlight.
   486→ *
   487→ * The highlight lives in the OUTER container (NOT the scroll area) so it can
   488→ * reach Trash, which sits below the scrollable vault list with `mt-auto`.
   489→ */
   490→function VaultActiveHighlight({
   491→  containerRef,
   492→  activeKey,
   493→}: {
   494→  containerRef: React.RefObject<HTMLDivElement | null>;
   495→  activeKey: string;
   496→}) {
   497→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   498→  const rafRef = React.useRef<number | null>(null);
   499→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   500→  const posRef = React.useRef({ x: 0, y: 0 });
   501→  const wasVisibleRef = React.useRef(false);
   502→  const [visible, setVisible] = React.useState(false);
   503→  const animateRef = React.useRef<(() => void) | null>(null);
   504→
   505→  const measure = React.useCallback(() => {
   506→    const container = containerRef.current;
   507→    if (!activeKey || !container) {
   508→      targetRef.current = null;
   509→      setVisible(false);
   510→      return;
   511→    }
   512→    const el = container.querySelector<HTMLElement>(
   513→      `[data-vault-key="${CSS.escape(activeKey)}"]`,
   514→    );
   515→    if (!el) {
   516→      targetRef.current = null;
   517→      setVisible(false);
   518→      return;
   519→    }
   520→    // Check if element is visible (not scrolled out of view)
   521→    const elRect = el.getBoundingClientRect();
   522→    const containerRect = container.getBoundingClientRect();
   523→    if (elRect.bottom < containerRect.top || elRect.top > containerRect.bottom) {
   524→      // Element is scrolled out of view — hide the indicator
   525→      setVisible(false);
   526→      return;
   527→    }
   528→    const next = {
   529→      x: elRect.left - containerRect.left,
   530→      y: elRect.top - containerRect.top,
   531→      w: elRect.width,
   532→      h: elRect.height,
   533→    };
   534→    if (highlightRef.current) {
   535→      highlightRef.current.style.width = `${next.w}px`;
   536→      highlightRef.current.style.height = `${next.h}px`;
   537→      if (!wasVisibleRef.current) {
   538→        highlightRef.current.style.transform = `translate(${next.x}px, ${next.y}px)`;
   539→        posRef.current = { x: next.x, y: next.y };
   540→      }
   541→    }
   542→    targetRef.current = next;
   543→    setVisible(true);
   544→  }, [activeKey, containerRef]);
   545→
   546→  React.useEffect(() => {
   547→    const step = () => {
   548→      const hl = highlightRef.current;
   549→      const target = targetRef.current;
   550→      if (!hl || !target) {
   551→        rafRef.current = null;
   552→        return;
   553→      }
   554→      const pos = posRef.current;
   555→      const FACTOR = 0.35;
   556→      const nx = pos.x + (target.x - pos.x) * FACTOR;
   557→      const ny = pos.y + (target.y - pos.y) * FACTOR;
   558→      const settled =
   559→        Math.abs(target.x - nx) < 0.5 && Math.abs(target.y - ny) < 0.5;
   560→      pos.x = settled ? target.x : nx;
   561→      pos.y = settled ? target.y : ny;
   562→      hl.style.transform = `translate(${pos.x}px, ${pos.y}px)`;
   563→      if (settled) {
   564→        rafRef.current = null;
   565→        return;
   566→      }
   567→      rafRef.current = requestAnimationFrame(step);
   568→    };
   569→    animateRef.current = step;
   570→    return () => { animateRef.current = null; };
   571→  }, []);
   572→
   573→  React.useEffect(() => {
   574→    wasVisibleRef.current = visible;
   575→  }, [visible]);
   576→
   577→  const kick = React.useCallback(() => {
   578→    measure();
   579→    if (!rafRef.current && animateRef.current) {
   580→      rafRef.current = requestAnimationFrame(animateRef.current);
   581→    }
   582→  }, [measure]);
   583→
   584→  React.useEffect(() => {
   585→    cancelAnimationFrame(rafRef.current ?? 0);
   586→    if (!activeKey) { setVisible(false); return; }
   587→    const timer = setTimeout(() => kick(), 16);
   588→    return () => { clearTimeout(timer); cancelAnimationFrame(rafRef.current ?? 0); };
   589→  }, [activeKey, kick]);
   590→
   591→  React.useEffect(() => {
   592→    const container = containerRef.current;
   593→    if (!container) return;
   594→    const scrollArea = container.querySelector(".lcked-scroll");
   595→    let pending = false;
   596→    const observer = new MutationObserver(() => {
   597→      if (pending) return;
   598→      pending = true;
   599→      requestAnimationFrame(() => { pending = false; kick(); });
   600→    });
   601→    observer.observe(container, { childList: true, subtree: true, attributes: true, attributeFilter: ["data-vault-key"] });
   602→    const handler = () => kick();
   603→    if (scrollArea) scrollArea.addEventListener("scroll", handler, { passive: true });
   604→    container.addEventListener("scroll", handler, { passive: true });
   605→    window.addEventListener("resize", handler, { passive: true });
   606→    return () => {
   607→      observer.disconnect();
   608→      if (scrollArea) scrollArea.removeEventListener("scroll", handler);
   609→      container.removeEventListener("scroll", handler);
   610→      window.removeEventListener("resize", handler);
   611→    };
   612→  }, [containerRef, kick]);
   613→
   614→  if (!visible) return null;
   615→  return (
   616→    <div
   617→      ref={highlightRef}
   618→      aria-hidden
   619→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   620→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   621→    />
   622→  );
   623→}

   632→  const setCreateVaultDialogOpen = useVault((s) => s.setCreateVaultDialogOpen);
   633→  const setVaultEditorOpen = useVault((s) => s.setVaultEditorOpen);
   634→  const deleteVault = useVault((s) => s.deleteVault);
   635→  const moveItemToVault = useVault((s) => s.moveItemToVault);
   636→  const hoverItemActions = useVault((s) => s.settings.hoverItemActions);
   637→
   638→  const containerRef = React.useRef<HTMLDivElement | null>(null);
   639→
   640→  const [overKey, setOverKey] = React.useState<string | null>(null);
   641→
   642→  const allCount = items.filter((i) => !i.trashed).length;
   643→  const favCount = items.filter((i) => !i.trashed && i.favorite).length;
   644→  const trashCount = items.filter((i) => i.trashed).length;
   645→  const vaultCount = (id: string) =>

   657→  const handleHide = (v: VaultDef) => {
   658→    try {
   659→      const hidden = JSON.parse(
   660→        window.localStorage.getItem("lcked-hidden-vaults") || "[]",
   661→      ) as string[];
   662→      if (!hidden.includes(v.id)) hidden.push(v.id);
   663→      window.localStorage.setItem("lcked-hidden-vaults", JSON.stringify(hidden));
   664→      toast.success(`Vault “${v.name}” hidden`, {
   665→        description: "Reload the app to apply.",
   666→      });
   667→    } catch {
   668→      toast.error("Could not hide vault");
   669→    }
   670→  };
   671→
   672→  const handleMoveAll = async (source: VaultDef, target: string | null) => {
   673→    const targets = items.filter((i) => !i.trashed && i.vaultId === source.id);
   674→    if (targets.length === 0) {
   675→      toast.info(`“${source.name}” has no items to move`);
   676→      return;
   677→    }
   678→    try {
   679→      await Promise.all(targets.map((i) => moveItemToVault(i.id, target)));
   680→      const label =
   681→        target === null
   682→          ? "All Items"
   683→          : vaults.find((v) => v.id === target)?.name ?? "target vault";
   684→      toast.success(`Moved ${targets.length} item${targets.length === 1 ? "" : "s"} to ${label}`);
   685→    } catch {
   686→      toast.error("Could not move items");
   687→    }
   688→  };

   690→  const renderCustomVault = (v: VaultDef) => (

   692→      key={v.id}
   693→      vault={v}
   694→      active={activeVault === v.id}
   695→      count={vaultCount(v.id)}
   696→      dragOver={overKey === `vault-${v.id}`}
   697→      hoverActions={hoverItemActions}
   698→      onSelect={() => setActiveVault(v.id)}
   699→      onRename={() => setVaultEditorOpen(true, v.id)}

   701→      onHide={() => handleHide(v)}
   702→      onMoveAll={(target) => handleMoveAll(v, target)}
   703→      onDragEnterRow={() => setOverKey(`vault-${v.id}`)}
   704→      onDragLeaveRow={(e) => {
   705→        const related = e.relatedTarget as Node | null;
   706→        if (related && (e.currentTarget as Node).contains(related)) return;
   707→        setOverKey((c) => (c === `vault-${v.id}` ? null : c));
   708→      }}
   709→      onDropClear={() => setOverKey(null)}
   710→    />
   711→  );
   712→
   713→  return (
   714→    <div
   715→      ref={containerRef}
   716→      className="relative flex h-full flex-col gap-1 px-2 py-2"
   717→    >
   718→      {/* Sliding highlight — lives in the OUTER container so it can reach
   719→          Trash, which sits below the scroll area with `mt-auto`. */}
   720→      <VaultActiveHighlight containerRef={containerRef} activeKey={activeVault} />
   721→
   722→      {/* Header */}

   724→        <span className="text-[10px] font-semibold uppercase tracking-[0.18em] text-muted-foreground">
   725→          Vaults
   726→        </span>
   727→        <Button
   728→          size="icon"
   729→          variant="ghost"
   730→          className="h-6 w-6 text-muted-foreground hover:text-foreground"
   731→          onClick={() => setCreateVaultDialogOpen(true)}
   732→          aria-label="Create vault"
   733→        >
   734→          <Plus className="h-4 w-4" />
   735→        </Button>
   736→      </div>
   737→

   740→      <div className="lcked-scroll relative min-h-0 flex-1 overflow-y-auto">
