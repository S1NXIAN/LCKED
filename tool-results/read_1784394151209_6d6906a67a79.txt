    15→ *   • Body: scrollable form. The Name field uses the flat borderless input
    16→ *     pattern from create-vault-dialog; the remaining fields use labelled
    17→ *     Inputs grouped into FieldCluster cards (matching item-detail's
    18→ *     FieldRow presentation). All type-specific fields (login/note/card/
    19→ *     identity), URLs, TOTP, notes, custom fields are preserved.
    20→ *   • Footer: a single Cancel button (left). Save/Create lives in the header.
    21→ *
    22→ * All existing functionality is preserved: blankItem(), itemToInput(),
    23→ * saveItem(), custom-field add/update/remove, login URL list helpers,
    24→ * favourite toggle, folder, and card-brand detection.

    68→function blankItem(type: ItemType): NewItemInput {
    69→  const base = {
    70→    name: "",
    71→    favorite: false,
    72→    folder: "",
    73→    customFields: [],
    74→    createdAt: Date.now(),
    75→    updatedAt: Date.now(),
    76→    vaultId: null as string | null,
    77→    trashed: false,
    78→    trashedAt: null as number | null,
    79→  };
    80→  if (type === "login") {
    81→    return { ...base, type, details: { username: "", password: "", urls: [""], totp: "", notes: "" } };
    82→  }
    83→  if (type === "note") {
    84→    return { ...base, type, details: { content: "" } };
    85→  }
    86→  if (type === "card") {
    87→    return {
    88→      ...base,
    89→      type,
    90→      details: { cardholder: "", number: "", brand: "", cvv: "", expiry: "", pin: "", notes: "" },
    91→    };
    92→  }
    93→  return {
    94→    ...base,
    95→    type,
    96→    details: {
    97→      firstName: "", lastName: "", email: "", phone: "", company: "",
    98→      address1: "", address2: "", city: "", state: "", zip: "", country: "", notes: "",
    99→    },
   100→  };
   101→}
   102→
   103→function itemToInput(item: VaultItem): NewItemInput {
   104→  const { id: _id, createdAt: _c, updatedAt: _u, ...rest } = item;
   105→  return rest as NewItemInput;
   106→}

   110→/** Bordered card grouping related editable rows, divided by 1px lines. */
   111→function FieldCluster({ children }: { children: React.ReactNode }) {
   112→  return (
   113→    <div className="overflow-hidden rounded-xl border border-border bg-secondary/10 dark:bg-secondary/10">
   114→      {children}
   115→    </div>
   116→  );
   117→}
   118→
   119→/**
   120→ * FieldCluster with a small section header above the card. Used for SECONDARY
   121→ * clusters (TOTP / Websites / Notes / Custom fields) to break up the visual
   122→ * monotony of a stack of identical cards — the primary credentials cluster
   123→ * stays bare for a cleaner look.
   124→ *
   125→ * The header is OUTSIDE the card (per spec), and may carry an optional
   126→ * `action` node on its right (e.g. the "Add" button for custom fields).
   127→ */
   128→function FieldClusterWithLabel({
   129→  label,
   130→  action,
   131→  children,
   132→}: {
   133→  label: string;
   134→  action?: React.ReactNode;
   135→  children: React.ReactNode;
   136→}) {
   137→  return (
   138→    <div>
   139→      <div className="mb-1.5 flex items-center justify-between px-1">
   140→        <span className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/60">
   141→          {label}
   142→        </span>
   143→        {action}
   144→      </div>
   145→      <FieldCluster>{children}</FieldCluster>
   146→    </div>
   147→  );
   148→}
   149→
   150→/** A labelled field row inside a FieldCluster. Flat borderless input.
   151→ *  `label` is optional — when omitted (e.g. inside a FieldClusterWithLabel
   152→ *  whose header already names the section), the row renders without the
   153→ *  small uppercase label so the section isn't visually redundant. */
   154→function FieldRowInput({
   155→  label,
   156→  first,
   157→  children,
   158→}: {
   159→  label?: string;
   160→  first?: boolean;
   161→  children: React.ReactNode;
   162→}) {
   163→  return (
   164→    <div className={cn("px-3.5 py-2", !first && "border-t border-border/50")}>
   165→      {label && (
   166→        <div className="mb-1 text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   167→          {label}
   168→        </div>
   169→      )}
   170→      {children}
   171→    </div>
   172→  );
   173→}

   187→export function ItemEditor() {
   188→  const open = useVault((s) => s.editorOpen);
   189→  const editorItemId = useVault((s) => s.editorItemId);
   190→  const items = useVault((s) => s.items);
   191→  const vaults = useVault((s) => s.vaults);
   192→  const settings = useVault((s) => s.settings);
   193→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   194→  const saveItem = useVault((s) => s.saveItem);
   195→
   196→  const [form, setForm] = React.useState<NewItemInput | null>(null);
   197→  const [busy, setBusy] = React.useState(false);
   198→
   199→  // Initialise form when the sheet opens.
   200→  React.useEffect(() => {
   201→    if (!open) {
   202→      setForm(null);
   203→      return;
   204→    }
   205→    if (editorItemId) {
   206→      const existing = items.find((i) => i.id === editorItemId);
   207→      if (existing) {
   208→        setForm(itemToInput(existing));
   209→        return;
   210→      }
   211→    }
   212→    // Honour a type pre-selected via ⌘K / icon rail, then clear the stash.
   213→    const initialType = consumeNewItemType() ?? "login";
   214→    setForm(blankItem(initialType));
   215→  }, [open, editorItemId, items]);
   216→
   217→  const isEditing = Boolean(editorItemId);
   218→
   219→  if (!open || !form) return null;
   220→
   221→  const update = (patch: Partial<NewItemInput>) => setForm((f) => (f ? { ...f, ...patch } : f));
   222→  const updateDetails = (patch: Record<string, unknown>) =>
   223→    setForm((f) => (f ? { ...f, details: { ...(f.details as object), ...patch } } : f));
   224→
   225→  const handleSave = async () => {
   226→    if (!form) return;
   227→    if (!form.name.trim()) {
   228→      toast.error("Please give the item a name");
   229→      return;
   230→    }
   231→    setBusy(true);
   232→    try {
   233→      await saveItem(form, editorItemId ?? undefined);
   234→      toast.success(isEditing ? "Item updated" : "Item created");
   235→      setEditorOpen(false);
   236→    } catch (err) {
   237→      console.error(err);
   238→      toast.error("Could not save item");
   239→    } finally {
   240→      setBusy(false);
   241→    }
   242→  };
   243→
   244→  const addCustomField = () =>
   245→    update({ customFields: [...form.customFields, { name: "", value: "", type: "text" }] });
   246→  const updateCustomField = (idx: number, patch: Partial<CustomField>) =>
   247→    update({
   248→      customFields: form.customFields.map((cf, i) => (i === idx ? { ...cf, ...patch } : cf)),
   249→    });
   250→  const removeCustomField = (idx: number) =>
   251→    update({ customFields: form.customFields.filter((_, i) => i !== idx) });
   252→
   253→  // Login URL list helpers.
   254→  const urls = form.type === "login" ? form.details.urls : [];
   255→  const setUrl = (idx: number, val: string) => {
   256→    const next = urls.map((u, i) => (i === idx ? val : u));
   257→    updateDetails({ urls: next });
   258→  };
   259→  const addUrl = () => updateDetails({ urls: [...urls, ""] });
   260→  const removeUrl = (idx: number) =>
   261→    updateDetails({ urls: urls.filter((_, i) => i !== idx) });
   262→
   263→  // Selected vault (for the header dropdown).
   264→  const selectedVault = form.vaultId ? vaults.find((v) => v.id === form.vaultId) ?? null : null;
   265→
   266→  return (
   267→    <Sheet open={open} onOpenChange={(o) => setEditorOpen(o)}>
   268→      <SheetContent
   269→        side="right"
   270→        className="w-full gap-0 overflow-hidden border-l border-border bg-background p-0 sm:max-w-[480px] [&>button:last-child]:hidden"
   271→      >
   272→        {/* Header — type icon + title (left); vault selector + Save (right) */}
   273→        <SheetHeader className="flex-row items-center justify-between gap-2 border-b border-border px-4 py-3">
   274→          <SheetTitle className="flex min-w-0 items-center gap-2.5 text-base font-semibold">
   275→            <ItemTypeIcon type={form.type} size="sm" />
   276→            <span className="truncate">{isEditing ? "Edit item" : "New item"}</span>
   277→          </SheetTitle>
   278→          <div className="flex shrink-0 items-center gap-1.5">
   279→            {/* Vault selector dropdown */}
   280→            <DropdownMenu>
   281→              <DropdownMenuTrigger asChild>
   282→                <button
   283→                  type="button"
   284→                  className="flex h-8 items-center gap-1.5 rounded-md border border-border bg-background px-2 text-xs text-foreground transition-colors hover:bg-muted/60"
   285→                  aria-label="Select vault"
   286→                  title={selectedVault ? selectedVault.name : "No vault"}
   287→                >
   288→                  {selectedVault ? (
   289→                    <VaultIcon icon={selectedVault.icon} color={selectedVault.color} size={18} />
   290→                  ) : (
   291→                    <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   292→                      —
   293→                    </span>
   294→                  )}
   295→                  <span className="max-w-[80px] truncate">{selectedVault ? selectedVault.name : "No vault"}</span>
   296→                  <ChevronsUpDown className="h-3 w-3 text-muted-foreground" />
   297→                </button>
   298→              </DropdownMenuTrigger>
   299→              <DropdownMenuContent align="end" className="min-w-[180px]">
   300→                <DropdownMenuItem
   301→                  onSelect={() => update({ vaultId: null })}
   302→                  className={cn("gap-2", !selectedVault && "bg-muted/60")}
   303→                >
   304→                  <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   305→                    —
   306→                  </span>
   307→                  No vault
   308→                </DropdownMenuItem>
   309→                {vaults.map((v) => (
   310→                  <DropdownMenuItem
   311→                    key={v.id}
   312→                    onSelect={() => update({ vaultId: v.id })}
   313→                    className={cn("gap-2", selectedVault?.id === v.id && "bg-muted/60")}
   314→                  >
   315→                    <VaultIcon icon={v.icon} color={v.color} size={18} />
   316→                    <span className="truncate">{v.name}</span>
   317→                    {selectedVault?.id === v.id && <Check className="ml-auto h-3.5 w-3.5" />}
   318→                  </DropdownMenuItem>
   319→                ))}
   320→              </DropdownMenuContent>
   321→            </DropdownMenu>
   322→
   323→            {/* Save / Create */}
   324→            <Button
   325→              type="button"
   326→              size="sm"
   327→              onClick={handleSave}
   328→              disabled={busy}
   329→              className="min-w-[72px]"
   330→            >
   331→              {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : isEditing ? "Save" : "Create"}
   332→            </Button>
   333→          </div>
   334→        </SheetHeader>
   335→        <SheetDescription className="sr-only">
   336→          {isEditing ? "Edit the selected vault item." : "Create a new vault item."}
   337→        </SheetDescription>
   338→
   339→        {/* Body — flat scrollable form */}
   340→        <div className="lcked-scroll flex-1 overflow-y-auto p-4">
   341→          {/* Type selector — only when creating */}
   342→          {!isEditing && (
   343→            <div className="mb-4 grid grid-cols-4 gap-2">
   344→              {(Object.keys(ITEM_TYPE_LABELS) as ItemType[]).map((t) => (
   345→                <button
   346→                  key={t}
   347→                  type="button"
   348→                  onClick={() => setForm(blankItem(t))}
   349→                  className={cn(
   350→                    "flex flex-col items-center gap-1.5 rounded-lg border p-3 text-xs transition-colors",
   351→                    form.type === t
   352→                      ? "border-primary bg-primary/10 text-foreground"
   353→                      : "border-border text-muted-foreground hover:bg-muted/50",
   354→                  )}
   355→                >
   356→                  <ItemTypeIcon type={t} size="sm" />
   357→                  {ITEM_TYPE_LABELS[t]}
   358→                </button>
   359→              ))}
   360→            </div>
   361→          )}
   362→
   363→          {/* Name + favorite star — inline to save space */}
   364→          <div className="mb-4">
   365→            <div className="mb-1 flex items-center justify-between">
   366→              <div className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   367→                Name
   368→              </div>
   369→              <button
   370→                type="button"
   371→                onClick={() => update({ favorite: !form.favorite })}
   372→                className="flex h-6 w-6 items-center justify-center rounded-md transition-colors hover:bg-muted"
   373→                aria-label={form.favorite ? "Remove favorite" : "Add favorite"}
   374→                title={form.favorite ? "Remove favorite" : "Add favorite"}
   375→              >
   376→                <Star className={cn("h-4 w-4 transition-colors", form.favorite ? "fill-amber-400 text-amber-400" : "text-muted-foreground")} />
   377→              </button>
   378→            </div>
   379→            <input
   380→              type="text"
   381→              value={form.name}
   382→              onChange={(e) => update({ name: e.target.value })}
   383→              placeholder="e.g. GitHub, Bank card, WiFi password"
   384→              autoFocus
   385→              className="w-full border-0 bg-transparent dark:bg-transparent px-0 py-0.5 text-base font-medium text-foreground placeholder:text-muted-foreground/60 focus:outline-none"
   386→              aria-label="Item name"
   387→            />
   388→          </div>
   389→
   390→          {/* Type-specific */}
   391→          {form.type === "login" && (
   392→            <div className="space-y-4">
   393→              {/* Username + Password */}
   394→              <FieldCluster>
   395→                <FieldRowInput label={isEmail(form.details.username) ? "Email" : "Username"} first>
   396→                  <Input
   397→                    value={form.details.username}
   398→                    onChange={(e) => updateDetails({ username: e.target.value })}
   399→                    placeholder="you@example.com"
   400→                    autoComplete="off"
   401→                    className={flatInputCls}
   402→                  />
   403→                </FieldRowInput>
   404→                <FieldRowInput label="Password">
   405→                  <PasswordField
   406→                    value={form.details.password}
   407→                    onChange={(v) => updateDetails({ password: v })}
   408→                    placeholder="Enter password"
   409→                    showStrength
   410→                    showGenerate
   411→                    generatorOptions={settings.generator}
   412→                    inputClassName={flatPasswordInputCls}
   413→                  />
   414→                </FieldRowInput>
   415→              </FieldCluster>
   416→
   417→              {/* TOTP */}
   418→              <FieldClusterWithLabel label="Verification">
   419→                <FieldRowInput first>
   420→                  <Input
   421→                    value={form.details.totp}
   422→                    onChange={(e) => updateDetails({ totp: e.target.value })}
   423→                    placeholder="Base32 secret or otpauth:// URI"
   424→                    className={cn(flatInputCls, "font-secret")}
   425→                    autoComplete="off"
   426→                  />
   427→                </FieldRowInput>
   428→              </FieldClusterWithLabel>
   429→
   430→              {/* URLs — wrapped in a labelled card cluster (Websites) with the
   431→                  "Add URL" action in the header. Matches the TOTP / Notes /
   432→                  Custom fields pattern so every secondary section reads as a
   433→                  consistent card. Each URL is a divided row inside the card. */}
   434→              <FieldClusterWithLabel
   435→                label="Websites"
   436→                action={
   437→                  <Button type="button" variant="ghost" size="sm" onClick={addUrl}>
   438→                    <Plus className="mr-1.5 h-3.5 w-3.5" />
   439→                    Add URL
   440→                  </Button>
   441→                }
   442→              >
   443→                {urls.map((url, idx) => (
   444→                  <div
   445→                    key={idx}
   446→                    className={cn(
   447→                      "flex items-center gap-2 px-3.5 py-2",
   448→                      idx !== 0 && "border-t border-border/50",
   449→                    )}
   450→                  >
   451→                    <Link2 className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
   452→                    <Input
   453→                      value={url}
   454→                      onChange={(e) => setUrl(idx, e.target.value)}
   455→                      placeholder="https://example.com"
   456→                      className={cn(flatInputCls, "font-secret")}
   457→                      autoComplete="off"
   458→                    />
   459→                    {urls.length > 1 && (
   460→                      <Button
   461→                        type="button"
   462→                        size="icon"
   463→                        variant="ghost"
   464→                        className="h-7 w-7 shrink-0 text-muted-foreground hover:text-red-400"
   465→                        onClick={() => removeUrl(idx)}
   466→                        aria-label={`Remove URL ${idx + 1}`}
   467→                      >
   468→                        <Trash2 className="h-3.5 w-3.5" />
   469→                      </Button>
   470→                    )}
   471→                  </div>
   472→                ))}
   473→              </FieldClusterWithLabel>
   474→
   475→              {/* Notes */}
   476→              <FieldClusterWithLabel label="Notes">
   477→                <FieldRowInput first>
   478→                  <Textarea
   479→                    value={form.details.notes}
   480→                    onChange={(e) => updateDetails({ notes: e.target.value })}
   481→                    placeholder="Optional secure notes…"
   482→                    rows={3}
   483→                    className={flatTextareaCls}
   484→                  />
   485→                </FieldRowInput>
   486→              </FieldClusterWithLabel>
   487→            </div>
   488→          )}
   489→
   490→          {form.type === "note" && (
   491→            <FieldCluster>
   492→              <FieldRowInput label="Content" first>
   493→                <Textarea
   494→                  value={form.details.content}
   495→                  onChange={(e) => updateDetails({ content: e.target.value })}
   496→                  placeholder="Write your secure note here…"
   497→                  rows={10}
   498→                  className={flatTextareaCls}
   499→                />
   500→              </FieldRowInput>
   501→            </FieldCluster>
   502→          )}
   503→
   504→          {form.type === "card" && (
   505→            <div className="space-y-4">
   506→              {/* Cardholder + Card number + CVV + Expiry + PIN */}
   507→              <FieldCluster>
   508→                <FieldRowInput label="Cardholder name" first>
   509→                  <Input
   510→                    value={form.details.cardholder}
   511→                    onChange={(e) => updateDetails({ cardholder: e.target.value })}
   512→                    placeholder="Name on card"
   513→                    autoComplete="off"
   514→                    className={flatInputCls}
   515→                  />
   516→                </FieldRowInput>
   517→                <FieldRowInput label="Card number">
   518→                  <PasswordField
   519→                    value={form.details.number}
   520→                    onChange={(v) => {
   521→                      const brand = detectCardBrand(v);
   522→                      updateDetails({ number: v, brand: brand || form.details.brand });
   523→                    }}
   524→                    placeholder="0000 0000 0000 0000"
   525→                    showGenerate={false}
   526→                    inputClassName={flatPasswordInputCls}
   527→                  />
   528→                  {form.details.brand && (
   529→                    <p className="mt-1 text-xs text-muted-foreground">Detected: {form.details.brand}</p>
   530→                  )}
   531→                </FieldRowInput>
   532→                <div className="grid grid-cols-2 divide-x divide-border/50">
   533→                  <FieldRowInput label="CVV" first>
   534→                    <PasswordField
   535→                      value={form.details.cvv}
   536→                      onChange={(v) => updateDetails({ cvv: v })}
   537→                      placeholder="123"
   538→                      showGenerate={false}
   539→                      inputClassName={flatPasswordInputCls}
   540→                    />
   541→                  </FieldRowInput>
   542→                  <FieldRowInput label="Expiry (MM/YY)" first>
   543→                    <Input
   544→                      value={form.details.expiry}
   545→                      onChange={(e) => updateDetails({ expiry: e.target.value })}
   546→                      placeholder="08/27"
   547→                      className={cn(flatInputCls, "font-secret")}
   548→                      autoComplete="off"
   549→                    />
