     1→/**
     2→ * LCKED — Vault store (Zustand)
     3→ * ---------------------------------------------------------------------------
     4→ * Owns the auth state machine + every vault mutation. CryptoKey objects live
     5→ * only in memory here — they are NEVER persisted. All async actions are
     6→ * fault-tolerant: a failed IndexedDB write reverts the optimistic UI change.
     7→ */
     8→
     9→import { create } from "zustand";
    10→import { persist } from "zustand/middleware";
    11→import {
    12→  buildVerifier,
    13→  bytesToBase64,
    14→  checkVerifier,
    15→  decryptJson,
    16→  deriveMasterKey,
    17→  encryptJson,
    18→  generateVaultKey,
    19→  PBKDF2_ITERATIONS,
    20→  randomBytes,
    21→  randomId,
    22→  unwrapVaultKey,
    23→  wrapVaultKey,
    24→} from "@/lib/crypto";
    25→import {
    26→  deleteStoredItem,
    27→  loadAllStoredItems,
    28→  loadVaultMeta,
    29→  putStoredItem,
    30→  saveVaultMeta,
    31→  vaultExists,
    32→  wipeVault,
    33→} from "@/lib/vault-db";
    34→import {
    35→  DEFAULT_VAULT_SETTINGS,
    36→  type GeneratorOptions,
    37→  type NewItemInput,
    38→  type VaultDef,
    39→  type VaultItem,
    40→  type VaultMeta,
    41→  type VaultSettings,
    42→} from "@/lib/types";
    43→import {
    44→  exportToCsv,
    45→  importFromText,
    46→  type ImportResult,
    47→  type LckedExport,
    48→} from "@/lib/import-export";
    49→
    50→export type VaultStatus = "loading" | "setup" | "locked" | "unlocked";
    51→
    52→interface VaultState {
    53→  status: VaultStatus;
    54→  items: VaultItem[];
    55→  /** User-defined vaults (colored containers, Proton Pass–style). */
    56→  vaults: VaultDef[];
    57→  /** Active vault filter: "all" | "trash" | vault id. */
    58→  activeVault: string;
    59→  /** Vault editor (create/rename/delete vault) dialog state. */
    60→  vaultEditorOpen: boolean;
    61→  editingVaultId: string | null;
    62→  createVaultDialogOpen: boolean;
    63→  /** IDs of vaults the user has hidden (not shown in sidebar, excluded from search). */
    64→  hiddenVaultIds: string[];
    65→  settings: VaultSettings;
    66→  /** UI-only selection / filter state (not persisted). */
    67→  selectedId: string | null;
    68→  searchQuery: string;
    69→  /** CryptoKey handles — held in memory only while unlocked. */
    70→  masterKey: CryptoKey | null;
    71→  vaultKey: CryptoKey | null;
    72→  /** Whether the active editor is open (and which item, if editing). */
    73→  editorOpen: boolean;
    74→  editorItemId: string | null;
    75→  generatorOpen: boolean;
    76→  importExportOpen: boolean;
    77→  settingsOpen: boolean;
    78→  commandOpen: boolean;
    79→
    80→  // lifecycle
    81→  init: () => Promise<void>;
    82→  setupVault: (masterPassword: string) => Promise<void>;
    83→  unlock: (masterPassword: string) => Promise<boolean>;
    84→  lock: () => void;
    85→  resetVault: () => Promise<void>;
    86→
    87→  // item CRUD
    88→  saveItem: (input: NewItemInput, existingId?: string) => Promise<VaultItem>;
    89→  /** Soft-delete — moves the item to Trash with a 30-day TTL. */
    90→  trashItem: (id: string) => Promise<void>;
    91→  restoreItem: (id: string) => Promise<void>;
    92→  permanentlyDeleteItem: (id: string) => Promise<void>;
    93→  emptyTrash: () => Promise<void>;
    94→  moveItemToVault: (itemId: string, vaultId: string | null) => Promise<void>;
    95→  moveAllItems: (fromVaultId: string | null, toVaultId: string | null) => Promise<void>;
    96→  toggleVaultHidden: (vaultId: string) => void;
    97→  toggleFavorite: (id: string) => Promise<void>;
    98→  duplicateItem: (id: string) => Promise<void>;
    99→  importItems: (filename: string, text: string) => Promise<ImportResult>;
   100→
   101→  // vault (custom containers) CRUD
   102→  createVault: (name: string, color: string, icon: string) => Promise<VaultDef>;
   103→  deleteVault: (id: string) => Promise<void>;
   104→  renameVault: (id: string, name: string) => Promise<void>;
   105→  updateVault: (id: string, patch: Partial<Omit<VaultDef, "id" | "createdAt">>) => Promise<void>;
   106→  setActiveVault: (v: string) => void;
   107→  setVaultEditorOpen: (open: boolean, vaultId?: string | null) => void;
   108→  setCreateVaultDialogOpen: (open: boolean) => void;
   109→
   110→  // settings
   111→  updateSettings: (patch: Partial<VaultSettings>) => Promise<void>;
   112→  updateGenerator: (patch: Partial<GeneratorOptions>) => void;
   113→  changeMasterPassword: (current: string, next: string) => Promise<boolean>;
   114→
   115→  // export
   116→  exportEncrypted: (password: string) => Promise<string>;
   117→  exportCsv: () => string;
   118→
   119→  // UI helpers
   120→  setSelected: (id: string | null) => void;
   121→  setSearch: (q: string) => void;
   122→  setEditorOpen: (open: boolean, itemId?: string | null) => void;
   123→  setGeneratorOpen: (open: boolean) => void;
   124→  setImportExportOpen: (open: boolean) => void;
   125→  setSettingsOpen: (open: boolean) => void;
   126→  setCommandOpen: (open: boolean) => void;
   127→}
   128→
   129→export const useVault = create<VaultState>()(
   130→  persist(
   131→    (set, get) => ({
   132→      status: "loading",
   133→      items: [],
   134→      vaults: [],
   135→      activeVault: "all",
   136→      vaultEditorOpen: false,
   137→      editingVaultId: null,
   138→      createVaultDialogOpen: false,
   139→      hiddenVaultIds: [],
   140→      settings: DEFAULT_VAULT_SETTINGS,
   141→      selectedId: null,
   142→      searchQuery: "",
   143→      masterKey: null,
   144→      vaultKey: null,
   145→      editorOpen: false,
   146→      editorItemId: null,
   147→      generatorOpen: false,
   148→      importExportOpen: false,
   149→      settingsOpen: false,
   150→      commandOpen: false,
   151→
   152→      /* ------------------------------ lifecycle ------------------------------ */
   153→
   154→      init: async () => {
   155→        try {
   156→          if (typeof window === "undefined") return;
   157→          const exists = await vaultExists();
   158→          set({ status: exists ? "locked" : "setup" });
   159→        } catch (err) {
   160→          console.error("init failed", err);
   161→          set({ status: "setup" });
   162→        }
   163→      },
   164→
   165→      setupVault: async (masterPassword) => {
   166→        const salt = bytesToBase64(randomBytes(16));
   167→        const masterKey = await deriveMasterKey(masterPassword, salt, PBKDF2_ITERATIONS);
   168→        const vaultKey = await generateVaultKey();
   169→        const { ciphertext, iv } = await wrapVaultKey(vaultKey, masterKey);
   170→        const verifier = await buildVerifier(masterKey);
   171→
   172→        const meta: VaultMeta = {
   173→          id: "singleton",
   174→          salt,
   175→          iterations: PBKDF2_ITERATIONS,
   176→          encryptedVaultKey: ciphertext,
   177→          vaultKeyIv: iv,
   178→          verifier: verifier.verifier,
   179→          verifierIv: verifier.verifierIv,
   180→          verifierToken: verifier.verifierToken,
   181→          createdAt: Date.now(),
   182→          settings: DEFAULT_VAULT_SETTINGS,
   183→          vaults: [],
   184→        };
   185→        await saveVaultMeta(meta);
   186→
   187→        set({
   188→          status: "unlocked",
   189→          masterKey,
   190→          vaultKey,
   191→          settings: DEFAULT_VAULT_SETTINGS,
   192→          items: [],
   193→          vaults: [],
   194→          activeVault: "all",
   195→          selectedId: null,
   196→        });
   197→      },
   198→
   199→      unlock: async (masterPassword) => {
   200→        const meta = await loadVaultMeta();
   201→        if (!meta) {
   202→          set({ status: "setup" });
   203→          return false;
   204→        }
   205→        const masterKey = await deriveMasterKey(
   206→          masterPassword,
   207→          meta.salt,
   208→          meta.iterations,
   209→        );
   210→        const ok = await checkVerifier(
   211→          masterKey,
   212→          meta.verifier,
   213→          meta.verifierIv,
   214→          meta.verifierToken,
   215→        );
   216→        if (!ok) return false;
   217→
   218→        const vaultKey = await unwrapVaultKey(
   219→          meta.encryptedVaultKey,
   220→          meta.vaultKeyIv,
   221→          masterKey,
   222→        );
   223→
   224→        // Decrypt every stored item.
   225→        const stored = await loadAllStoredItems();
   226→        const items: VaultItem[] = [];
   227→        const toReencrypt: VaultItem[] = [];
   228→        let needsMetaSave = false;
   229→        const TRASH_TTL_MS = 30 * 24 * 60 * 60 * 1000; // 30 days
   230→        const now = Date.now();
   231→        for (const s of stored) {
   232→          try {
   233→            const item = await decryptJson<VaultItem>(s.ciphertext, s.iv, vaultKey);
   234→            // Migration: ensure the new soft-delete + vaultId fields exist on
   235→            // items written by older LCKED versions. Mutated records are queued
   236→            // for a transparent re-encrypt + persist.
   237→            let migrated: VaultItem | null = null;
   238→            if (item.vaultId === undefined) {
   239→              item.vaultId = null;
   240→              migrated = item;
   241→            }
   242→            if (item.trashed === undefined) {
   243→              item.trashed = false;
   244→              migrated = item;
   245→            }
   246→            if (item.trashedAt === undefined) {
   247→              item.trashedAt = null;
   248→              migrated = item;
   249→            }
   250→            // 30-day auto-delete: drop trashed items whose TTL has expired.
   251→            if (item.trashed && item.trashedAt && now - item.trashedAt > TRASH_TTL_MS) {
   252→              try {
   253→                await deleteStoredItem(item.id);
   254→              } catch {
   255→                // best-effort; keep going
   256→              }
   257→              continue;
   258→            }
   259→            if (migrated) toReencrypt.push(migrated);
   260→            items.push(item);
   261→          } catch {
   262→            // skip corrupt record but keep the vault usable
   263→            console.warn("Failed to decrypt item", s.id);
   264→          }
   265→        }
   266→        // Persist any migrated records so we never re-migrate them again.
   267→        if (toReencrypt.length > 0) {
   268→          await Promise.all(
   269→            toReencrypt.map(async (it) => {
   270→              const { ciphertext, iv } = await encryptJson(it, vaultKey);
   271→              await putStoredItem({
   272→                id: it.id,
   273→                type: it.type,
   274→                ciphertext,
   275→                iv,
   276→                createdAt: it.createdAt,
   277→                updatedAt: it.updatedAt,
   278→              });
   279→            }),
   280→          );
   281→        }
   282→        items.sort((a, b) => b.updatedAt - a.updatedAt);
   283→
   284→        // Hydrate vaults (custom containers) — migrate older meta that lacks
   285→        // the field by persisting an empty array.
   286→        const vaults: VaultDef[] = Array.isArray(meta.vaults) ? meta.vaults : [];
   287→        if (!Array.isArray(meta.vaults)) {
   288→          needsMetaSave = true;
   289→        }
   290→        if (needsMetaSave) {
   291→          await saveVaultMeta({ ...meta, vaults });
   292→        }
   293→
   294→        set({
   295→          status: "unlocked",
   296→          masterKey,
   297→          vaultKey,
   298→          items,
   299→          vaults,
   300→          activeVault: "all",
   301→          settings: meta.settings,
   302→        });
   303→        return true;
   304→      },
   305→
   306→      lock: () => {
   307→        set({
   308→          status: "locked",
   309→          masterKey: null,
   310→          vaultKey: null,
   311→          items: [],
   312→          vaults: [],
   313→          activeVault: "all",
   314→          selectedId: null,
   315→          searchQuery: "",
   316→          editorOpen: false,
   317→          generatorOpen: false,
   318→          settingsOpen: false,
   319→          commandOpen: false,
   320→          vaultEditorOpen: false,
   321→          editingVaultId: null,
   322→          createVaultDialogOpen: false,
   323→        });
   324→      },
   325→
   326→      resetVault: async () => {
   327→        await wipeVault();
   328→        set({
   329→          status: "setup",
   330→          items: [],
   331→          vaults: [],
   332→          activeVault: "all",
   333→          masterKey: null,
   334→          vaultKey: null,
   335→          selectedId: null,
   336→          settings: DEFAULT_VAULT_SETTINGS,
   337→          vaultEditorOpen: false,
   338→          editingVaultId: null,
   339→          createVaultDialogOpen: false,
   340→        });
   341→      },
   342→
   343→      /* ------------------------------- CRUD -------------------------------- */
   344→
   345→      saveItem: async (input, existingId) => {
   346→        const { vaultKey } = get();
   347→        if (!vaultKey) throw new Error("Vault is locked");
   348→
   349→        const now = Date.now();
   350→        const id = existingId ?? randomId();
   351→        const existing = existingId ? get().items.find((i) => i.id === existingId) : undefined;
   352→        // Never inherit trashed state from a duplicated item — duplicates land
   353→        // in the active view, not in Trash.
   354→        const baseTrashed = existing ? existing.trashed : false;
   355→        const baseTrashedAt = existing ? existing.trashedAt : null;
   356→        const baseVaultId = existing ? existing.vaultId : (input as VaultItem).vaultId ?? null;
   357→        const item: VaultItem = {
   358→          ...(input as VaultItem),
   359→          id,
   360→          createdAt: existing?.createdAt ?? now,
   361→          updatedAt: now,
   362→          vaultId: baseVaultId,
   363→          trashed: baseTrashed,
   364→          trashedAt: baseTrashedAt,
   365→        } as VaultItem;
   366→
   367→        const { ciphertext, iv } = await encryptJson(item, vaultKey);
   368→        await putStoredItem({ id, type: item.type, ciphertext, iv, createdAt: item.createdAt, updatedAt: now });
   369→
   370→        set((state) => {
   371→          const others = state.items.filter((i) => i.id !== id);
   372→          const next = [item, ...others].sort((a, b) => b.updatedAt - a.updatedAt);
   373→          return { items: next, selectedId: id };
   374→        });
   375→        return item;
   376→      },
   377→
   378→      /**
   379→       * Soft-delete: marks the item as trashed and stamps trashedAt. The record
   380→       * stays in IndexedDB (still encrypted) so it can be restored. Auto-purged
   381→       * 30 days later on the next unlock.
   382→       */
   383→      trashItem: async (id) => {
   384→        await updateItemFlags(id, { trashed: true, trashedAt: Date.now() }, get, set);
   385→      },
   386→
   387→      restoreItem: async (id) => {
   388→        await updateItemFlags(id, { trashed: false, trashedAt: null }, get, set);
   389→      },
   390→
   391→      permanentlyDeleteItem: async (id) => {
   392→        const prev = get().items;
   393→        set((state) => ({
   394→          items: state.items.filter((i) => i.id !== id),
   395→          selectedId: state.selectedId === id ? null : state.selectedId,
   396→        }));
   397→        try {
   398→          await deleteStoredItem(id);
   399→        } catch (err) {
   400→          console.error("permanent delete failed", err);
   401→          set({ items: prev });
   402→          throw err;
   403→        }
   404→      },
   405→
   406→      emptyTrash: async () => {
   407→        const trashed = get().items.filter((i) => i.trashed);
   408→        const prev = get().items;
   409→        // Optimistic: drop all trashed items from UI.
   410→        set((state) => ({
   411→          items: state.items.filter((i) => !i.trashed),
   412→          selectedId:
   413→            state.selectedId && trashed.some((t) => t.id === state.selectedId)
   414→              ? null
   415→              : state.selectedId,
   416→        }));
   417→        try {
   418→          await Promise.all(trashed.map((t) => deleteStoredItem(t.id)));
   419→        } catch (err) {
   420→          console.error("emptyTrash failed", err);
   421→          set({ items: prev });
   422→          throw err;
   423→        }
   424→      },
   425→
   426→      moveItemToVault: async (itemId, vaultId) => {
   427→        await updateItemFlags(itemId, { vaultId, updatedAt: Date.now() }, get, set);
   428→      },
   429→
   430→      moveAllItems: async (fromVaultId, toVaultId) => {
   431→        const items = get().items.filter(
   432→          (i) => !i.trashed && i.vaultId === fromVaultId,
   433→        );
   434→        for (const item of items) {
   435→          await updateItemFlags(item.id, { vaultId: toVaultId, updatedAt: Date.now() }, get, set);
   436→        }
   437→      },
   438→
   439→      toggleVaultHidden: (vaultId) => {
   440→        set((state) => {
   441→          const hidden = state.hiddenVaultIds.includes(vaultId)
   442→            ? state.hiddenVaultIds.filter((id) => id !== vaultId)
   443→            : [...state.hiddenVaultIds, vaultId];
   444→          return { hiddenVaultIds: hidden };
   445→        });
   446→      },
   447→
   448→      toggleFavorite: async (id) => {
   449→        const item = get().items.find((i) => i.id === id);
   450→        if (!item) return;
   451→        const updated: VaultItem = {
   452→          ...item,
   453→          favorite: !item.favorite,
   454→          updatedAt: Date.now(),
   455→        } as VaultItem;
   456→        const { vaultKey } = get();
   457→        if (vaultKey) {
   458→          const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   459→          await putStoredItem({ id, type: updated.type, ciphertext, iv, createdAt: updated.createdAt, updatedAt: updated.updatedAt });
   460→        }
   461→        set((state) => ({
   462→          items: state.items
   463→            .map((i) => (i.id === id ? updated : i))
   464→            .sort((a, b) => b.updatedAt - a.updatedAt),
   465→        }));
   466→      },
   467→
   468→      duplicateItem: async (id) => {
   469→        const item = get().items.find((i) => i.id === id);
   470→        if (!item) return;
   471→        const { id: _id, createdAt: _c, updatedAt: _u, trashed: _t, trashedAt: _ta, ...rest } = item;
   472→        void _id; void _c; void _u; void _t; void _ta;
   473→        await get().saveItem(rest as NewItemInput);
   474→      },
   475→
   476→      importItems: async (filename, text) => {
   477→        const { vaultKey } = get();
   478→        if (!vaultKey) throw new Error("Vault is locked");
   479→        const { result, items } = importFromText(filename, text);
   480→
   481→        // Encrypted LCKED imports require the export's password — handled by a
   482→        // dedicated UI flow. Here we surface a clear hint instead of failing.
   483→        if ((result as any).__raw) {
   484→          return {
   485→            imported: 0,
   486→            skipped: 0,
   487→            warnings: ["Encrypted LCKED files need their own password — use the dedicated import path."],
   488→          };
   489→        }
   490→
   491→        let imported = 0;
   492→        for (const input of items) {
   493→          try {
   494→            await get().saveItem(input);
   495→            imported++;
   496→          } catch {
   497→            result.skipped++;
   498→          }
   499→        }
   500→        result.imported = imported;
   501→        return result;
   502→      },
   503→
   504→      /* --------------------------- vaults (containers) ---------------------- */
   505→
   506→      createVault: async (name, color, icon) => {
   507→        const meta = await loadVaultMeta();
   508→        if (!meta) throw new Error("Vault meta missing");
   509→        const vault: VaultDef = {
   510→          id: randomId(),
   511→          name: name.trim() || "Untitled vault",
   512→          color,
   513→          icon,
   514→          createdAt: Date.now(),
   515→        };
   516→        const vaults = [...(meta.vaults ?? []), vault];
   517→        await saveVaultMeta({ ...meta, vaults });
   518→        set({ vaults, createVaultDialogOpen: false, vaultEditorOpen: false, editingVaultId: null });
   519→        return vault;
   520→      },
   521→
   522→      deleteVault: async (id) => {
   523→        const meta = await loadVaultMeta();
   524→        if (!meta) return;
   525→        const vaults = (meta.vaults ?? []).filter((v) => v.id !== id);
   526→        await saveVaultMeta({ ...meta, vaults });
   527→        // Orphan any items that lived in this vault — they fall back to the
   528→        // default vault (vaultId = null). Re-encrypt + persist each.
   529→        const { vaultKey, items } = get();
   530→        const orphaned = items.filter((i) => i.vaultId === id);
   531→        if (vaultKey && orphaned.length > 0) {
   532→          await Promise.all(
   533→            orphaned.map(async (it) => {
   534→              const next: VaultItem = { ...it, vaultId: null, updatedAt: Date.now() } as VaultItem;
   535→              const { ciphertext, iv } = await encryptJson(next, vaultKey);
   536→              await putStoredItem({
   537→                id: next.id,
   538→                type: next.type,
   539→                ciphertext,
   540→                iv,
   541→                createdAt: next.createdAt,
   542→                updatedAt: next.updatedAt,
   543→              });
   544→            }),
   545→          );
   546→          set((state) => ({
   547→            items: state.items.map((i) =>
   548→              i.vaultId === id ? ({ ...i, vaultId: null, updatedAt: Date.now() } as VaultItem) : i,
   549→            ),
   550→          }));
   551→        }
   552→        set((state) => ({
   553→          vaults,
   554→          activeVault: state.activeVault === id ? "all" : state.activeVault,
   555→          vaultEditorOpen: state.editingVaultId === id ? false : state.vaultEditorOpen,
   556→          editingVaultId: state.editingVaultId === id ? null : state.editingVaultId,
   557→        }));
   558→      },
   559→
   560→      renameVault: async (id, name) => {
   561→        await get().updateVault(id, { name });
   562→      },
   563→
   564→      updateVault: async (id, patch) => {
   565→        const meta = await loadVaultMeta();
   566→        if (!meta) return;
   567→        const vaults = (meta.vaults ?? []).map((v) =>
   568→          v.id === id ? { ...v, ...patch, name: patch.name ?? v.name } : v,
   569→        );
   570→        await saveVaultMeta({ ...meta, vaults });
   571→        set({ vaults, vaultEditorOpen: false, editingVaultId: null });
   572→      },
   573→
   574→      setActiveVault: (v) => set({ activeVault: v }),
   575→      setVaultEditorOpen: (open, vaultId = null) =>
   576→        set({ vaultEditorOpen: open, editingVaultId: vaultId }),
   577→      setCreateVaultDialogOpen: (open) => set({ createVaultDialogOpen: open }),
   578→
   579→      /* ----------------------------- settings ------------------------------- */
   580→
   581→      updateSettings: async (patch) => {
   582→        const next = { ...get().settings, ...patch };
   583→        set({ settings: next });
   584→        const meta = await loadVaultMeta();
   585→        if (meta) {
   586→          await saveVaultMeta({ ...meta, settings: next });
   587→        }
   588→      },

   590→      updateGenerator: (patch) => {
   591→        set((state) => ({
   592→          settings: { ...state.settings, generator: { ...state.settings.generator, ...patch } },
   593→        }));
   594→      },

   596→      changeMasterPassword: async (current, next) => {
   597→        const meta = await loadVaultMeta();
   598→        if (!meta) return false;
   599→        const currentKey = await deriveMasterKey(current, meta.salt, meta.iterations);
   600→        const ok = await checkVerifier(currentKey, meta.verifier, meta.verifierIv, meta.verifierToken);
   601→        if (!ok) return false;
   602→
   603→        // New salt + new master key; re-wrap the SAME vault key so no item
   604→        // needs re-encryption.
   605→        const newSalt = bytesToBase64(randomBytes(16));
   606→        const newMasterKey = await deriveMasterKey(next, newSalt, PBKDF2_ITERATIONS);
   607→        const { vaultKey } = get();
   608→        if (!vaultKey) return false;
   609→        const wrapped = await wrapVaultKey(vaultKey, newMasterKey);
   610→        const verifier = await buildVerifier(newMasterKey);

   612→          ...meta,
   613→          salt: newSalt,
   614→          iterations: PBKDF2_ITERATIONS,

   616→          vaultKeyIv: wrapped.iv,
   617→          verifier: verifier.verifier,
   618→          verifierIv: verifier.verifierIv,
   619→        });

   621→        return true;
   622→      },

   626→      exportEncrypted: async (password) => {
   627→        const items = get().items;
   628→        const vaults = get().vaults;
   629→        const salt = bytesToBase64(randomBytes(16));
   630→        const exportMasterKey = await deriveMasterKey(password, salt, PBKDF2_ITERATIONS);
   631→        const exportVaultKey = await generateVaultKey();
   632→        const verifier = await buildVerifier(exportMasterKey);

   637→        // plaintext-side metadata (names/colors/icons only) but rides inside

   643→          vaults,

   654→          verifierIv: verifier.verifierIv,
   655→          data: fullCipher,
   656→          dataIv: fullIv,
   657→        };

   686→/**
   687→ * Patch one or more flags on an item (vaultId / trashed / trashedAt / favorite)
   688→ * and re-encrypt + persist it. Used by trashItem / restoreItem / moveItemToVault
   689→ * so we have a single fault-tolerant code path.
   690→ */
   691→async function updateItemFlags(
   692→  id: string,
   693→  patch: Partial<Pick<VaultItem, "vaultId" | "trashed" | "trashedAt" | "updatedAt">>,
   694→  get: () => VaultState,
   695→  set: (partial: Partial<VaultState>) => void,
   696→): Promise<void> {
   697→  const item = get().items.find((i) => i.id === id);
   698→  if (!item) return;
   699→  const now = patch.updatedAt ?? Date.now();
   700→  const updated: VaultItem = { ...item, ...patch, updatedAt: now } as VaultItem;
   701→  const { vaultKey } = get();
   702→  if (vaultKey) {
   703→    const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   704→    await putStoredItem({
   705→      id,
   706→      type: updated.type,
   707→      ciphertext,
   708→      iv,
   709→      createdAt: updated.createdAt,
   710→      updatedAt: now,
   711→    });
   712→  }
   713→  set({
   714→    items: get()
   715→      .items.map((i) => (i.id === id ? updated : i))
   716→      .sort((a, b) => b.updatedAt - a.updatedAt),
   717→  });
   718→}

   720→/**
   721→ * Copy-to-clipboard with an automatic 30-second clear. Returns the timeout id
   722→ * so callers can cancel early if needed. Sensitive values never linger.
   723→ */
   724→const clipboardTimers = new Map<string, ReturnType<typeof setTimeout>>();
   725→
   726→export async function copyWithAutoClear(
   727→  value: string,
   728→  key = "default",
   729→  clearMs = 30_000,
   730→): Promise<void> {
   731→  if (typeof navigator === "undefined" || !navigator.clipboard?.writeText) {
   732→    throw new Error("Clipboard API unavailable");
   733→  }
   734→  await navigator.clipboard.writeText(value);
   735→  // Clear any prior timer for this key.
   736→  const prior = clipboardTimers.get(key);
   737→  if (prior) clearTimeout(prior);
   738→  const timer = setTimeout(async () => {
   739→    try {
   740→      const current = await navigator.clipboard.readText().catch(() => "");
   741→      if (current === value) {
   742→        await navigator.clipboard.writeText("");
   743→      }
   744→    } catch {
   745→      // readText may be denied — best-effort clear.
   746→    }
   747→    clipboardTimers.delete(key);
   748→  }, clearMs);
   749→  clipboardTimers.set(key, timer);
   750→}
   751→
   752→export function cancelClipboardClear(key = "default"): void {
   753→  const t = clipboardTimers.get(key);
   754→  if (t) {
   755→    clearTimeout(t);
   756→    clipboardTimers.delete(key);
   757→  }
   758→}
