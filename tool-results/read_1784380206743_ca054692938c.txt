    69→    69→function blankItem(type: ItemType): NewItemInput {
    70→    70→  const base = {
    71→    71→    name: "",
    72→    72→    favorite: false,
    73→    73→    folder: "",
    74→    74→    customFields: [],
    75→    75→    createdAt: Date.now(),
    76→    76→    updatedAt: Date.now(),
    77→    77→    vaultId: null as string | null,
    78→    78→    trashed: false,
    79→    79→    trashedAt: null as number | null,
    80→    80→  };
    81→    81→  if (type === "login") {
    82→    82→    return { ...base, type, details: { username: "", password: "", urls: [""], totp: "", notes: "" } };
    83→    83→  }
    84→    84→  if (type === "note") {
    85→    85→    return { ...base, type, details: { content: "" } };
    86→    86→  }
    87→    87→  if (type === "card") {
    88→    88→    return {
    89→    89→      ...base,
    90→    90→      type,
    91→    91→      details: { cardholder: "", number: "", brand: "", cvv: "", expiry: "", pin: "", notes: "" },
    92→    92→    };
    93→    93→  }
    94→    94→  return {
    95→    95→    ...base,
    96→    96→    type,
    97→    97→    details: {
    98→    98→      firstName: "", lastName: "", email: "", phone: "", company: "",
    99→    99→      address1: "", address2: "", city: "", state: "", zip: "", country: "", notes: "",
   100→   100→    },
   101→   101→  };
   102→   102→}

   104→   104→function itemToInput(item: VaultItem): NewItemInput {
   105→   105→  const { id: _id, createdAt: _c, updatedAt: _u, ...rest } = item;
   106→   106→  return rest as NewItemInput;
   107→   107→}

   144→   144→export function ItemEditor() {
   145→   145→  const open = useVault((s) => s.editorOpen);
   146→   146→  const editorItemId = useVault((s) => s.editorItemId);
   147→   147→  const items = useVault((s) => s.items);
   148→   148→  const vaults = useVault((s) => s.vaults);
   149→   149→  const settings = useVault((s) => s.settings);
   150→   150→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   151→   151→  const saveItem = useVault((s) => s.saveItem);
   152→   152→
   153→   153→  const [form, setForm] = React.useState<NewItemInput | null>(null);
   154→   154→  const [busy, setBusy] = React.useState(false);
   155→   155→
   156→   156→  // Initialise form when the sheet opens.
   157→   157→  React.useEffect(() => {
   158→   158→    if (!open) {
   159→   159→      setForm(null);
   160→   160→      return;
   161→   161→    }
   162→   162→    if (editorItemId) {
   163→   163→      const existing = items.find((i) => i.id === editorItemId);
   164→   164→      if (existing) {
   165→   165→        setForm(itemToInput(existing));
   166→   166→        return;
   167→   167→      }
   168→   168→    }
   169→   169→    // Honour a type pre-selected via ⌘K / icon rail, then clear the stash.
   170→   170→    const initialType = consumeNewItemType() ?? "login";
   171→   171→    setForm(blankItem(initialType));
   172→   172→  }, [open, editorItemId, items]);
   173→   173→
   174→   174→  const isEditing = Boolean(editorItemId);
   175→   175→
   176→   176→  if (!open || !form) return null;
   177→   177→
   178→   178→  const update = (patch: Partial<NewItemInput>) => setForm((f) => (f ? { ...f, ...patch } : f));
   179→   179→  const updateDetails = (patch: Record<string, unknown>) =>
   180→   180→    setForm((f) => (f ? { ...f, details: { ...(f.details as object), ...patch } } : f));
   181→   181→
   182→   182→  const handleSave = async () => {
   183→   183→    if (!form) return;
   184→   184→    if (!form.name.trim()) {
   185→   185→      toast.error("Please give the item a name");
   186→   186→      return;
   187→   187→    }
   188→   188→    setBusy(true);
   189→   189→    try {
   190→   190→      await saveItem(form, editorItemId ?? undefined);
   191→   191→      toast.success(isEditing ? "Item updated" : "Item created");
   192→   192→      setEditorOpen(false);
   193→   193→    } catch (err) {
   194→   194→      console.error(err);
   195→   195→      toast.error("Could not save item");
   196→   196→    } finally {
   197→   197→      setBusy(false);
   198→   198→    }
   199→   199→  };
   200→   200→
   201→   201→  const addCustomField = () =>
   202→   202→    update({ customFields: [...form.customFields, { name: "", value: "", type: "text" }] });
   203→   203→  const updateCustomField = (idx: number, patch: Partial<CustomField>) =>
   204→   204→    update({
   205→   205→      customFields: form.customFields.map((cf, i) => (i === idx ? { ...cf, ...patch } : cf)),
   206→   206→    });
   207→   207→  const removeCustomField = (idx: number) =>
   208→   208→    update({ customFields: form.customFields.filter((_, i) => i !== idx) });
   209→   209→
   210→   210→  // Login URL list helpers.
   211→   211→  const urls = form.type === "login" ? form.details.urls : [];
   212→   212→  const setUrl = (idx: number, val: string) => {
   213→   213→    const next = urls.map((u, i) => (i === idx ? val : u));
   214→   214→    updateDetails({ urls: next });
   215→   215→  };
   216→   216→  const addUrl = () => updateDetails({ urls: [...urls, ""] });
   217→   217→  const removeUrl = (idx: number) =>
   218→   218→    updateDetails({ urls: urls.filter((_, i) => i !== idx) });
   219→   219→
   220→   220→  // Selected vault (for the header dropdown).
   221→   221→  const selectedVault = form.vaultId ? vaults.find((v) => v.id === form.vaultId) ?? null : null;
   222→   222→
   223→   223→  return (
   224→   224→    <Sheet open={open} onOpenChange={(o) => setEditorOpen(o)}>
   225→   225→      <SheetContent
   226→   226→        side="right"
   227→   227→        className="w-full gap-0 overflow-hidden border-l border-border bg-background p-0 sm:max-w-[480px] [&>button:last-child]:hidden"
   228→   228→      >
   229→   229→        {/* Header — type icon + title (left); vault selector + Save (right) */}
   230→   230→        <SheetHeader className="flex-row items-center justify-between gap-2 border-b border-border px-4 py-3">
   231→   231→          <SheetTitle className="flex min-w-0 items-center gap-2.5 text-base font-semibold">
   232→   232→            <ItemTypeIcon type={form.type} size="sm" />
   233→   233→            <span className="truncate">{isEditing ? "Edit item" : "New item"}</span>
   234→   234→          </SheetTitle>
   235→   235→          <div className="flex shrink-0 items-center gap-1.5">
   236→   236→            {/* Vault selector dropdown */}
   237→   237→            <DropdownMenu>
   238→   238→              <DropdownMenuTrigger asChild>
   239→   239→                <button
   240→   240→                  type="button"
   241→   241→                  className="flex h-8 items-center gap-1.5 rounded-md border border-border bg-background px-2 text-xs text-foreground transition-colors hover:bg-muted/60"
   242→   242→                  aria-label="Select vault"
   243→   243→                  title={selectedVault ? selectedVault.name : "No vault"}
   244→   244→                >
   245→   245→                  {selectedVault ? (
   246→   246→                    <VaultIcon icon={selectedVault.icon} color={selectedVault.color} size={18} />
   247→   247→                  ) : (
   248→   248→                    <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   249→   249→                      —
   250→   250→                    </span>
   251→   251→                  )}
   252→   252→                  <span className="max-w-[80px] truncate">{selectedVault ? selectedVault.name : "No vault"}</span>
   253→   253→                  <ChevronsUpDown className="h-3 w-3 text-muted-foreground" />
   254→   254→                </button>
   255→   255→              </DropdownMenuTrigger>
   256→   256→              <DropdownMenuContent align="end" className="min-w-[180px]">
   257→   257→                <DropdownMenuItem
   258→   258→                  onSelect={() => update({ vaultId: null })}
   259→   259→                  className={cn("gap-2", !selectedVault && "bg-muted/60")}
   260→   260→                >
   261→   261→                  <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   262→   262→                    —
   263→   263→                  </span>
   264→   264→                  No vault
   265→   265→                </DropdownMenuItem>
   266→   266→                {vaults.map((v) => (
   267→   267→                  <DropdownMenuItem
   268→   268→                    key={v.id}
   269→   269→                    onSelect={() => update({ vaultId: v.id })}
   270→   270→                    className={cn("gap-2", selectedVault?.id === v.id && "bg-muted/60")}
   271→   271→                  >
   272→   272→                    <VaultIcon icon={v.icon} color={v.color} size={18} />
   273→   273→                    <span className="truncate">{v.name}</span>
   274→   274→                    {selectedVault?.id === v.id && <Check className="ml-auto h-3.5 w-3.5" />}
   275→   275→                  </DropdownMenuItem>
   276→   276→                ))}
   277→   277→              </DropdownMenuContent>
   278→   278→            </DropdownMenu>
   279→   279→
   280→   280→            {/* Save / Create */}
   281→   281→            <Button
   282→   282→              type="button"
   283→   283→              size="sm"
   284→   284→              onClick={handleSave}
   285→   285→              disabled={busy}
   286→   286→              className="min-w-[72px]"
   287→   287→            >
   288→   288→              {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : isEditing ? "Save" : "Create"}
   289→   289→            </Button>
   290→   290→          </div>
   291→   291→        </SheetHeader>
   292→   292→        <SheetDescription className="sr-only">
   293→   293→          {isEditing ? "Edit the selected vault item." : "Create a new vault item."}
   294→   294→        </SheetDescription>
   295→   295→
   296→   296→        {/* Body — flat scrollable form */}
   297→   297→        <div className="lcked-scroll flex-1 overflow-y-auto p-4">
   298→   298→          {/* Type selector — only when creating */}
   299→   299→          {!isEditing && (
   300→   300→            <div className="mb-4 grid grid-cols-4 gap-2">
   301→   301→              {(Object.keys(ITEM_TYPE_LABELS) as ItemType[]).map((t) => (
   302→   302→                <button
   303→   303→                  key={t}
   304→   304→                  type="button"
   305→   305→                  onClick={() => setForm(blankItem(t))}
   306→   306→                  className={cn(
   307→   307→                    "flex flex-col items-center gap-1.5 rounded-lg border p-3 text-xs transition-colors",
   308→   308→                    form.type === t
   309→   309→                      ? "border-primary bg-primary/10 text-foreground"
   310→   310→                      : "border-border text-muted-foreground hover:bg-muted/50",
   311→   311→                  )}
   312→   312→                >
   313→   313→                  <ItemTypeIcon type={t} size="sm" />
   314→   314→                  {ITEM_TYPE_LABELS[t]}
   315→   315→                </button>
   316→   316→              ))}
   317→   317→            </div>
   318→   318→          )}
   319→   319→
   320→   320→          {/* Name — flat borderless input (matches create-vault-dialog pattern) */}
   321→   321→          <div className="mb-4">
   322→   322→            <div className="mb-1 text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   323→   323→              Name
   324→   324→            </div>
   325→   325→            <input
   326→   326→              type="text"
   327→   327→              value={form.name}
   328→   328→              onChange={(e) => update({ name: e.target.value })}
   329→   329→              placeholder="e.g. GitHub, Bank card, WiFi password"
   330→   330→              autoFocus
   331→   331→              className="w-full border-0 bg-transparent px-0 py-0.5 text-lg font-medium text-foreground placeholder:text-muted-foreground/60 focus:outline-none"
   332→   332→              aria-label="Item name"
   333→   333→            />
   334→   334→          </div>
   335→   335→
   336→   336→          {/* Favorite toggle — flat row */}
   337→   337→          <label className="mb-4 flex cursor-pointer items-center justify-between rounded-lg border border-border bg-secondary/10 px-3.5 py-2 dark:bg-secondary/10">
   338→   338→            <span className="flex items-center gap-2 text-sm">
   339→   339→              <Star className="h-4 w-4 text-amber-400" />
   340→   340→              Mark as favorite
   341→   341→            </span>
   342→   342→            <Switch checked={form.favorite} onCheckedChange={(v) => update({ favorite: v })} />
   343→   343→          </label>
   344→   344→
   345→   345→          {/* Type-specific */}
   346→   346→          {form.type === "login" && (
   347→   347→            <div className="space-y-4">
   348→   348→              <FieldCluster>
   349→   349→                <FieldRowInput label="Username / email" first>
   350→   350→                  <Input
   351→   351→                    value={form.details.username}
   352→   352→                    onChange={(e) => updateDetails({ username: e.target.value })}
   353→   353→                    placeholder="you@example.com"
   354→   354→                    autoComplete="off"
   355→   355→                    className={flatInputCls}
   356→   356→                  />
   357→   357→                </FieldRowInput>
   358→   358→              </FieldCluster>
   359→   359→
   360→   360→              <PasswordField
   361→   361→                label="Password"
   362→   362→                value={form.details.password}
   363→   363→                onChange={(v) => updateDetails({ password: v })}
   364→   364→                showStrength
   365→   365→                showGenerate
   366→   366→                generatorOptions={settings.generator}
   367→   367→              />
   368→   368→
   369→   369→              <div className="space-y-1.5">
   370→   370→                <Label>Verification code (TOTP)</Label>
   371→   371→                <Input
   372→   372→                  value={form.details.totp}
   373→   373→                  onChange={(e) => updateDetails({ totp: e.target.value })}
   374→   374→                  placeholder="Base32 secret or otpauth:// URI"
   375→   375→                  className="font-secret"
   376→   376→                  autoComplete="off"
   377→   377→                />
   378→   378→              </div>
   379→   379→
   380→   380→              <div className="space-y-1.5">
   381→   381→                <Label>URLs</Label>
   382→   382→                <div className="space-y-2">
   383→   383→                  {urls.map((url, idx) => (
   384→   384→                    <div key={idx} className="flex items-center gap-2">
   385→   385→                      <Link2 className="h-4 w-4 shrink-0 text-muted-foreground" />
   386→   386→                      <Input
   387→   387→                        value={url}
   388→   388→                        onChange={(e) => setUrl(idx, e.target.value)}
   389→   389→                        placeholder="https://example.com"
   390→   390→                        className="font-secret"
   391→   391→                        autoComplete="off"
   392→   392→                      />
   393→   393→                      {urls.length > 1 && (
   394→   394→                        <Button
   395→   395→                          type="button"
   396→   396→                          size="icon"
   397→   397→                          variant="ghost"
   398→   398→                          className="h-8 w-8 shrink-0 text-muted-foreground hover:text-red-400"
   399→   399→                          onClick={() => removeUrl(idx)}
   400→   400→                        >
   401→   401→                          <Trash2 className="h-3.5 w-3.5" />
   402→   402→                        </Button>
   403→   403→                      )}
   404→   404→                    </div>
   405→   405→                  ))}
   406→   406→                  <Button type="button" variant="outline" size="sm" onClick={addUrl}>
   407→   407→                    <Plus className="mr-1.5 h-3.5 w-3.5" />
   408→   408→                    Add URL
   409→   409→                  </Button>
   410→   410→                </div>
   411→   411→              </div>
   412→   412→
   413→   413→              <div className="space-y-1.5">
   414→   414→                <Label>Notes</Label>
   415→   415→                <Textarea
   416→   416→                  value={form.details.notes}
   417→   417→                  onChange={(e) => updateDetails({ notes: e.target.value })}
   418→   418→                  placeholder="Optional secure notes…"
   419→   419→                  rows={3}
   420→   420→                />
   421→   421→              </div>
   422→   422→            </div>
   423→   423→          )}
   424→   424→
   425→   425→          {form.type === "note" && (
   426→   426→            <div className="space-y-1.5">
   427→   427→              <Label>Content</Label>
   428→   428→              <Textarea
   429→   429→                value={form.details.content}
   430→   430→                onChange={(e) => updateDetails({ content: e.target.value })}
   431→   431→                placeholder="Write your secure note here…"
   432→   432→                rows={10}
   433→   433→                className="resize-y"
   434→   434→              />
   435→   435→            </div>
   436→   436→          )}
   437→   437→
   438→   438→          {form.type === "card" && (
   439→   439→            <div className="space-y-4">
   440→   440→              <div className="space-y-1.5">
   441→   441→                <Label>Cardholder name</Label>
   442→   442→                <Input
   443→   443→                  value={form.details.cardholder}
   444→   444→                  onChange={(e) => updateDetails({ cardholder: e.target.value })}
   445→   445→                  placeholder="Name on card"
   446→   446→                  autoComplete="off"
   447→   447→                />
   448→   448→              </div>
   449→   449→              <PasswordField
   450→   450→                label="Card number"
   451→   451→                value={form.details.number}
   452→   452→                onChange={(v) => {
   453→   453→                  const brand = detectCardBrand(v);
   454→   454→                  updateDetails({ number: v, brand: brand || form.details.brand });
   455→   455→                }}
   456→   456→                placeholder="0000 0000 0000 0000"
   457→   457→                showGenerate={false}
