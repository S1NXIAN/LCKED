     3→     3→/**
     4→     4→ * LCKED — VaultsSidebar
     5→     5→ * ---------------------------------------------------------------------------
     6→     6→ * Proton Pass–style vault selector. Renders the fixed rows (All / Favorites /
     7→     7→ * Trash) plus every user-defined vault. Each row is a drag-and-drop drop
     8→     8→ * target so an item dragged from the list can be moved between vaults.
     9→     9→ *
    10→    10→ * Pattern notes:
    11→    11→ *   • Rows that host a ⋮ dropdown are rendered as <div role="button"> (NOT
    12→    12→ *     <button>) so the trigger button doesn't nest inside another button —
    13→    13→ *     nested buttons are invalid HTML and break accessibility tree.
    14→    14→ *   • A fixed-width slot is reserved for the menu button on EVERY row (even
    15→    15→ *     the menu-less ones) so all the right-aligned counts line up.
    16→    16→ *   • "Rename" opens the vault editor sheet via setVaultEditorOpen(true, id);
    17→    17→ *     "Delete" calls deleteVault (which orphan-rescues any items inside).
    18→    18→ *   • Each row wrapper carries a `data-vault-key` attribute consumed by the
    19→    19→ *     sliding VaultActiveHighlight (rAF spring, same pattern as item-list's
    20→    20→ *     ActiveRowHighlight). The highlight lives in the OUTER container so it
    21→    21→ *     can reach Trash (which sits outside the scroll area).
    22→    22→ *   • CustomVaultRow is wrapped in <ContextMenu> with the same items as the
    23→    23→ *     3-dots dropdown: Edit, Move all items (submenu), Hide vault, Delete.
    24→    24→ */

    84→    84→export interface VaultIconProps {
    85→    85→  /** Vault icon id (see VAULT_ICONS). */
    86→    86→  icon: string;
    87→    87→  /** Vault color id (see VAULT_COLORS). */
    88→    88→  color: string;
    89→    89→  /** Pixel size of the square swatch. Default 28. */
    90→    90→  size?: number;
    91→    91→  className?: string;

    94→    94→/**
    95→    95→ * Render a vault's icon inside a colored rounded swatch. The fill is the
    96→    96→ * vault color at 18% opacity; the icon glyph uses the full hex for contrast.
    97→    97→ * Used by VaultsSidebar rows AND the create-vault-dialog picker preview.
    98→    98→ * The Lucide-component lookup lives in `./vault-lucide-icons` so both this
    99→    99→ * module and the picker share a single source of truth.
   100→   100→ */
   101→   101→export function VaultIcon({ icon, color, size = 28, className, bare = false }: VaultIconProps & { bare?: boolean }) {
   102→   102→  const hex = vaultColorHex(color);
   103→   103→  const Resolved = VAULT_LUCIDE_BY_ID[icon] ?? Home;
   104→   104→  const px = `${size}px`;
   105→   105→  const glyph = Math.round(size * 0.55);
   106→   106→  if (bare) {
   107→   107→    return (
   108→   108→      <Resolved
   109→   109→        size={glyph}
   110→   110→        strokeWidth={2}
   111→   111→        style={{ color: hex }}
   112→   112→        className={cn("shrink-0", className)}
   113→   113→        aria-hidden="true"
   114→   114→      />
   115→   115→    );
   116→   116→  }
   117→   117→  return (
   118→   118→    <span
   119→   119→      className={cn("inline-flex shrink-0 items-center justify-center rounded-lg", className)}
   120→   120→      style={{
   121→   121→        width: px,
   122→   122→        height: px,
   123→   123→        backgroundColor: `${hex}29`, // ~16% opacity
   124→   124→        color: hex,
   125→   125→      }}
   126→   126→      aria-hidden="true"
   127→   127→    >
   128→   128→      <Resolved size={glyph} strokeWidth={2} />
   129→   129→    </span>
   130→   130→  );
   131→   131→}

   159→   159→/**
   160→   160→ * A single sidebar row. When `menu` is provided we render a div instead of a
   161→   161→ * button to avoid invalid nested-button HTML. The reserved `menuSlot` keeps
   162→   162→ * every row's right edge aligned whether or not a menu is present.
   163→   163→ *
   164→   164→ * The active state keeps `text-accent-foreground` but NO `bg-accent` — the
   165→   165→ * sliding VaultActiveHighlight provides the background, so a static one
   166→   166→ * would visually fight the spring.
   167→   167→ */
   168→   168→function VaultRow({
   169→   169→  icon,
   170→   170→  label,
   171→   171→  count,
   172→   172→  active,
   173→   173→  onSelect,
   174→   174→  dropVaultId,
   175→   175→  warn,
   176→   176→  menu,
   177→   177→  dragOver,
   178→   178→}: VaultRowProps) {
   179→   179→  const Tag = menu ? "div" : "button";
   180→   180→  const tagProps = menu
   181→   181→    ? { role: "button", tabIndex: 0 }
   182→   182→    : { type: "button" as const };
   183→   183→
   184→   184→  const handleDrop = (e: React.DragEvent) => {
   185→   185→    if (dropVaultId === undefined) return;
   186→   186→    e.preventDefault();
   187→   187→    const id = e.dataTransfer.getData("text/lcked-item");
   188→   188→    if (!id) return;
   189→   189→    useVault.getState().moveItemToVault(id, dropVaultId ?? null);
   190→   190→    toast.success(`Moved to ${label}`);
   191→   191→  };
   192→   192→  const handleDragOver = (e: React.DragEvent) => {
   193→   193→    if (dropVaultId === undefined) return;
   194→   194→    e.preventDefault();
   195→   195→    e.dataTransfer.dropEffect = "move";
   196→   196→  };
   197→   197→  const handleKeyDown = (e: React.KeyboardEvent) => {
   198→   198→    if (!menu) return;
   199→   199→    if (e.key === "Enter" || e.key === " ") {
   200→   200→      e.preventDefault();
   201→   201→      onSelect();
   202→   202→    }
   203→   203→  };
   204→   204→
   205→   205→  return (
   206→   206→    <Tag
   207→   207→      {...tagProps}
   208→   208→      onClick={(e: React.MouseEvent) => {
   209→   209→        const target = e.target as HTMLElement;
   210→   210→        if (target.closest("[data-menu-slot]")) return;
   211→   211→        onSelect();
   212→   212→      }}
   213→   213→      onKeyDown={menu ? handleKeyDown : undefined}
   214→   214→      onDrop={handleDrop}
   215→   215→      onDragOver={handleDragOver}
   216→   216→      aria-current={active ? "true" : undefined}
   217→   217→      className={cn(
   218→   218→        "group relative flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition-colors duration-100",
   219→   219→        "focus:outline-none focus-visible:ring-2 focus-visible:ring-ring/60",
   220→   220→        active
   221→   221→          ? "text-accent-foreground"
   222→   222→          : warn
   223→   223→            ? "text-foreground hover:bg-amber-500/10"
   224→   224→            : "text-foreground/80 hover:bg-muted/60 hover:text-foreground",
   225→   225→        dragOver && "ring-2 ring-primary/60 ring-offset-1 ring-offset-background",
   226→   226→      )}
   227→   227→    >
   228→   228→      <span className="flex h-7 w-7 shrink-0 items-center justify-center">{icon}</span>
   229→   229→      <span className="min-w-0 flex-1">
   230→   230→        <span className="block truncate text-sm font-medium leading-tight">{label}</span>
   231→   231→        <span
   232→   232→          className={cn(
   233→   233→            "block text-xs leading-tight tabular-nums",
   234→   234→            warn && !active ? "text-amber-400/80" : active ? "text-accent-foreground/70" : "text-muted-foreground",
   235→   235→          )}
   236→   236→        >
   237→   237→          {count} {count === 1 ? "item" : "items"}
   238→   238→        </span>
   239→   239→      </span>
   240→   240→      <span
   241→   241→        data-menu-slot
   242→   242→        className="flex h-7 w-7 shrink-0 items-center justify-center"
   243→   243→      >
   244→   244→        {menu}
   245→   245→      </span>
   246→   246→    </Tag>
   247→   247→  );
   248→   248→}

   479→   479→/* --------------------- VaultActiveHighlight (rAF spring) --------------------- */
   480→   480→
   481→   481→/**
   482→   482→ * Single persistent highlight rendered ONCE inside the sidebar outer
   483→   483→ * container. Slides toward the active row's bounding box via a
   484→   484→ * requestAnimationFrame spring (exponential lerp, factor 0.22). Uses the
   485→   485→ * same `animateRef` pattern as item-list's ActiveRowHighlight.
   486→   486→ *
   487→   487→ * The highlight lives in the OUTER container (NOT the scroll area) so it can
   488→   488→ * reach Trash, which sits below the scrollable vault list with `mt-auto`.
   489→   489→ */
   490→   490→function VaultActiveHighlight({
   491→   491→  containerRef,
   492→   492→  activeKey,
   493→   493→}: {
   494→   494→  containerRef: React.RefObject<HTMLDivElement | null>;
   495→   495→  activeKey: string;
   496→   496→}) {
   497→   497→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   498→   498→  const rafRef = React.useRef<number | null>(null);
   499→   499→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   500→   500→  const posRef = React.useRef({ x: 0, y: 0 });
   501→   501→  const wasVisibleRef = React.useRef(false);
   502→   502→  const [visible, setVisible] = React.useState(false);
   503→   503→  const animateRef = React.useRef<(() => void) | null>(null);
   504→   504→
   505→   505→  const measure = React.useCallback(() => {
   506→   506→    const container = containerRef.current;
   507→   507→    if (!activeKey || !container) {
   508→   508→      targetRef.current = null;
   509→   509→      setVisible(false);
   510→   510→      return;
   511→   511→    }
   512→   512→    const el = container.querySelector<HTMLElement>(
   513→   513→      `[data-vault-key="${CSS.escape(activeKey)}"]`,
   514→   514→    );
   515→   515→    if (!el) {
   516→   516→      targetRef.current = null;
   517→   517→      setVisible(false);
   518→   518→      return;
   519→   519→    }
   520→   520→    // Check if element is visible (not scrolled out of view)
   521→   521→    const elRect = el.getBoundingClientRect();
   522→   522→    const containerRect = container.getBoundingClientRect();
   523→   523→    if (elRect.bottom < containerRect.top || elRect.top > containerRect.bottom) {
   524→   524→      // Element is scrolled out of view — hide the indicator
   525→   525→      setVisible(false);
   526→   526→      return;
   527→   527→    }
   528→   528→    const next = {
   529→   529→      x: elRect.left - containerRect.left,
   530→   530→      y: elRect.top - containerRect.top,
   531→   531→      w: elRect.width,
   532→   532→      h: elRect.height,
   533→   533→    };
   534→   534→    if (highlightRef.current) {
   535→   535→      highlightRef.current.style.width = `${next.w}px`;
   536→   536→      highlightRef.current.style.height = `${next.h}px`;
   537→   537→      if (!wasVisibleRef.current) {
   538→   538→        highlightRef.current.style.transform = `translate(${next.x}px, ${next.y}px)`;
   539→   539→        posRef.current = { x: next.x, y: next.y };
   540→   540→      }
   541→   541→    }
   542→   542→    targetRef.current = next;
   543→   543→    setVisible(true);
   544→   544→  }, [activeKey, containerRef]);
   545→   545→
   546→   546→  React.useEffect(() => {
   547→   547→    const step = () => {
   548→   548→      const hl = highlightRef.current;
   549→   549→      const target = targetRef.current;
   550→   550→      if (!hl || !target) {
   551→   551→        rafRef.current = null;
   552→   552→        return;
   553→   553→      }
   554→   554→      const pos = posRef.current;
   555→   555→      const FACTOR = 0.35;
   556→   556→      const nx = pos.x + (target.x - pos.x) * FACTOR;
   557→   557→      const ny = pos.y + (target.y - pos.y) * FACTOR;
   558→   558→      const settled =
   559→   559→        Math.abs(target.x - nx) < 0.5 && Math.abs(target.y - ny) < 0.5;
   560→   560→      pos.x = settled ? target.x : nx;
   561→   561→      pos.y = settled ? target.y : ny;
   562→   562→      hl.style.transform = `translate(${pos.x}px, ${pos.y}px)`;
   563→   563→      if (settled) {
   564→   564→        rafRef.current = null;
   565→   565→        return;
   566→   566→      }
   567→   567→      rafRef.current = requestAnimationFrame(step);
   568→   568→    };
   569→   569→    animateRef.current = step;
   570→   570→    return () => { animateRef.current = null; };
   571→   571→  }, []);
   572→   572→
   573→   573→  React.useEffect(() => {
   574→   574→    wasVisibleRef.current = visible;
   575→   575→  }, [visible]);
   576→   576→
   577→   577→  const kick = React.useCallback(() => {
   578→   578→    measure();
   579→   579→    if (!rafRef.current && animateRef.current) {
   580→   580→      rafRef.current = requestAnimationFrame(animateRef.current);
   581→   581→    }
   582→   582→  }, [measure]);
   583→   583→
   584→   584→  React.useEffect(() => {
   585→   585→    cancelAnimationFrame(rafRef.current ?? 0);
   586→   586→    if (!activeKey) { setVisible(false); return; }
   587→   587→    const timer = setTimeout(() => kick(), 16);
   588→   588→    return () => { clearTimeout(timer); cancelAnimationFrame(rafRef.current ?? 0); };
   589→   589→  }, [activeKey, kick]);
   590→   590→
   591→   591→  React.useEffect(() => {
   592→   592→    const container = containerRef.current;
   593→   593→    if (!container) return;
   594→   594→    const scrollArea = container.querySelector(".lcked-scroll");
   595→   595→    let pending = false;
   596→   596→    const observer = new MutationObserver(() => {
   597→   597→      if (pending) return;
   598→   598→      pending = true;
   599→   599→      requestAnimationFrame(() => { pending = false; kick(); });
   600→   600→    });
   601→   601→    observer.observe(container, { childList: true, subtree: true, attributes: true, attributeFilter: ["data-vault-key"] });
   602→   602→    const handler = () => kick();
   603→   603→    if (scrollArea) scrollArea.addEventListener("scroll", handler, { passive: true });
   604→   604→    container.addEventListener("scroll", handler, { passive: true });
   605→   605→    window.addEventListener("resize", handler, { passive: true });
   606→   606→    return () => {
   607→   607→      observer.disconnect();
   608→   608→      if (scrollArea) scrollArea.removeEventListener("scroll", handler);
   609→   609→      container.removeEventListener("scroll", handler);
   610→   610→      window.removeEventListener("resize", handler);
   611→   611→    };
   612→   612→  }, [containerRef, kick]);
   613→   613→
   614→   614→  if (!visible) return null;
   615→   615→  return (
   616→   616→    <div
   617→   617→      ref={highlightRef}
   618→   618→      aria-hidden
   619→   619→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   620→   620→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   621→   621→    />
   622→   622→  );
   623→   623→}
