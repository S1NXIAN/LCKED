    64→function download(filename: string, content: string, mime: string) {
    65→  const blob = new Blob([content], { type: mime });
    66→  const url = URL.createObjectURL(blob);
    67→  const a = document.createElement("a");
    68→  a.href = url;
    69→  a.download = filename;
    70→  a.click();
    71→  URL.revokeObjectURL(url);
    72→}

    74→function formatBytes(bytes: number) {
    75→  if (!bytes) return "0 B";
    76→  if (bytes < 1024) return `${bytes} B`;
    77→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
    78→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
    79→}

   416→function ImportTab() {
   417→  const importItems = useVault((s) => s.importItems);
   418→  const setOpen = useVault((s) => s.setSettingsOpen);
   419→  const [selectedSource, setSelectedSource] = React.useState<ImportSource | null>(null);
   420→  const [file, setFile] = React.useState<File | null>(null);
   421→  const [filePreview, setFilePreview] = React.useState("");
   422→  const [detectedFormat, setDetectedFormat] = React.useState("");
   423→  const [importing, setImporting] = React.useState(false);
   424→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   425→
   426→  const handlePickSource = (source: ImportSource) => {
   427→    setSelectedSource(source);
   428→    setFile(null);
   429→    setFilePreview("");
   430→    setDetectedFormat("");
   431→    // Trigger file picker immediately.
   432→    setTimeout(() => fileInputRef.current?.click(), 50);
   433→  };
   434→
   435→  const handleFile = async (f: File) => {
   436→    setFile(f);
   437→    const text = await f.text();
   438→    setFilePreview(text.slice(0, 600));
   439→    setDetectedFormat(detectFormat(f.name, text));
   440→  };
   441→
   442→  const handleImport = async () => {
   443→    if (!file) return;
   444→    setImporting(true);
   445→    try {
   446→      const text = await file.text();
   447→      const result = await importItems(file.name, text);
   448→      toast.success(`Imported ${result.imported} item${result.imported === 1 ? "" : "s"}`, {
   449→        description:
   450→          result.skipped > 0
   451→            ? `${result.skipped} skipped. ${result.warnings[0] ?? ""}`
   452→            : undefined,
   453→      });
   454→      setOpen(false);
   455→    } catch (err) {
   456→      console.error(err);
   457→      toast.error("Import failed", { description: "The file may be corrupted or unsupported." });
   458→    } finally {
   459→      setImporting(false);
   460→    }
   461→  };
   462→
   463→  return (
   464→    <div className="mx-auto max-w-2xl space-y-6 px-4 py-6 md:px-6">
   465→      <section className="space-y-3">
   466→        <div>
   467→          <h3 className="text-sm font-medium">Import from another password manager</h3>
   468→          <p className="text-xs text-muted-foreground">
   469→            Select where your data is coming from, then choose the exported file.
   470→          </p>
   471→        </div>
   472→        <div className="grid grid-cols-1 gap-2 sm:grid-cols-3">
   473→          {IMPORT_SOURCES.map((source) => {
   474→            const active = selectedSource?.id === source.id;
   475→            return (
   476→              <button
   477→                key={source.id}
   478→                type="button"
   479→                onClick={() => handlePickSource(source)}
   480→                className={cn(
   481→                  "flex flex-col items-start gap-2 rounded-xl border p-3 text-left transition-all duration-100",
   482→                  active
   483→                    ? "border-primary bg-primary/5 ring-1 ring-primary/40"
   484→                    : "border-border hover:bg-muted/30",
   485→                )}
   486→                aria-pressed={active}
   487→              >
   488→                <div className="flex items-center gap-2">
   489→                  <source.icon className={cn("h-5 w-5", source.color)} />
   490→                  <span className="text-sm font-medium">{source.name}</span>
   491→                </div>
   492→                <span className="text-[10px] text-muted-foreground">{source.fileType}</span>
   493→              </button>
   494→            );
   495→          })}
   496→        </div>
   497→      </section>
   498→
   499→      {/* Hidden file input — triggered when a source is picked */}
   500→      <input
   501→        ref={fileInputRef}
   502→        type="file"
   503→        accept={selectedSource?.accept ?? ".json,.csv"}
   504→        className="sr-only"
   505→        onChange={(e) => e.target.files?.[0] && handleFile(e.target.files[0])}
   506→      />
   507→
   508→      {file && (
   509→        <section className="space-y-3 rounded-xl border border-border bg-muted/20 p-4">
   510→          <div className="flex items-center gap-2 text-xs">
   511→            <FileUp className="h-4 w-4 text-muted-foreground" />
   512→            <span className="truncate font-medium text-foreground">{file.name}</span>
   513→          </div>
   514→          <div className="flex items-center gap-2 text-xs">
   515→            <span className="text-muted-foreground">Detected format:</span>
   516→            <span className="rounded bg-primary/10 px-1.5 py-0.5 font-medium text-primary">
   517→              {detectedFormat}
   518→            </span>
   519→          </div>
   520→          {filePreview && (
   521→            <pre className="lcked-scroll max-h-32 overflow-auto rounded-lg bg-muted/50 p-2 text-[10px] text-muted-foreground">
   522→              {filePreview}
   523→            </pre>
   524→          )}
   525→          <Button className="w-full" disabled={importing} onClick={handleImport}>
   526→            {importing ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Upload className="mr-2 h-4 w-4" />}
   527→            Import items
   528→          </Button>
   529→        </section>
   530→      )}
   531→
   532→      <div className="flex items-start gap-2 rounded-lg border border-border/60 bg-muted/10 px-3 py-2 text-xs text-muted-foreground">
   533→        <AlertTriangle className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   534→        <span>
   535→          Exported files from other managers are read locally and never uploaded.
   536→          After import, delete the original export file for safety.
   537→        </span>
   538→      </div>
   539→    </div>
   540→  );
   541→}

   551→  const [format, setFormat] = React.useState<ExportFormat>("pgp");
   552→  const [passphrase, setPassphrase] = React.useState("");
   553→  const [confirmPassphrase, setConfirmPassphrase] = React.useState("");
   554→  const [csvConfirm, setCsvConfirm] = React.useState(false);
   555→  const [busy, setBusy] = React.useState(false);
   556→
   557→  const passphraseValid = passphrase.length >= 8;
   558→  const passphrasesMatch = passphrase === confirmPassphrase && passphraseValid;
   559→  const canExport = format === "csv" || (passphraseValid && passphrasesMatch);
   560→
   561→  const stamp = () => new Date().toISOString().slice(0, 10);
   562→
   563→  const handleExport = async () => {
   564→    if (!canExport) {
   565→      if (!passphraseValid) {
   566→        toast.error("Passphrase must be at least 8 characters");
   567→        return;
   568→      }
   569→      if (!passphrasesMatch) {
   570→        toast.error("Passphrases don't match");
   571→        return;
   572→      }
   573→      return;
   574→    }
   575→    setBusy(true);
   576→    try {
   577→      if (format === "pgp") {
   578→        const json = await exportEncrypted(passphrase);
   579→        download(`lcked-vault-${stamp()}.json`, json, "application/json");
   580→        toast.success("Encrypted export downloaded", {
   581→          description: "Keep this file and the passphrase safe — both are required to restore.",
   582→        });
   583→      } else if (format === "zip") {
   584→        // ZIP uses the same encrypted JSON payload, wrapped conceptually as
   585→        // a protected archive. We emit the encrypted JSON (the passphrase is
   586→        // embedded in the encryption envelope) with a .zip-style extension so
   587→        // the user's workflow matches. Real ZIP encryption would need a
   588→        // client-side zip lib; the encrypted JSON itself is already AES-GCM.
   589→        const json = await exportEncrypted(passphrase);
   590→        download(`lcked-vault-${stamp()}.lckz`, json, "application/octet-stream");
   591→        toast.success("Encrypted archive downloaded", {
   592→          description: "Passphrase-protected — keep it safe.",
   593→        });
   594→      } else {
   595→        // CSV
   596→        const csv = exportCsv();
   597→        download(`lcked-vault-${stamp()}.csv`, csv, "text/csv");
   598→        toast.success("CSV export downloaded");
   599→        setCsvConfirm(false);
   600→      }
   601→      setPassphrase("");
   602→      setConfirmPassphrase("");
   603→      setOpen(false);
   604→    } catch (err) {
   605→      console.error(err);
   606→      toast.error("Export failed");
   607→    } finally {
   608→      setBusy(false);
   609→    }
   610→  };
   611→
   612→  return (
   613→    <div className="mx-auto max-w-2xl space-y-6 px-4 py-6 md:px-6">
   614→      <section className="space-y-3">
   615→        <div>
   616→          <h3 className="text-sm font-medium">Export format</h3>
   617→          <p className="text-xs text-muted-foreground">
   618→            {items.length} item{items.length === 1 ? "" : "s"} in your vault. Choose a format below.
   619→          </p>
   620→        </div>
   621→        <div className="space-y-2">
   622→          {EXPORT_OPTIONS.map((opt) => {
   623→            const active = format === opt.id;
   624→            return (
   625→              <button
   626→                key={opt.id}
   627→                type="button"
   628→                onClick={() => { setFormat(opt.id); setCsvConfirm(false); }}
   629→                className={cn(
   630→                  "flex w-full items-start gap-3 rounded-xl border p-3 text-left transition-all duration-100",
   631→                  active
   632→                    ? "border-primary bg-primary/5 ring-1 ring-primary/40"
   633→                    : "border-border hover:bg-muted/30",
   634→                )}
   635→                aria-pressed={active}
   636→              >
   637→                <opt.icon className={cn("mt-0.5 h-5 w-5 shrink-0", opt.color)} />
   638→                <div className="min-w-0 flex-1">
   639→                  <div className="flex items-center gap-2">
   640→                    <span className="text-sm font-medium">{opt.name}</span>
   641→                    {opt.badge && (
   642→                      <span className="rounded-full bg-emerald-500/15 px-2 py-0.5 text-[10px] font-medium text-emerald-400">
   643→                        {opt.badge}
   644→                      </span>
   645→                    )}
   646→                    {active && <Check className="h-3.5 w-3.5 text-primary" />}
   647→                  </div>
   648→                  <div className="text-xs text-muted-foreground">{opt.description}</div>
   649→                </div>
   650→              </button>
   651→            );
   652→          })}
   653→        </div>
   654→      </section>
   655→
   656→      {/* CSV warning confirmation */}
   657→      {format === "csv" && (
   658→        <AlertDialog open={csvConfirm} onOpenChange={setCsvConfirm}>
   659→          <div className="rounded-xl border border-amber-500/40 bg-amber-500/10 p-4">
   660→            <div className="flex items-start gap-2">
   661→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" />
   662→              <div className="text-xs text-amber-200/90">
   663→                The CSV will contain your passwords in plain text. Only do this on a
   664→                trusted device and delete the file after use. No passphrase protects a CSV.
   665→              </div>
   666→            </div>
   667→            <Button
   668→              className="mt-3 w-full bg-amber-500 text-white hover:bg-amber-600"
   669→              disabled={items.length === 0 || busy}
   670→              onClick={handleExport}
   671→            >
   672→              {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Download className="mr-2 h-4 w-4" />}
   673→              I understand, export CSV
   674→            </Button>
   675→          </div>
   676→        </AlertDialog>
   677→      )}
   678→
   679→      {/* Passphrase (PGP + ZIP only) */}
   680→      {format !== "csv" && (
   681→        <section className="space-y-3">
   682→          <div className="flex items-center gap-2 text-sm font-medium">
   683→            <Lock className="h-4 w-4 text-muted-foreground" />
   684→            Passphrase
   685→            <span className="text-[10px] font-normal text-red-400">required</span>
   686→          </div>
   687→          <div className="space-y-2">
   688→            <div className="space-y-1.5">
   689→              <Label htmlFor="exp-pass" className="text-xs text-muted-foreground">Passphrase</Label>
   690→              <Input
   691→                id="exp-pass"
   692→                type="password"
   693→                value={passphrase}
   694→                onChange={(e) => setPassphrase(e.target.value)}
   695→                placeholder="At least 8 characters"
   696→                className="font-secret"
   697→              />
   698→            </div>
   699→            <div className="space-y-1.5">
   700→              <Label htmlFor="exp-pass2" className="text-xs text-muted-foreground">Confirm passphrase</Label>
   701→              <Input
   702→                id="exp-pass2"
   703→                type="password"
   704→                value={confirmPassphrase}
   705→                onChange={(e) => setConfirmPassphrase(e.target.value)}
   706→                placeholder="Repeat passphrase"
   707→                className="font-secret"
   708→              />
   709→              {confirmPassphrase.length > 0 && !passphrasesMatch && (
   710→                <p className="text-[11px] text-red-400">Passphrases don&apos;t match</p>
   711→              )}
   712→            </div>
   713→            <p className="text-[11px] text-muted-foreground">
   714→              This passphrase is independent of your master password. You&apos;ll need it to restore.
   715→            </p>
   716→          </div>
   717→          <Button
   718→            className="w-full"
   719→            disabled={busy || !canExport || items.length === 0}
   720→            onClick={handleExport}
   721→          >
   722→            {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Download className="mr-2 h-4 w-4" />}
   723→            Export {format === "pgp" ? "encrypted file" : "archive"}
   724→          </Button>
   725→        </section>
   726→      )}
   727→    </div>
   728→  );
   729→}
