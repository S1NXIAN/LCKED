   140→function download(filename: string, content: string, mime: string) {
   141→  const blob = new Blob([content], { type: mime });
   142→  const url = URL.createObjectURL(blob);
   143→  const a = document.createElement("a");
   144→  a.href = url;
   145→  a.download = filename;
   146→  a.click();
   147→  URL.revokeObjectURL(url);
   148→}

   150→function formatBytes(bytes: number) {
   151→  if (!bytes) return "0 B";
   152→  if (bytes < 1024) return `${bytes} B`;
   153→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   154→  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
   155→}

   791→  const [detectedFmt, setDetectedFmt] = React.useState<string>("");
   792→
   793→  const triggerFilePicker = (src: ImportSource) => {
   794→    setPendingSource(src);
   795→    fileInputRef.current?.click();

   797→

   799→    setFile(f);
   800→    try {
   801→      const text = await f.text();
   802→      setDetectedFmt(detectFormat(f.name, text));
   803→    } catch {
   804→      setDetectedFmt("");
   805→    }
   806→  };
   807→
   808→  const handleImport = async () => {
   809→    if (!file) return;
   810→    setImporting(true);

   818→            : undefined,
   819→      });
   820→      setFile(null);
   821→      setPendingSource(null);
   822→      setDetectedFmt("");
