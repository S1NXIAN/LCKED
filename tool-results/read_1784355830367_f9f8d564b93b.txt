    92→    92→/* ----------------------- ActiveRowHighlight (rAF spring) --------------- */
    93→    93→
    94→    94→/**
    95→    95→ * Single persistent highlight element rendered ONCE inside the <ul>.
    96→    96→ *
    97→    97→ * Slides toward the active item's bounding box via a requestAnimationFrame
    98→    98→ * spring (exponential lerp, factor 0.22). Uses the `animateRef` pattern —
    99→    99→ * the latest step function is stashed in a ref so external listeners
   100→   100→ * (scroll, MutationObserver) can always kick the loop without re-binding.
   101→   101→ *
   102→   102→ * On `activeId` change: measure + start the animation. If the item isn't
   103→   103→ * found in the DOM, `setVisible(false)` and bail. A MutationObserver re-
   104→   104→ * checks when the list DOM changes; a scroll listener re-measures on
   105→   105→ * scroll so the highlight stays glued to the active row.
   106→   106→ */
   107→   107→function ActiveRowHighlight({
   108→   108→  listRef,
   109→   109→  activeId,
   110→   110→}: {
   111→   111→  listRef: React.RefObject<HTMLUListElement | null>;
   112→   112→  activeId: string | null;
   113→   113→}) {
   114→   114→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   115→   115→  const rafRef = React.useRef<number | null>(null);
   116→   116→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   117→   117→  const wasVisibleRef = React.useRef(false);
   118→   118→  const [visible, setVisible] = React.useState(false);
   119→   119→  // animateRef holds the latest step function so external listeners can
   120→   120→  // kick it without re-binding their own dependencies.
   121→   121→  const animateRef = React.useRef<(() => void) | null>(null);
   122→   122→
   123→   123→  // Measure the active item's rect relative to the <ul>. If not found,
   124→   124→  // hide the highlight.
   125→   125→  const measure = React.useCallback(() => {
   126→   126→    const list = listRef.current;
   127→   127→    if (!activeId || !list) {
   128→   128→      targetRef.current = null;
   129→   129→      setVisible(false);
   130→   130→      return;
   131→   131→    }
   132→   132→    const el = list.querySelector<HTMLElement>(
   133→   133→      `[data-item-id="${CSS.escape(activeId)}"]`,
   134→   134→    );
   135→   135→    if (!el) {
   136→   136→      targetRef.current = null;
   137→   137→      setVisible(false);
   138→   138→      return;
   139→   139→    }
   140→   140→    const r = el.getBoundingClientRect();
   141→   141→    const lr = list.getBoundingClientRect();
   142→   142→    const next = {
   143→   143→      x: r.left - lr.left,
   144→   144→      y: r.top - lr.top,
   145→   145→      w: r.width,
   146→   146→      h: r.height,
   147→   147→    };
   148→   148→    // First activation (or returning from invisible): snap to target so
   149→   149→    // we don't slide in from (0, 0).
   150→   150→    if (!wasVisibleRef.current) {
   151→   151→      const hl = highlightRef.current;
   152→   152→      if (hl) {
   153→   153→        hl.style.transform = `translate(${next.x}px, ${next.y}px)`;
   154→   154→        hl.style.width = `${next.w}px`;
   155→   155→        hl.style.height = `${next.h}px`;
   156→   156→      }
   157→   157→    }
   158→   158→    targetRef.current = next;
   159→   159→    setVisible(true);
   160→   160→  }, [activeId, listRef]);
   161→   161→
   162→   162→  // (Re)define the animation step on every render so it captures the
   163→   163→  // latest refs. Store it in animateRef for external listeners.
   164→   164→  React.useEffect(() => {
   165→   165→    const step = () => {
   166→   166→      const hl = highlightRef.current;
   167→   167→      const target = targetRef.current;
   168→   168→      const list = listRef.current;
   169→   169→      if (!hl || !target || !list) {
   170→   170→        rafRef.current = null;
   171→   171→        return;
   172→   172→      }
   173→   173→      const cur = hl.getBoundingClientRect();
   174→   174→      const lr = list.getBoundingClientRect();
   175→   175→      const cx = cur.left - lr.left;
   176→   176→      const cy = cur.top - lr.top;
   177→   177→      const FACTOR = 0.22;
   178→   178→      const nx = cx + (target.x - cx) * FACTOR;
   179→   179→      const ny = cy + (target.y - cy) * FACTOR;
   180→   180→      const nw = cur.width + (target.w - cur.width) * FACTOR;
   181→   181→      const nh = cur.height + (target.h - cur.height) * FACTOR;
   182→   182→      const settled =
   183→   183→        Math.abs(target.x - nx) < 0.5 &&
   184→   184→        Math.abs(target.y - ny) < 0.5 &&
   185→   185→        Math.abs(target.w - nw) < 0.5 &&
   186→   186→        Math.abs(target.h - nh) < 0.5;
   187→   187→      const fx = settled ? target.x : nx;
   188→   188→      const fy = settled ? target.y : ny;
   189→   189→      const fw = settled ? target.w : nw;
   190→   190→      const fh = settled ? target.h : nh;
   191→   191→      hl.style.transform = `translate(${fx}px, ${fy}px)`;
   192→   192→      hl.style.width = `${fw}px`;
   193→   193→      hl.style.height = `${fh}px`;
   194→   194→      if (settled) {
   195→   195→        rafRef.current = null;
   196→   196→        return;
   197→   197→      }
   198→   198→      rafRef.current = requestAnimationFrame(step);
   199→   199→    };
   200→   200→    animateRef.current = step;
   201→   201→    return () => {
   202→   202→      animateRef.current = null;
   203→   203→    };
   204→   204→  }, [listRef]);
   205→   205→
   206→   206→  // Track visibility transitions for the snap-on-first-show behavior.
   207→   207→  React.useEffect(() => {
   208→   208→    wasVisibleRef.current = visible;
   209→   209→  }, [visible]);
   210→   210→
   211→   211→  // Re-measure + kick the animation if it isn't already running.
   212→   212→  const kick = React.useCallback(() => {
   213→   213→    measure();
   214→   214→    if (!rafRef.current && animateRef.current) {
   215→   215→      rafRef.current = requestAnimationFrame(animateRef.current);
   216→   216→    }
   217→   217→  }, [measure]);
   218→   218→
   219→   219→  // On activeId change: measure + start the animation. Cancel any in-
   220→   220→  // flight loop on cleanup.
   221→   221→  React.useEffect(() => {
   222→   222→    kick();
   223→   223→    return () => {
   224→   224→      if (rafRef.current) {
   225→   225→        cancelAnimationFrame(rafRef.current);
   226→   226→        rafRef.current = null;
   227→   227→      }
   228→   228→    };
   229→   229→  }, [activeId, kick]);
   230→   230→
   231→   231→  // MutationObserver — re-check when the list DOM changes (items added,
   232→   232→  // removed, reordered by sort, attributes toggled). Uses a microtask delay
   233→   233→  // so the DOM has time to settle after AnimatePresence exit animations.
   234→   234→  React.useEffect(() => {
   235→   235→    const list = listRef.current;
   236→   236→    if (!list) return;
   237→   237→    let pending = false;
   238→   238→    const observer = new MutationObserver(() => {
   239→   239→      if (pending) return;
   240→   240→      pending = true;
   241→   241→      requestAnimationFrame(() => {
   242→   242→        pending = false;
   243→   243→        kick();
   244→   244→      });
   245→   245→    });
   246→   246→    observer.observe(list, {
   247→   247→      childList: true,
   248→   248→      subtree: true,
   249→   249→      attributes: true,
   250→   250→      attributeFilter: ["data-item-id"],
   251→   251→    });
   252→   252→    return () => observer.disconnect();
   253→   253→  }, [listRef, kick]);
   254→   254→
   255→   255→  // Scroll listener — re-measure + re-animate on scroll so the highlight
   256→   256→  // stays glued to the active row (both inside the scroll viewport and
   257→   257→  // when the page itself scrolls).
   258→   258→  React.useEffect(() => {
   259→   259→    const list = listRef.current;
   260→   260→    if (!list) return;
   261→   261→    const container =
   262→   262→      list.closest('[data-radix-scroll-area-viewport]') ??
   263→   263→      list.parentElement ??
   264→   264→      null;
   265→   265→    const handler = () => kick();
   266→   266→    container?.addEventListener("scroll", handler, { passive: true });
   267→   267→    window.addEventListener("scroll", handler, { passive: true });
   268→   268→    return () => {
   269→   269→      container?.removeEventListener("scroll", handler);
   270→   270→      window.removeEventListener("scroll", handler);
   271→   271→    };
   272→   272→  }, [listRef, kick]);
   273→   273→
   274→   274→  if (!visible) return null;
   275→   275→  return (
   276→   276→    <div
   277→   277→      ref={highlightRef}
   278→   278→      aria-hidden
   279→   279→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   280→   280→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   281→   281→    />
   282→   282→  );
   283→   283→}

   286→   286→
   287→   287→export function ItemList({
   288→   288→  filter,
   289→   289→  setFilter,
   290→   290→  activeVault: activeVaultProp,
   291→   291→  onMobileBack,
   292→   292→}: {
   293→   293→  /** Type filter (All / Login / Note / Card / Identity). */
   294→   294→  filter: FilterType;
   295→   295→  setFilter: (f: FilterType) => void;
   296→   296→  /** Active vault filter from the store — passed explicitly so the list re-
   297→   297→   *  renders when the sidebar switches between All/Favorites/Trash/vault. */
   298→   298→  activeVault?: string;
   299→   299→  /** Mobile-only callback when the user taps Back from a Trash detail. */
   300→   300→  onMobileBack?: () => void;
   301→   301→}) {
   302→   302→  // Store hooks — we read activeVault directly here as well so the list
   303→   303→  // always re-renders even if a parent forgets to thread the prop through.
   304→   304→  const items = useVault((s) => s.items);
   305→   305→  const vaults = useVault((s) => s.vaults);
   306→   306→  const selectedId = useVault((s) => s.selectedId);
   307→   307→  const setSelected = useVault((s) => s.setSelected);
   308→   308→  const searchQuery = useVault((s) => s.searchQuery);
   309→   309→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   310→   310→  const trashItem = useVault((s) => s.trashItem);
   311→   311→  const restoreItem = useVault((s) => s.restoreItem);
   312→   312→  const permanentlyDeleteItem = useVault((s) => s.permanentlyDeleteItem);
   313→   313→  const moveItemToVault = useVault((s) => s.moveItemToVault);
   314→   314→  const toggleFavorite = useVault((s) => s.toggleFavorite);
   315→   315→  const sortFavoritesFirst = useVault((s) => s.settings.sortFavoritesFirst);
   316→   316→
   317→   317→  const activeVault = activeVaultProp ?? useVault.getState().activeVault;
   318→   318→  const isTrashView = activeVault === "trash";
   319→   319→
   320→   320→  // Sort persistence — read the initial value from localStorage so the
   321→   321→  // user's last sort preference survives reloads.
   322→   322→  const [sort, setSortState] = React.useState<SortKey>(() => {
   323→   323→    if (typeof window === "undefined") return "newest";
   324→   324→    const stored = window.localStorage.getItem("lcked-sort") as SortKey | null;
   325→   325→    return stored === "newest" || stored === "oldest" || stored === "alphabetical"
   326→   326→      ? stored
   327→   327→      : "newest";
   328→   328→  });
   329→   329→  const setSort = React.useCallback((next: SortKey) => {
   330→   330→    setSortState(next);
   331→   331→    try {
   332→   332→      window.localStorage.setItem("lcked-sort", next);
   333→   333→    } catch {
   334→   334→      // localStorage may be denied (private mode, etc.) — best-effort.
   335→   335→    }
   336→   336→  }, []);
   337→   337→
   338→   338→  const [multiSelect, setMultiSelect] = React.useState(false);
   339→   339→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
   340→   340→  // Reset multi-select whenever we leave multi-select mode.
   341→   341→  React.useEffect(() => {
   342→   342→    if (!multiSelect) setSelectedIds(new Set());
   343→   343→  }, [multiSelect]);
   344→   344→
   345→   345→  // Deferred search query — keeps typing responsive on large vaults by
   346→   346→  // letting the filter/sort work happen at lower priority.
   347→   347→  const deferredSearch = React.useDeferredValue(searchQuery);
   348→   348→
   349→   349→  // List ref for the ActiveRowHighlight.
   350→   350→  const listRef = React.useRef<HTMLUListElement | null>(null);
   351→   351→
   352→   352→  /* --------------------------- filtering + sorting --------------------------- */
   353→   353→  const filtered = React.useMemo(() => {
   354→   354→    let list = items;
   355→   355→    // Vault-level filter.
   356→   356→    if (activeVault === "trash") list = list.filter((i) => i.trashed);
   357→   357→    else if (activeVault === "favorites") list = list.filter((i) => !i.trashed && i.favorite);
   358→   358→    else if (activeVault && activeVault !== "all")
   359→   359→      list = list.filter((i) => !i.trashed && i.vaultId === activeVault);
   360→   360→    else list = list.filter((i) => !i.trashed);
   361→   361→    // Type filter (secondary, list-header dropdown).
   362→   362→    if (filter !== "all" && typeof filter === "string") {
   363→   363→      list = list.filter((i) => i.type === filter);
   364→   364→    }
   365→   365→    // Fuzzy search.
   366→   366→    list = searchItems(list, deferredSearch);
   367→   367→    // Sort.
   368→   368→    const sorted = [...list];
   369→   369→    sorted.sort((a, b) => {
   370→   370→      // Pin ALWAYS pins to top (unconditional). The schema doesn't have a
   371→   371→      // dedicated `pinned` flag yet — this branch is a placeholder so the
   372→   372→      // behavior is correct the moment one is added.
   373→   373→      const aPin = false;
   374→   374→      const bPin = false;
   375→   375→      // Favorite only when sortFavoritesFirst is on.
   376→   376→      const aFav = sortFavoritesFirst && a.favorite;
   377→   377→      const bFav = sortFavoritesFirst && b.favorite;
   378→   378→      // Favorite takes priority over pin when both active.
   379→   379→      if (aFav !== bFav) return aFav ? -1 : 1;
   380→   380→      if (aPin !== bPin) return aPin ? -1 : 1;
   381→   381→      // Primary sort.
   382→   382→      if (sort === "newest") return b.updatedAt - a.updatedAt;
   383→   383→      if (sort === "oldest") return a.updatedAt - b.updatedAt;
   384→   384→      return a.name.localeCompare(b.name);
   385→   385→    });
   386→   386→    return sorted;
   387→   387→  }, [items, activeVault, filter, deferredSearch, sort, sortFavoritesFirst]);
   388→   388→
   389→   389→  // Scroll-into-view when selection changes via keyboard.
   390→   390→  React.useEffect(() => {
   391→   391→    if (!selectedId) return;
   392→   392→    const el = document.querySelector(`[data-item-id="${CSS.escape(selectedId)}"]`);
   393→   393→    el?.scrollIntoView({ block: "nearest", behavior: "smooth" });
   394→   394→  }, [selectedId]);
   395→   395→
   396→   396→  /* --------------------------- multi-select actions --------------------------- */
   397→   397→  const toggleSelected = (id: string) => {
   398→   398→    setSelectedIds((prev) => {
   399→   399→      const next = new Set(prev);
   400→   400→      if (next.has(id)) next.delete(id);
   401→   401→      else next.add(id);
   402→   402→      return next;
   403→   403→    });
   404→   404→  };
   405→   405→  const selectAll = () => setSelectedIds(new Set(filtered.map((i) => i.id)));
   406→   406→  const deselectAll = () => setSelectedIds(new Set());
   407→   407→
   408→   408→  const handleMultiTrash = async () => {
   409→   409→    if (selectedIds.size === 0) return;
   410→   410→    const ids = Array.from(selectedIds);
   411→   411→    try {
   412→   412→      await Promise.all(ids.map((id) => trashItem(id)));
   413→   413→      toast.success(`Moved ${ids.length} to Trash`);
   414→   414→      setMultiSelect(false);
   415→   415→    } catch {
   416→   416→      toast.error("Could not move items");
   417→   417→    }
   418→   418→  };
   419→   419→  const handleMultiMove = async (vaultId: string | null) => {
   420→   420→    if (selectedIds.size === 0) return;
   421→   421→    const ids = Array.from(selectedIds);
   422→   422→    try {
   423→   423→      await Promise.all(ids.map((id) => moveItemToVault(id, vaultId)));
   424→   424→      const label =
   425→   425→        vaultId === null
   426→   426→          ? "All Items"
   427→   427→          : vaults.find((v) => v.id === vaultId)?.name ?? "vault";
   428→   428→      toast.success(`Moved ${ids.length} to ${label}`);
   429→   429→      setMultiSelect(false);
   430→   430→    } catch {
   431→   431→      toast.error("Could not move items");
   432→   432→    }
   433→   433→  };
   434→   434→
   435→   435→  /* ------------------------------- item helpers ------------------------------- */
   436→   436→  // Helper used by the empty-area context menu (and any other caller that
   437→   437→  // wants to spawn the editor pre-seeded with a type).
   438→   438→  const createItem = React.useCallback((type: ItemType) => {
   439→   439→    stashNewItemType(type);
   440→   440→    setEditorOpen(true);
   441→   441→  }, [setEditorOpen]);
   442→   442→
   443→   443→  const copyField = React.useCallback(
   444→   444→    async (value: string | undefined, label: string) => {
   445→   445→      if (!value) return;
   446→   446→      try {
   447→   447→        await copyWithAutoClear(value, label);
   448→   448→        toast.success(`${label} copied`, { description: "Auto-clears in 30s" });
   449→   449→      } catch {
   450→   450→        toast.error("Clipboard access denied");
   451→   451→      }
   452→   452→    },
   453→   453→    [],
   454→   454→  );
   455→   455→
   456→   456→  /* ------------------------------- rendering ------------------------------- */
   457→   457→  return (
   458→   458→    <div className="flex h-full flex-col">
   459→   459→      {/* Filter bar — type + sort + multi-select menu */}
