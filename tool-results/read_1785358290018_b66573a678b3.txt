     3→/**
     4→ * LCKED — VaultsSidebar
     5→ * ---------------------------------------------------------------------------
     6→ * Proton Pass–style vault selector. Renders the fixed rows (All / Favorites /
     7→ * Trash) plus every user-defined vault. Each row is a drag-and-drop drop
     8→ * target so an item dragged from the list can be moved between vaults.
     9→ *
    10→ * Pattern notes:
    11→ *   • Rows that host a ⋮ dropdown are rendered as <div role="button"> (NOT
    12→ *     <button>) so the trigger button doesn't nest inside another button —
    13→ *     nested buttons are invalid HTML and break accessibility tree.
    14→ *   • A fixed-width slot is reserved for the menu button on EVERY row (even
    15→ *     the menu-less ones) so all the right-aligned counts line up.
    16→ *   • "Rename" opens the vault editor sheet via setVaultEditorOpen(true, id);
    17→ *     "Delete" calls deleteVault (which orphan-rescues any items inside).
    18→ *   • Each row wrapper carries a `data-vault-key` attribute consumed by the
    19→ *     shared sliding ActiveHighlight (rAF spring — same component used by
    20→ *     item-list). TWO instances are rendered: one inside the scroll area's
    21→ *     content wrapper (for All / Favorites / custom vaults) and one inside
    22→ *     the Trash wrapper (Trash lives outside the scroll area). Because the
    23→ *     scroll-area instance lives INSIDE the scrolling content, it is
    24→ *     naturally clipped by `overflow` when the active vault row scrolls out
    25→ *     of view — exactly like the item-list indicator. No floating overlay.
    26→ *   • CustomVaultRow is wrapped in <ContextMenu> with the same items as the
    27→ *     3-dots dropdown: Edit, Move all items (submenu), Hide vault, Delete.
    28→ */

    90→/* ----------------------- drag-drop helper (shared) ----------------------- */
    91→
    92→/**
    93→ * Extract item IDs from a drag-drop event. Handles BOTH:
    94→ *   - `text/lcked-items` → JSON array of IDs (multi-select drag)
    95→ *   - `text/lcked-item`  → single ID string (classic single-item drag)
    96→ * Returns an empty array if neither is present.
    97→ */
    98→function parseDraggedIds(e: React.DragEvent): string[] {
    99→  // Multi-select drag carries a JSON array of IDs.
   100→  const multi = e.dataTransfer.getData("text/lcked-items");
   101→  if (multi) {
   102→    try {
   103→      const ids = JSON.parse(multi);
   104→      if (Array.isArray(ids) && ids.every((id) => typeof id === "string")) {
   105→        return ids as string[];
   106→      }
   107→    } catch {
   108→      // Malformed JSON — fall through to single-item.
   109→    }
   110→  }
   111→  // Single-item drag.
   112→  const single = e.dataTransfer.getData("text/lcked-item");
   113→  return single ? [single] : [];
   114→}

   118→function exitMultiSelect() {
   119→  window.dispatchEvent(new CustomEvent("lcked:exit-multi-select"));

   124→export interface VaultIconProps {
   125→  /** Vault icon id (see VAULT_ICONS). */
   126→  icon: string;
   127→  /** Vault color id (see VAULT_COLORS). */
   128→  color: string;
   129→  /** Pixel size of the square swatch. Default 28. */
   130→  size?: number;
   131→  className?: string;

   133→
   134→/**
   135→ * Render a vault's icon inside a colored rounded swatch. The fill is the
   136→ * vault color at 18% opacity; the icon glyph uses the full hex for contrast.
   137→ * Used by VaultsSidebar rows AND the create-vault-dialog picker preview.
   138→ * The Lucide-component lookup lives in `./vault-lucide-icons` so both this
   139→ * module and the picker share a single source of truth.
   140→ */
   141→export function VaultIcon({ icon, color, size = 28, className, bare = false }: VaultIconProps & { bare?: boolean }) {
   142→  const hex = vaultColorHex(color);
   143→  const Resolved = VAULT_LUCIDE_BY_ID[icon] ?? Home;
   144→  const px = `${size}px`;
   145→  const glyph = Math.round(size * 0.55);
   146→  if (bare) {
   147→    return (
   148→      <Resolved
   149→        size={glyph}
   150→        strokeWidth={2}
   151→        style={{ color: hex }}
   152→        className={cn("shrink-0", className)}
   153→        aria-hidden="true"
   154→      />
   155→    );
   156→  }
   157→  return (
   158→    <span
   159→      className={cn("inline-flex shrink-0 items-center justify-center rounded-lg", className)}
   160→      style={{
   161→        width: px,
   162→        height: px,
   163→        backgroundColor: `${hex}29`, // ~16% opacity
   164→        color: hex,
   165→      }}
   166→      aria-hidden="true"
   167→    >
   168→      <Resolved size={glyph} strokeWidth={2} />
   169→    </span>
   170→  );
   171→}

   178→interface VaultRowProps {
   179→  /** Glyph rendered in the swatch. Pass a Lucide component for fixed rows. */
   180→  icon: React.ReactNode;
   181→  /** Row label. */
   182→  label: string;
   183→  /** Item count shown on the right. */
   184→  count: number;
   185→  /** Whether this row is the active filter. */
   186→  active: boolean;
   187→  /** Click handler — sets the active vault filter. */
   188→  onSelect: () => void;
   189→  /** Optional drag-and-drop target vault id (null = no drop). */
   190→  dropVaultId?: string | null;
   191→  /** Optional amber-tinted style (used by Trash when non-empty). */
   192→  warn?: boolean;
   193→  /** Optional ⋮ menu node (rendered in the reserved slot). */
   194→  menu?: React.ReactNode;
   195→  /** Whether the row is being dragged over. */
   196→  dragOver?: boolean;
   197→}
   198→
   199→/**
   200→ * A single sidebar row. When `menu` is provided we render a div instead of a
   201→ * button to avoid invalid nested-button HTML. The reserved `menuSlot` keeps
   202→ * every row's right edge aligned whether or not a menu is present.
   203→ *
   204→ * The active state keeps `text-accent-foreground` but NO `bg-accent` — the
   205→ * sliding VaultActiveHighlight provides the background, so a static one
   206→ * would visually fight the spring.
   207→ */
   208→function VaultRow({
   209→  icon,
   210→  label,
   211→  count,
   212→  active,
   213→  onSelect,
   214→  dropVaultId,
   215→  warn,
   216→  menu,
   217→  dragOver,
   218→}: VaultRowProps) {
   219→  const Tag = menu ? "div" : "button";
   220→  const tagProps = menu
   221→    ? { role: "button", tabIndex: 0 }
   222→    : { type: "button" as const };
   223→
   224→  const handleDrop = async (e: React.DragEvent) => {
   225→    if (dropVaultId === undefined) return;
   226→    e.preventDefault();
   227→    const ids = parseDraggedIds(e);
   228→    if (ids.length === 0) return;
   229→    const { moveItemsToVault } = useVault.getState();
   230→    const { moved, failed } = await moveItemsToVault(ids, dropVaultId ?? null);
   231→    if (moved > 0 && failed === 0) {
   232→      toast.success(`Moved ${moved} item${moved === 1 ? "" : "s"} to ${label}`);
   233→    } else if (moved > 0 && failed > 0) {
   234→      toast.warning(`Moved ${moved}, ${failed} failed`, { description: `${label}` });
   235→    } else if (failed > 0) {
   236→      toast.error(`Could not move ${failed} item${failed === 1 ? "" : "s"}`);
   237→    }
   238→    // Exit multi-select if more than one item was dragged.
   239→    if (ids.length > 1 && moved > 0) exitMultiSelect();
   240→  };
   241→  const handleDragOver = (e: React.DragEvent) => {
   242→    if (dropVaultId === undefined) return;
   243→    e.preventDefault();
   244→    e.dataTransfer.dropEffect = "move";
   245→  };
   246→  const handleKeyDown = (e: React.KeyboardEvent) => {
   247→    if (!menu) return;
   248→    if (e.key === "Enter" || e.key === " ") {
   249→      e.preventDefault();
   250→      onSelect();
   251→    }
   252→  };
   253→
   254→  return (
   255→    <Tag
   256→      {...tagProps}
   257→      onClick={(e: React.MouseEvent) => {
   258→        const target = e.target as HTMLElement;
   259→        if (target.closest("[data-menu-slot]")) return;
   260→        onSelect();
   261→      }}
   262→      onKeyDown={menu ? handleKeyDown : undefined}
   263→      onDrop={handleDrop}
   264→      onDragOver={handleDragOver}
   265→      aria-current={active ? "true" : undefined}
   266→      className={cn(
   267→        "group relative flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition-colors duration-100",
   268→        "focus:outline-none focus-visible:ring-2 focus-visible:ring-ring/60",
   269→        active
   270→          ? "text-accent-foreground"
   271→          : warn
   272→            ? "text-foreground hover:bg-amber-500/10"
   273→            : "text-foreground/80 hover:bg-muted/60 hover:text-foreground",
   274→        dragOver && "ring-2 ring-primary/60 ring-offset-1 ring-offset-background",
   275→      )}
   276→    >
   277→      <span className="flex h-7 w-7 shrink-0 items-center justify-center">{icon}</span>
   278→      <span className="min-w-0 flex-1">
   279→        <span className="block truncate text-sm font-medium leading-tight">{label}</span>
   280→        <span
   281→          className={cn(
   282→            "block text-xs leading-tight tabular-nums",
   283→            warn && !active ? "text-amber-400/80" : active ? "text-accent-foreground/70" : "text-muted-foreground",
   284→          )}
   285→        >
   286→          {count} {count === 1 ? "item" : "items"}
   287→        </span>
   288→      </span>
   289→      <span
   290→        data-menu-slot
   291→        className="flex h-7 w-7 shrink-0 items-center justify-center"
   292→      >
   293→        {menu}
   294→      </span>
   295→    </Tag>
   296→  );
   297→}

   375→/**
   376→ * A custom-vault row with a ⋮ menu (Edit / Move all items / Delete) AND a
   377→ * right-click ContextMenu mirroring the same items. Uses local state to
   378→ * control the delete-confirm AlertDialog; the delete is async and the
   379→ * dialog stays open (with a spinner) until it resolves (VS-3).
   380→ */
   381→function CustomVaultRow({
   382→  vault,
   383→  active,
   384→  count,
   385→  dragOver,
   386→  hoverActions,
   387→  otherVaults,
   388→  onSelect,
   389→  onRename,
   390→  onDelete,
   391→  onMoveAll,
   392→  onDragEnterRow,
   393→  onDragLeaveRow,
   394→  onDropClear,
   395→}: CustomVaultRowProps) {
   396→  const [confirmOpen, setConfirmOpen] = React.useState(false);
   397→  const [deleting, setDeleting] = React.useState(false);
   398→
   399→  const handleRename = () => onRename();
   400→  const handleDelete = () => setConfirmOpen(true);
   401→
   402→  // Await the async delete before closing the dialog (VS-3). If it fails, the
   403→  // dialog stays open and the user sees the error toast.
   404→  const confirmDelete = async () => {
   405→    setDeleting(true);
   406→    try {
   407→      await onDelete();
   408→      setConfirmOpen(false);
   409→    } catch {
   410→      // Error toast is surfaced by the parent's handleDelete.
   411→    } finally {
   412→      setDeleting(false);
   413→    }
   414→  };
   415→
   416→  return (
   417→    <div
   418→      data-vault-key={vault.id}
   419→      onDragEnter={onDragEnterRow}
   420→      onDragLeave={onDragLeaveRow}
   421→      onDrop={onDropClear}
   422→    >
   423→      <ContextMenu>
   424→        <ContextMenuTrigger asChild>
   425→          <VaultRow
   426→            icon={<VaultIcon icon={vault.icon} color={vault.color} size={28} />}
   427→            label={vault.name}
   428→            count={count}
   429→            active={active}
   430→            onSelect={onSelect}
   431→            dropVaultId={vault.id}
   432→            dragOver={dragOver}
   433→            menu={
   434→              <DropdownMenu>
   435→                <DropdownMenuTrigger asChild>
   436→                  <button
   437→                    type="button"
   438→                    className={cn(
   439→                      "flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground transition-all duration-150 hover:bg-muted hover:text-foreground",
   440→                      active && "opacity-60",
   441→                      hoverActions && "opacity-0 group-hover:opacity-100 focus-visible:opacity-100",
   442→                    )}
   443→                    aria-label={`Vault “${vault.name}” options`}
   444→                    onClick={(e) => e.stopPropagation()}
   445→                  >
   446→                    <MoreVertical className="h-4 w-4" />
   447→                  </button>
   448→                </DropdownMenuTrigger>
   449→                <DropdownMenuContent align="end" className="w-44">
   450→                  {/* Mirror the ContextMenu items inside the dropdown so both
   451→                      entry points offer identical actions. */}
   452→                  <DropdownMenuItem onSelect={handleRename}>
   453→                    <Pencil className="h-3.5 w-3.5" />
   454→                    Edit
   455→                  </DropdownMenuItem>
   456→                  <DropdownMenuSub>
   457→                    <DropdownMenuSubTrigger>
   458→                      <FolderInput className="h-3.5 w-3.5" />
   459→                      Move all items
   460→                    </DropdownMenuSubTrigger>
   461→                    <DropdownMenuSubContent className="w-48">
   462→                      <DropdownMenuItem onSelect={() => onMoveAll(null)}>
   463→                        <ShieldCheck className="h-3.5 w-3.5 text-primary" />
   464→                        All Items
   465→                      </DropdownMenuItem>
   466→                      {otherVaults.length > 0 && <DropdownMenuSeparator />}
   467→                      {otherVaults.map((v) => (
   468→                        <DropdownMenuItem key={v.id} onSelect={() => onMoveAll(v.id)}>
   469→                          <VaultIcon icon={v.icon} color={v.color} size={18} />
   470→                          <span className="truncate">{v.name}</span>
   471→                        </DropdownMenuItem>
   472→                      ))}
   473→                    </DropdownMenuSubContent>
   474→                  </DropdownMenuSub>
   475→                  <DropdownMenuSeparator />
   476→                  <DropdownMenuItem
   477→                    variant="destructive"
   478→                    onSelect={(e) => {
   479→                      e.preventDefault();
   480→                      setConfirmOpen(true);
   481→                    }}
   482→                  >
   483→                    <Trash2 className="h-3.5 w-3.5" />
   484→                    Delete vault
   485→                  </DropdownMenuItem>
   486→                </DropdownMenuContent>
   487→              </DropdownMenu>
   488→            }
   489→          />
   490→        </ContextMenuTrigger>
   491→        <ContextMenuContent className="w-44">
   492→          <VaultMenuItems
   493→            vault={vault}
   494→            otherVaults={otherVaults}
   495→            onRename={handleRename}
   496→            onDelete={handleDelete}
   497→            onMoveAll={onMoveAll}
   498→          />
   499→        </ContextMenuContent>
   500→      </ContextMenu>
   501→
   502→      <AlertDialog open={confirmOpen} onOpenChange={(o) => { if (!deleting) setConfirmOpen(o); }}>
   503→        <AlertDialogContent>
   504→          <AlertDialogHeader>
   505→            <AlertDialogTitle>Delete “{vault.name}”?</AlertDialogTitle>
   506→            <AlertDialogDescription>
   507→              The vault will be removed. Items inside it will be moved to your
   508→              default vault (not deleted). This cannot be undone.
   509→            </AlertDialogDescription>
   510→          </AlertDialogHeader>
   511→          <AlertDialogFooter>
   512→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>
   513→            <AlertDialogAction
   514→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   515→              disabled={deleting}
   516→              onClick={(e) => {
   517→                e.preventDefault();
   518→                confirmDelete();
   519→              }}
   520→            >
   521→              {deleting ? "Deleting…" : "Delete vault"}
   522→            </AlertDialogAction>
   523→          </AlertDialogFooter>
   524→        </AlertDialogContent>
   525→      </AlertDialog>
   526→    </div>
   527→  );
   528→}

   546→
   547→  // Refs for the two ActiveHighlight instances:
   548→  //   • listRef  — the relative content wrapper INSIDE the scroll area
   549→  //     (holds All / Favorites / custom vaults). The highlight rendered here
   550→  //     is clipped by the scroll area's overflow, so it disappears with the
   551→  //     active row when it scrolls out of view — exactly like item-list.
   552→  //   • trashRef — the Trash wrapper, which lives OUTSIDE the scroll area
   553→  //     (Trash is pinned at the bottom). Its highlight is always visible
   554→  //     when Trash is the active filter.
   555→  const listRef = React.useRef<HTMLDivElement | null>(null);
   556→  const trashRef = React.useRef<HTMLDivElement | null>(null);
   557→
   558→  const [overKey, setOverKey] = React.useState<string | null>(null);
   559→  const [organizeOpen, setOrganizeOpen] = React.useState(false);
   560→
   561→  const allCount = items.filter((i) => !i.trashed).length;
   562→  const favCount = items.filter((i) => !i.trashed && i.favorite).length;
   563→  const trashCount = items.filter((i) => i.trashed).length;
   564→  const vaultCount = (id: string) =>
   565→    items.filter((i) => !i.trashed && i.vaultId === id).length;
   566→
   567→  const handleDelete = async (v: VaultDef) => {
   568→    await deleteVault(v.id);
   569→    toast.success(`Deleted vault “${v.name}”`);
   570→  };
   571→
   572→  const handleClearFavorites = async () => {
   573→    const { cleared, failed } = await clearFavorites();
   574→    if (cleared > 0 && failed === 0) {
   575→      toast.success(`Cleared ${cleared} favorite${cleared === 1 ? "" : "s"}`);
   576→    } else if (cleared > 0 && failed > 0) {
   577→      toast.warning(`Cleared ${cleared}, ${failed} failed`);
   578→    } else if (failed > 0) {
   579→      toast.error(`Could not clear ${failed} favorite${failed === 1 ? "" : "s"}`);
   580→    }
   581→  };
   582→
   583→  const handleEmptyTrash = async () => {
   584→    try {
   585→      await emptyTrash();
   586→      toast.success("Trash emptied");
   587→    } catch {
   588→      toast.error("Could not empty trash");
   589→    }
   590→  };
   591→
   592→  const handleRestoreAllTrash = async () => {
   593→    const { restored, failed } = await restoreAllTrash();
   594→    if (restored > 0) toast.success(`Restored ${restored} item${restored === 1 ? "" : "s"}`);
   595→    if (failed > 0) toast.error(`Could not restore ${failed} item${failed === 1 ? "" : "s"}`);
   596→  };
   597→
   598→  const handleMoveAll = async (source: VaultDef, target: string | null) => {
   599→    const targets = items.filter((i) => !i.trashed && i.vaultId === source.id);
   600→    if (targets.length === 0) {
   601→      toast.info(`“${source.name}” has no items to move`);
   602→      return;
   603→    }
   604→    try {
   605→      await Promise.all(targets.map((i) => moveItemToVault(i.id, target)));
   606→      const label =
   607→        target === null
   608→          ? "All Items"
   609→          : vaults.find((v) => v.id === target)?.name ?? "target vault";
   610→      toast.success(`Moved ${targets.length} item${targets.length === 1 ? "" : "s"} to ${label}`);
   611→    } catch {
   612→      toast.error("Could not move items");
   613→    }
   614→  };
   615→
   616→  const renderCustomVault = (v: VaultDef) => (
   617→    <CustomVaultRow
   618→      key={v.id}
   619→      vault={v}
   620→      active={activeVault === v.id}
   621→      count={vaultCount(v.id)}
   622→      dragOver={overKey === `vault-${v.id}`}
   623→      hoverActions={hoverItemActions}
   624→      otherVaults={vaults.filter((x) => x.id !== v.id)}
   625→      onSelect={() => setActiveVault(v.id)}
   626→      onRename={() => setVaultEditorOpen(true, v.id)}
   627→      onDelete={() => handleDelete(v)}
   628→      onMoveAll={(target) => handleMoveAll(v, target)}
   629→      onDragEnterRow={() => setOverKey(`vault-${v.id}`)}
   630→      onDragLeaveRow={(e) => {
   631→        const related = e.relatedTarget as Node | null;
   632→        if (related && (e.currentTarget as Node).contains(related)) return;
   633→        setOverKey((c) => (c === `vault-${v.id}` ? null : c));
   634→      }}
   635→      onDropClear={() => setOverKey(null)}
   636→    />
   637→  );
   638→
   639→  return (
   640→    <div className="flex h-full flex-col gap-1 px-2 py-2">
   641→      {/* Header */}
   642→      <div className="relative flex items-center justify-between px-3 py-2">
   643→        <span className="text-[10px] font-semibold uppercase tracking-[0.18em] text-muted-foreground">
   644→          Vaults
   645→        </span>
   646→        <Button
   647→          size="icon"
   648→          variant="ghost"
   649→          className="h-6 w-6 text-muted-foreground hover:text-foreground"
   650→          onClick={() => setCreateVaultDialogOpen(true)}
   651→          aria-label="Create vault"
   652→        >
   653→          <Plus className="h-4 w-4" />
   654→        </Button>
   655→      </div>
   656→
   657→      {/* Scrollable vault list area — All Items, Favorites, custom vaults.
   658→          Trash is OUTSIDE this scroll area so it stays pinned at the bottom.
   659→
   660→          The highlight lives INSIDE the `relative` content wrapper below,
   661→          which itself scrolls with the content — so the highlight is
   662→          clipped by the scroll area's `overflow-y-auto` the moment the
   663→          active vault row leaves the visible viewport. No floating overlay,
   664→          no follow-the-item-off-screen: it disappears with the row, exactly
   665→          like the item-list indicator. (Shared ActiveHighlight component.) */}
   666→      <div className="lcked-scroll min-h-0 flex-1 overflow-y-auto">
   667→        <div ref={listRef} className="relative">
   668→          {/* Sliding highlight — only active for non-trash vaults. Clipped
   669→              by the scroll area's overflow when the row scrolls away. */}
   670→          <ActiveHighlight
   671→            containerRef={listRef}
   672→            activeKey={settingsOpen || activeVault === "trash" ? null : activeVault}
   673→            selectorAttr="data-vault-key"
   674→          />
   675→
   676→          {/* All items — drop = move to default vault (vaultId = null). */}
   677→          <div
   678→            data-vault-key="all"
   679→            onDragEnter={() => setOverKey("all")}
   680→            onDragLeave={(e) => {
   681→              const related = e.relatedTarget as Node | null;
   682→              if (related && (e.currentTarget as Node).contains(related)) return;
   683→              setOverKey((c) => (c === "all" ? null : c));
   684→            }}
   685→            onDrop={async (e) => {
   686→              setOverKey(null);
   687→              const ids = parseDraggedIds(e);
   688→              if (ids.length === 0) return;
   689→              const { moveItemsToVault } = useVault.getState();
   690→              const { moved, failed } = await moveItemsToVault(ids, null);
   691→              if (moved > 0) toast.success(`Moved ${moved} item${moved === 1 ? "" : "s"} to All Items`);
   692→              else if (failed > 0) toast.error(`Could not move ${failed} item${failed === 1 ? "" : "s"}`);
   693→              if (ids.length > 1 && moved > 0) exitMultiSelect();
   694→            }}
   695→            onDragOver={(e) => {
   696→              e.preventDefault();
   697→              e.dataTransfer.dropEffect = "move";
   698→            }}
   699→          >
   700→            <VaultRow
   701→              icon={
   702→                <span
   703→                  className="inline-flex h-7 w-7 items-center justify-center rounded-lg"
   704→                  style={{ backgroundColor: "#7777F829", color: "#7777F8" }}
   705→                  aria-hidden="true"
   706→                >
   707→                  <ShieldCheck className="h-4 w-4" />
   708→                </span>
   709→              }
   710→              label="All Items"
   711→              count={allCount}
   712→              active={activeVault === "all"}
   713→              onSelect={() => setActiveVault("all")}
   714→              dragOver={overKey === "all"}
   715→              menu={
   716→                <DropdownMenu>
   717→                  <DropdownMenuTrigger asChild>
   718→                    <button
   719→                      type="button"
   720→                      className={cn(
   721→                        "flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground transition-all duration-150 hover:bg-muted hover:text-foreground",
   722→                        hoverItemActions && "opacity-0 group-hover:opacity-100 focus-visible:opacity-100",
   723→                      )}
   724→                      aria-label="All items options"
   725→                      onClick={(e) => e.stopPropagation()}
   726→                    >
   727→                      <MoreVertical className="h-4 w-4" />
   728→                    </button>
   729→                  </DropdownMenuTrigger>
   730→                  <DropdownMenuContent align="end" className="w-44">
   731→                    <DropdownMenuItem onSelect={() => setOrganizeOpen(true)}>
   732→                      <LayoutGrid className="h-3.5 w-3.5" />
   733→                      Organize vault
   734→                    </DropdownMenuItem>
   735→                  </DropdownMenuContent>
   736→                </DropdownMenu>
   737→              }
   738→            />
   739→          </div>
   740→
   741→          {/* Favorites — drop = toggle favorite on. */}
   742→          <div
   743→            data-vault-key="favorites"
   744→            onDragEnter={() => setOverKey("favorites")}
   745→            onDragLeave={(e) => {
   746→              const related = e.relatedTarget as Node | null;
   747→              if (related && (e.currentTarget as Node).contains(related)) return;
   748→              setOverKey((c) => (c === "favorites" ? null : c));
   749→            }}
   750→            onDrop={async (e) => {
   751→              setOverKey(null);
   752→              const ids = parseDraggedIds(e);
   753→              if (ids.length === 0) return;
   754→              const store = useVault.getState();
   755→              // Only favorite items that aren't already favorites.
   756→              const toFav = store.items.filter((i) => ids.includes(i.id) && !i.favorite);
   757→              if (toFav.length === 0) {
   758→                toast.info(ids.length === 1 ? "Already a favorite" : "All already favorites");
   759→                return;
   760→              }
   761→              try {
   762→                await Promise.all(toFav.map((it) => store.toggleFavorite(it.id)));
   763→                toast.success(`Added ${toFav.length} item${toFav.length === 1 ? "" : "s"} to Favorites`);
   764→                if (ids.length > 1) exitMultiSelect();
   765→              } catch {
   766→                toast.error("Could not favorite items");
   767→              }
   768→            }}
   769→            onDragOver={(e) => {
   770→              e.preventDefault();
   771→              e.dataTransfer.dropEffect = "move";
   772→            }}
   773→          >
   774→            <VaultRow
   775→              icon={
   776→                <span
   777→                  className="inline-flex h-7 w-7 items-center justify-center rounded-lg"
   778→                  style={{ backgroundColor: "#FFB84D29", color: "#FFB84D" }}
   779→                  aria-hidden="true"
   780→                >
   781→                  <Star className="h-4 w-4" />
   782→                </span>
   783→              }
   784→              label="Favorites"
   785→              count={favCount}
   786→              active={activeVault === "favorites"}
   787→              onSelect={() => setActiveVault("favorites")}
   788→              dragOver={overKey === "favorites"}
   789→              menu={
   790→                <DropdownMenu>
   791→                  <DropdownMenuTrigger asChild>
   792→                    <button
   793→                      type="button"
   794→                      className={cn(
   795→                        "flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground transition-all duration-150 hover:bg-muted hover:text-foreground",
   796→                        hoverItemActions && "opacity-0 group-hover:opacity-100 focus-visible:opacity-100",
   797→                      )}
   798→                      aria-label="Favorites options"
   799→                      onClick={(e) => e.stopPropagation()}
   800→                    >
   801→                      <MoreVertical className="h-4 w-4" />
   802→                    </button>
   803→                  </DropdownMenuTrigger>
   804→                  <DropdownMenuContent align="end" className="w-44">
   805→                    <DropdownMenuItem
   806→                      variant="destructive"
   807→                      disabled={favCount === 0}
   808→                      onSelect={handleClearFavorites}
   809→                    >
   810→                      <StarOff className="h-3.5 w-3.5" />
   811→                      Clear favorites
   812→                    </DropdownMenuItem>
   813→                  </DropdownMenuContent>
   814→                </DropdownMenu>
   815→              }
   816→            />
   817→          </div>
   818→
   819→          {/* Custom vaults */}
   820→          {vaults.length > 0 && (
   821→            <div className="mt-1 flex flex-col gap-0.5">
   822→              {vaults.map(renderCustomVault)}
   823→            </div>
   824→          )}
   825→        </div>
   826→      </div>{/* End scrollable vault list area */}
   827→
   828→      {/* Trash — pinned at the bottom, OUTSIDE the scroll area.
   829→          A thin border-t separates it from the scrollable vault list.
   830→
   831→          Its own ActiveHighlight instance is only active when Trash is the
   832→          current filter. Since Trash never scrolls, this highlight is always
   833→          fully visible (no clipping needed). */}
   834→      <div
   835→        ref={trashRef}
   836→        data-vault-key="trash"
   837→        className="relative shrink-0 border-t border-border/60 pt-1"
   838→        onDragEnter={() => setOverKey("trash")}
   839→        onDragLeave={(e) => {
   840→          const related = e.relatedTarget as Node | null;
   841→          if (related && (e.currentTarget as Node).contains(related)) return;
   842→          setOverKey((c) => (c === "trash" ? null : c));
   843→        }}
   844→        onDrop={async (e) => {
   845→          setOverKey(null);
   846→          const ids = parseDraggedIds(e);
   847→          if (ids.length === 0) return;
   848→          const { trashItems } = useVault.getState();
   849→          const { moved, failed } = await trashItems(ids);
   850→          if (moved > 0) toast.success(`Moved ${moved} item${moved === 1 ? "" : "s"} to Trash`);
   851→          else if (ids.length > 0 && moved === 0) toast.info("Already in Trash");
   852→          if (failed > 0) toast.error(`Could not move ${failed} item${failed === 1 ? "" : "s"}`);
   853→          if (ids.length > 1 && moved > 0) exitMultiSelect();
   854→        }}
   855→        onDragOver={(e) => {
   856→          e.preventDefault();
   857→          e.dataTransfer.dropEffect = "move";
   858→        }}
   859→      >
   860→        <ActiveHighlight
   861→          containerRef={trashRef}
   862→          activeKey={!settingsOpen && activeVault === "trash" ? "trash" : null}
   863→          selectorAttr="data-vault-key"
   864→        />
   865→        <VaultRow
   866→          icon={
   867→            <span
   868→              className={cn(
   869→                "inline-flex h-7 w-7 items-center justify-center rounded-lg",
   870→                trashCount > 0
   871→                  ? "bg-amber-500/15 text-amber-400"
   872→                  : "bg-muted text-muted-foreground",
   873→              )}
   874→              aria-hidden="true"
   875→            >
   876→              <Trash2 className="h-4 w-4" />
   877→            </span>
   878→          }
   879→          label="Trash"
   880→          count={trashCount}
   881→          active={activeVault === "trash"}
   882→          warn={trashCount > 0}
   883→          onSelect={() => setActiveVault("trash")}
   884→          dragOver={overKey === "trash"}
   885→          menu={
   886→            <DropdownMenu>
   887→              <DropdownMenuTrigger asChild>
   888→                <button
   889→                  type="button"
   890→                  className={cn(
   891→                    "flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground transition-all duration-150 hover:bg-muted hover:text-foreground",
   892→                    hoverItemActions && "opacity-0 group-hover:opacity-100 focus-visible:opacity-100",
   893→                  )}
   894→                  aria-label="Trash options"
   895→                  onClick={(e) => e.stopPropagation()}
   896→                >
   897→                  <MoreVertical className="h-4 w-4" />
   898→                </button>
   899→              </DropdownMenuTrigger>
   900→              <DropdownMenuContent align="end" className="w-44">
   901→                <DropdownMenuItem
   902→                  onSelect={handleRestoreAllTrash}
   903→                  disabled={trashCount === 0}
   904→                >
   905→                  <RotateCcw className="h-3.5 w-3.5" />
   906→                  Restore all
   907→                </DropdownMenuItem>
   908→                <DropdownMenuSeparator />
   909→                <DropdownMenuItem
   910→                  variant="destructive"
   911→                  onSelect={handleEmptyTrash}
   912→                  disabled={trashCount === 0}
   913→                >
   914→                  <Trash2 className="h-3.5 w-3.5" />
   915→                  Empty trash
   916→                </DropdownMenuItem>
   917→              </DropdownMenuContent>
   918→            </DropdownMenu>
   919→          }
   920→        />
   921→      </div>
   922→
   923→      {/* Organize vault dialog — opened from the "All Items" 3-dots menu. */}
   924→      <OrganizeVaultDialog open={organizeOpen} onOpenChange={setOrganizeOpen} />
   925→    </div>
   926→  );
   927→}
