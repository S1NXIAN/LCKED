    65→function download(filename: string, content: string, mime: string) {
    66→  const blob = new Blob([content], { type: mime });
    67→  const url = URL.createObjectURL(blob);
    68→  const a = document.createElement("a");
    69→  a.href = url;
    70→  a.download = filename;
    71→  a.click();
    72→  URL.revokeObjectURL(url);
    73→}

   455→/* ============================== Import tab ============================== */
   456→
   457→function ImportTab() {
   458→  const importItems = useVault((s) => s.importItems);
   459→  const setOpen = useVault((s) => s.setSettingsOpen);
   460→  const [selectedSource, setSelectedSource] = React.useState<ImportSource | null>(null);
   461→  const [file, setFile] = React.useState<File | null>(null);
   462→  const [filePreview, setFilePreview] = React.useState("");
   463→  const [detectedFormat, setDetectedFormat] = React.useState("");
   464→  const [importing, setImporting] = React.useState(false);
   465→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   466→
   467→  const handlePickSource = (source: ImportSource) => {
   468→    setSelectedSource(source);
   469→    setFile(null);
   470→    setFilePreview("");
   471→    setDetectedFormat("");
   472→    setTimeout(() => fileInputRef.current?.click(), 50);
   473→  };
   474→
   475→  const handleFile = async (f: File) => {
   476→    setFile(f);
   477→    const text = await f.text();
   478→    setFilePreview(text.slice(0, 600));
   479→    setDetectedFormat(detectFormat(f.name, text));
   480→  };
   481→
   482→  const handleImport = async () => {
   483→    if (!file) return;
   484→    setImporting(true);
   485→    try {
   486→      const text = await file.text();
   487→      const result = await importItems(file.name, text);
   488→      toast.success(`Imported ${result.imported} item${result.imported === 1 ? "" : "s"}`, {
   489→        description:
   490→          result.skipped > 0
   491→            ? `${result.skipped} skipped. ${result.warnings[0] ?? ""}`
   492→            : undefined,
   493→      });
   494→      setOpen(false);
   495→    } catch (err) {
   496→      console.error(err);
   497→      toast.error("Import failed", { description: "The file may be corrupted or unsupported." });
   498→    } finally {
   499→      setImporting(false);
   500→    }
   501→  };
   502→
   503→  return (
   504→    <div className="mx-auto max-w-2xl space-y-6 px-4 py-6 md:px-6">
   505→      <section className="space-y-3">
   506→        <div>
   507→          <h3 className="text-sm font-medium">Import from another password manager</h3>
   508→          <p className="text-xs text-muted-foreground">
   509→            Select where your data is coming from, then choose the exported file.
   510→          </p>
   511→        </div>
   512→        <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   513→          {IMPORT_SOURCES.map((source) => {
   514→            const active = selectedSource?.id === source.id;
   515→            return (
   516→              <button
   517→                key={source.id}
   518→                type="button"
   519→                onClick={() => handlePickSource(source)}
   520→                className={cn(
   521→                  "flex flex-col items-center gap-2 rounded-xl border p-3 text-center transition-colors duration-100",
   522→                  active
   523→                    ? "border-primary bg-primary/5 ring-1 ring-primary/40"
   524→                    : "border-border hover:bg-muted/30",
   525→                )}
   526→                aria-pressed={active}
   527→              >
   528→                <img
   529→                  src={source.icon}
   530→                  alt=""
   531→                  className="h-7 w-7"
   532→                  width={28}
   533→                  height={28}
   534→                />
   535→                <div>
   536→                  <div className="text-xs font-medium">{source.name}</div>
   537→                  <div className="text-[10px] text-muted-foreground">{source.fileType}</div>
   538→                </div>
   539→              </button>
   540→            );
   541→          })}
   542→        </div>
   543→      </section>
   544→
   545→      {/* Hidden file input */}
   546→      <input
   547→        ref={fileInputRef}
   548→        type="file"
   549→        accept={selectedSource?.accept ?? ".json,.csv"}
   550→        className="sr-only"
   551→        onChange={(e) => e.target.files?.[0] && handleFile(e.target.files[0])}
   552→      />
   553→
   554→      {/* File preview — animated, no layout shift */}
   555→      <AnimatePresence initial={false}>
   556→        {file && (
   557→          <motion.section
   558→            initial={{ height: 0, opacity: 0 }}
   559→            animate={{ height: "auto", opacity: 1 }}
   560→            exit={{ height: 0, opacity: 0 }}
   561→            transition={{ duration: 0.2, ease: [0.16, 1, 0.3, 1] }}
   562→            className="overflow-hidden"
   563→          >
   564→            <div className="space-y-3 rounded-xl border border-border bg-muted/20 p-4">
   565→              <div className="flex items-center gap-2 text-xs">
   566→                <FileUp className="h-4 w-4 text-muted-foreground" />
   567→                <span className="truncate font-medium text-foreground">{file.name}</span>
   568→              </div>
   569→              <div className="flex items-center gap-2 text-xs">
   570→                <span className="text-muted-foreground">Detected format:</span>
   571→                <span className="rounded bg-primary/10 px-1.5 py-0.5 font-medium text-primary">
   572→                  {detectedFormat}
   573→                </span>
   574→              </div>
   575→              {filePreview && (
   576→                <pre className="lcked-scroll max-h-32 overflow-auto rounded-lg bg-muted/50 p-2 text-[10px] text-muted-foreground">
   577→                  {filePreview}
   578→                </pre>
   579→              )}
   580→              <Button className="w-full" disabled={importing} onClick={handleImport}>
   581→                {importing ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Upload className="mr-2 h-4 w-4" />}
   582→                Import items
   583→              </Button>
   584→            </div>
   585→          </motion.section>
   586→        )}
   587→      </AnimatePresence>
   588→
   589→      <div className="flex items-start gap-2 rounded-lg border border-border/60 bg-muted/10 px-3 py-2 text-xs text-muted-foreground">
   590→        <AlertTriangle className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   591→        <span>
   592→          Exported files from other managers are read locally and never uploaded.
   593→          After import, delete the original export file for safety.
   594→        </span>
   595→      </div>
   596→    </div>
   597→  );
   598→}

   600→/* ============================== Export tab ============================== */
   601→
   602→function ExportTab() {
   603→  const exportEncrypted = useVault((s) => s.exportEncrypted);
   604→  const exportCsv = useVault((s) => s.exportCsv);
   605→  const setOpen = useVault((s) => s.setSettingsOpen);
   606→  const items = useVault((s) => s.items);
   607→
   608→  const [format, setFormat] = React.useState<ExportFormat>("pgp");
   609→  const [passphrase, setPassphrase] = React.useState("");
   610→  const [confirmPassphrase, setConfirmPassphrase] = React.useState("");
   611→  const [busy, setBusy] = React.useState(false);
   612→
   613→  const passphraseValid = passphrase.length >= 8;
   614→  const passphrasesMatch = passphrase === confirmPassphrase && passphraseValid;
   615→  const canExport = format === "csv" || (passphraseValid && passphrasesMatch);
   616→  const isCsv = format === "csv";
   617→
   618→  const stamp = () => new Date().toISOString().slice(0, 10);
   619→
   620→  const handleExport = async () => {
   621→    if (!canExport) {
   622→      if (!passphraseValid) {
   623→        toast.error("Passphrase must be at least 8 characters");
   624→        return;
   625→      }
   626→      if (!passphrasesMatch) {
   627→        toast.error("Passphrases don't match");
   628→        return;
   629→      }
   630→      return;
   631→    }
   632→    setBusy(true);
   633→    try {
   634→      if (format === "pgp") {
   635→        const json = await exportEncrypted(passphrase);
   636→        download(`lcked-vault-${stamp()}.json`, json, "application/json");
   637→        toast.success("Encrypted export downloaded", {
   638→          description: "Keep this file and the passphrase safe — both are required to restore.",
   639→        });
   640→      } else if (format === "zip") {
   641→        const json = await exportEncrypted(passphrase);
   642→        download(`lcked-vault-${stamp()}.lckz`, json, "application/octet-stream");
   643→        toast.success("Encrypted archive downloaded", {
   644→          description: "Passphrase-protected — keep it safe.",
   645→        });
   646→      } else {
   647→        const csv = exportCsv();
   648→        download(`lcked-vault-${stamp()}.csv`, csv, "text/csv");
   649→        toast.success("CSV export downloaded");
   650→      }
   651→      setPassphrase("");
   652→      setConfirmPassphrase("");
   653→      setOpen(false);
   654→    } catch (err) {
   655→      console.error(err);
   656→      toast.error("Export failed");
   657→    } finally {
   658→      setBusy(false);
   659→    }
   660→  };
   661→
   662→  return (
   663→    <div className="mx-auto max-w-2xl space-y-6 px-4 py-6 md:px-6">
   664→      <section className="space-y-3">
   665→        <div>
   666→          <h3 className="text-sm font-medium">Export format</h3>
   667→          <p className="text-xs text-muted-foreground">
   668→            {items.length} item{items.length === 1 ? "" : "s"} in your vault. Choose a format below.
   669→          </p>
   670→        </div>
   671→        <div className="space-y-2">
   672→          {EXPORT_OPTIONS.map((opt) => {
   673→            const active = format === opt.id;
   674→            return (
   675→              <button
   676→                key={opt.id}
   677→                type="button"
   678→                onClick={() => { setFormat(opt.id); setPassphrase(""); setConfirmPassphrase(""); }}
   679→                className={cn(
   680→                  "flex w-full items-start gap-3 rounded-xl border p-3 text-left transition-colors duration-100",
   681→                  active
   682→                    ? "border-primary bg-primary/5 ring-1 ring-primary/40"
   683→                    : "border-border hover:bg-muted/30",
   684→                )}
   685→                aria-pressed={active}
   686→              >
   687→                <opt.icon className={cn("mt-0.5 h-5 w-5 shrink-0", opt.color)} />
   688→                <div className="min-w-0 flex-1">
   689→                  <div className="flex items-center gap-2">
   690→                    <span className="text-sm font-medium">{opt.name}</span>
   691→                    {opt.badge && (
   692→                      <span className="rounded-full bg-emerald-500/15 px-2 py-0.5 text-[10px] font-medium text-emerald-400">
   693→                        {opt.badge}
   694→                      </span>
   695→                    )}
   696→                    {active && <Check className="h-3.5 w-3.5 text-primary" />}
   697→                  </div>
   698→                  <div className="text-xs text-muted-foreground">{opt.description}</div>
   699→                </div>
   700→              </button>
   701→            );
   702→          })}
   703→        </div>
   704→      </section>
   705→
   706→      {/* Animated swap between passphrase (PGP/ZIP) and CSV warning — no layout shift */}
   707→      <AnimatePresence mode="wait" initial={false}>
   708→        {isCsv ? (
   709→          <motion.div
   710→            key="csv-warning"
   711→            initial={{ opacity: 0, height: 0 }}
   712→            animate={{ opacity: 1, height: "auto" }}
   713→            exit={{ opacity: 0, height: 0 }}
   714→            transition={{ duration: 0.2, ease: [0.16, 1, 0.3, 1] }}
   715→            className="overflow-hidden"
   716→          >
   717→            <div className="rounded-xl border border-amber-500/40 bg-amber-500/10 p-4">
   718→              <div className="flex items-start gap-2">
   719→                <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-400" />
   720→                <div className="text-xs text-amber-200/90">
   721→                  The CSV will contain your passwords in plain text. Only do this on a
   722→                  trusted device and delete the file after use. No passphrase protects a CSV.
   723→                </div>
   724→              </div>
   725→              <Button
   726→                className="mt-3 w-full bg-amber-500 text-white hover:bg-amber-600"
   727→                disabled={items.length === 0 || busy}
   728→                onClick={handleExport}
   729→              >
   730→                {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Download className="mr-2 h-4 w-4" />}
   731→                I understand, export CSV
   732→              </Button>
   733→            </div>
   734→          </motion.div>
   735→        ) : (
   736→          <motion.section
   737→            key="passphrase"
   738→            initial={{ opacity: 0, height: 0 }}
   739→            animate={{ opacity: 1, height: "auto" }}
   740→            exit={{ opacity: 0, height: 0 }}
   741→            transition={{ duration: 0.2, ease: [0.16, 1, 0.3, 1] }}
   742→            className="overflow-hidden"
   743→          >
   744→            <div className="space-y-3">
   745→              <div className="flex items-center gap-2 text-sm font-medium">
   746→                <Lock className="h-4 w-4 text-muted-foreground" />
   747→                Passphrase
   748→                <span className="text-[10px] font-normal text-red-400">required</span>
   749→              </div>
   750→              <div className="space-y-2">
   751→                <div className="space-y-1.5">
   752→                  <Label htmlFor="exp-pass" className="text-xs text-muted-foreground">Passphrase</Label>
   753→                  <Input
   754→                    id="exp-pass"
   755→                    type="password"
   756→                    value={passphrase}
   757→                    onChange={(e) => setPassphrase(e.target.value)}
   758→                    placeholder="At least 8 characters"
   759→                    className="font-secret"
   760→                  />
   761→                </div>
   762→                <div className="space-y-1.5">
   763→                  <Label htmlFor="exp-pass2" className="text-xs text-muted-foreground">Confirm passphrase</Label>
   764→                  <Input
   765→                    id="exp-pass2"
   766→                    type="password"
   767→                    value={confirmPassphrase}
   768→                    onChange={(e) => setConfirmPassphrase(e.target.value)}
   769→                    placeholder="Repeat passphrase"
   770→                    className="font-secret"
   771→                  />
   772→                  {confirmPassphrase.length > 0 && !passphrasesMatch && (
   773→                    <p className="text-[11px] text-red-400">Passphrases don&apos;t match</p>
   774→                  )}
   775→                </div>
   776→                <p className="text-[11px] text-muted-foreground">
   777→                  This passphrase is independent of your master password. You&apos;ll need it to restore.
   778→                </p>
   779→              </div>
   780→              <Button
   781→                className="w-full"
   782→                disabled={busy || !canExport || items.length === 0}
   783→                onClick={handleExport}
   784→              >
   785→                {busy ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Download className="mr-2 h-4 w-4" />}
   786→                Export {format === "pgp" ? "encrypted file" : "archive"}
   787→              </Button>
   788→            </div>
   789→          </motion.section>
   790→        )}
   791→      </AnimatePresence>
   792→    </div>
   793→  );
   794→}
