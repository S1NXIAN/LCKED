    24→ * favourite toggle, folder, and card-brand detection.

    68→function blankItem(type: ItemType): NewItemInput {
    69→  const base = {
    70→    name: "",
    71→    favorite: false,
    72→    folder: "",
    73→    customFields: [],
    74→    createdAt: Date.now(),
    75→    updatedAt: Date.now(),
    76→    vaultId: null as string | null,
    77→    trashed: false,
    78→    trashedAt: null as number | null,
    79→  };
    80→  if (type === "login") {
    81→    return { ...base, type, details: { username: "", password: "", urls: [""], totp: "", notes: "" } };
    82→  }
    83→  if (type === "note") {
    84→    return { ...base, type, details: { content: "" } };
    85→  }
    86→  if (type === "card") {
    87→    return {
    88→      ...base,
    89→      type,
    90→      details: { cardholder: "", number: "", brand: "", cvv: "", expiry: "", pin: "", notes: "" },
    91→    };
    92→  }
    93→  return {
    94→    ...base,
    95→    type,
    96→    details: {
    97→      firstName: "", lastName: "", email: "", phone: "", company: "",
    98→      address1: "", address2: "", city: "", state: "", zip: "", country: "", notes: "",
    99→    },
   100→  };
   101→}
   102→
   103→function itemToInput(item: VaultItem): NewItemInput {
   104→  const { id: _id, createdAt: _c, updatedAt: _u, ...rest } = item;
   105→  return rest as NewItemInput;
   106→}

   110→/** Bordered card grouping related editable rows, divided by 1px lines. */
   111→function FieldCluster({ children }: { children: React.ReactNode }) {
   112→  return (
   113→    <div className="overflow-hidden rounded-xl border border-border bg-secondary/10 dark:bg-secondary/10">
   114→      {children}
   115→    </div>
   116→  );
   117→}

   119→/** A labelled field row inside a FieldCluster. Flat borderless input. */
   120→function FieldRowInput({
   121→  label,
   122→  first,
   123→  children,
   124→}: {
   125→  label: string;
   126→  first?: boolean;
   127→  children: React.ReactNode;
   128→}) {
   129→  return (
   130→    <div className={cn("px-3.5 py-2", !first && "border-t border-border/50")}>
   131→      <div className="mb-1 text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   132→        {label}
   133→      </div>
   134→      {children}
   135→    </div>
   136→  );
   137→}

   151→export function ItemEditor() {
   152→  const open = useVault((s) => s.editorOpen);
   153→  const editorItemId = useVault((s) => s.editorItemId);
   154→  const items = useVault((s) => s.items);
   155→  const vaults = useVault((s) => s.vaults);
   156→  const settings = useVault((s) => s.settings);
   157→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   158→  const saveItem = useVault((s) => s.saveItem);
   159→
   160→  const [form, setForm] = React.useState<NewItemInput | null>(null);
   161→  const [busy, setBusy] = React.useState(false);
   162→
   163→  // Initialise form when the sheet opens.
   164→  React.useEffect(() => {
   165→    if (!open) {
   166→      setForm(null);
   167→      return;
   168→    }
   169→    if (editorItemId) {
   170→      const existing = items.find((i) => i.id === editorItemId);
   171→      if (existing) {
   172→        setForm(itemToInput(existing));
   173→        return;
   174→      }
   175→    }
   176→    // Honour a type pre-selected via ⌘K / icon rail, then clear the stash.
   177→    const initialType = consumeNewItemType() ?? "login";
   178→    setForm(blankItem(initialType));
   179→  }, [open, editorItemId, items]);
   180→
   181→  const isEditing = Boolean(editorItemId);
   182→
   183→  if (!open || !form) return null;
   184→
   185→  const update = (patch: Partial<NewItemInput>) => setForm((f) => (f ? { ...f, ...patch } : f));
   186→  const updateDetails = (patch: Record<string, unknown>) =>
   187→    setForm((f) => (f ? { ...f, details: { ...(f.details as object), ...patch } } : f));
   188→
   189→  const handleSave = async () => {
   190→    if (!form) return;
   191→    if (!form.name.trim()) {
   192→      toast.error("Please give the item a name");
   193→      return;
   194→    }
   195→    setBusy(true);
   196→    try {
   197→      await saveItem(form, editorItemId ?? undefined);
   198→      toast.success(isEditing ? "Item updated" : "Item created");
   199→      setEditorOpen(false);
   200→    } catch (err) {
   201→      console.error(err);
   202→      toast.error("Could not save item");
   203→    } finally {
   204→      setBusy(false);
   205→    }
   206→  };
   207→
   208→  const addCustomField = () =>
   209→    update({ customFields: [...form.customFields, { name: "", value: "", type: "text" }] });
   210→  const updateCustomField = (idx: number, patch: Partial<CustomField>) =>
   211→    update({
   212→      customFields: form.customFields.map((cf, i) => (i === idx ? { ...cf, ...patch } : cf)),
   213→    });
   214→  const removeCustomField = (idx: number) =>
   215→    update({ customFields: form.customFields.filter((_, i) => i !== idx) });
   216→
   217→  // Login URL list helpers.
   218→  const urls = form.type === "login" ? form.details.urls : [];
   219→  const setUrl = (idx: number, val: string) => {
   220→    const next = urls.map((u, i) => (i === idx ? val : u));
   221→    updateDetails({ urls: next });
   222→  };
   223→  const addUrl = () => updateDetails({ urls: [...urls, ""] });
   224→  const removeUrl = (idx: number) =>
   225→    updateDetails({ urls: urls.filter((_, i) => i !== idx) });
   226→
   227→  // Selected vault (for the header dropdown).
   228→  const selectedVault = form.vaultId ? vaults.find((v) => v.id === form.vaultId) ?? null : null;
   229→
   230→  return (
   231→    <Sheet open={open} onOpenChange={(o) => setEditorOpen(o)}>
   232→      <SheetContent
   233→        side="right"
   234→        className="w-full gap-0 overflow-hidden border-l border-border bg-background p-0 sm:max-w-[480px] [&>button:last-child]:hidden"
   235→      >
   236→        {/* Header — type icon + title (left); vault selector + Save (right) */}
   237→        <SheetHeader className="flex-row items-center justify-between gap-2 border-b border-border px-4 py-3">
   238→          <SheetTitle className="flex min-w-0 items-center gap-2.5 text-base font-semibold">
   239→            <ItemTypeIcon type={form.type} size="sm" />
   240→            <span className="truncate">{isEditing ? "Edit item" : "New item"}</span>
   241→          </SheetTitle>
   242→          <div className="flex shrink-0 items-center gap-1.5">
   243→            {/* Vault selector dropdown */}
   244→            <DropdownMenu>
   245→              <DropdownMenuTrigger asChild>
   246→                <button
   247→                  type="button"
   248→                  className="flex h-8 items-center gap-1.5 rounded-md border border-border bg-background px-2 text-xs text-foreground transition-colors hover:bg-muted/60"
   249→                  aria-label="Select vault"
   250→                  title={selectedVault ? selectedVault.name : "No vault"}
   251→                >
   252→                  {selectedVault ? (
   253→                    <VaultIcon icon={selectedVault.icon} color={selectedVault.color} size={18} />
   254→                  ) : (
   255→                    <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   256→                      —
   257→                    </span>
   258→                  )}
   259→                  <span className="max-w-[80px] truncate">{selectedVault ? selectedVault.name : "No vault"}</span>
   260→                  <ChevronsUpDown className="h-3 w-3 text-muted-foreground" />
   261→                </button>
   262→              </DropdownMenuTrigger>
   263→              <DropdownMenuContent align="end" className="min-w-[180px]">
   264→                <DropdownMenuItem
   265→                  onSelect={() => update({ vaultId: null })}
   266→                  className={cn("gap-2", !selectedVault && "bg-muted/60")}
   267→                >
   268→                  <span className="inline-flex h-[18px] w-[18px] items-center justify-center rounded-md bg-muted text-[10px] text-muted-foreground">
   269→                    —
   270→                  </span>
   271→                  No vault
   272→                </DropdownMenuItem>
   273→                {vaults.map((v) => (
   274→                  <DropdownMenuItem
   275→                    key={v.id}
   276→                    onSelect={() => update({ vaultId: v.id })}
   277→                    className={cn("gap-2", selectedVault?.id === v.id && "bg-muted/60")}
   278→                  >
   279→                    <VaultIcon icon={v.icon} color={v.color} size={18} />
   280→                    <span className="truncate">{v.name}</span>
   281→                    {selectedVault?.id === v.id && <Check className="ml-auto h-3.5 w-3.5" />}
   282→                  </DropdownMenuItem>
   283→                ))}
   284→              </DropdownMenuContent>
   285→            </DropdownMenu>
   286→
   287→            {/* Save / Create */}
   288→            <Button
   289→              type="button"
   290→              size="sm"
   291→              onClick={handleSave}
   292→              disabled={busy}
   293→              className="min-w-[72px]"
   294→            >
   295→              {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : isEditing ? "Save" : "Create"}
   296→            </Button>
   297→          </div>
   298→        </SheetHeader>
   299→        <SheetDescription className="sr-only">
   300→          {isEditing ? "Edit the selected vault item." : "Create a new vault item."}
   301→        </SheetDescription>
   302→
   303→        {/* Body — flat scrollable form */}
   304→        <div className="lcked-scroll flex-1 overflow-y-auto p-4">
   305→          {/* Type selector — only when creating */}
   306→          {!isEditing && (
   307→            <div className="mb-4 grid grid-cols-4 gap-2">
   308→              {(Object.keys(ITEM_TYPE_LABELS) as ItemType[]).map((t) => (
   309→                <button
   310→                  key={t}
   311→                  type="button"
   312→                  onClick={() => setForm(blankItem(t))}
   313→                  className={cn(
   314→                    "flex flex-col items-center gap-1.5 rounded-lg border p-3 text-xs transition-colors",
   315→                    form.type === t
   316→                      ? "border-primary bg-primary/10 text-foreground"
   317→                      : "border-border text-muted-foreground hover:bg-muted/50",
   318→                  )}
   319→                >
   320→                  <ItemTypeIcon type={t} size="sm" />
   321→                  {ITEM_TYPE_LABELS[t]}
   322→                </button>
   323→              ))}
   324→            </div>
   325→          )}
   326→
   327→          {/* Name + favorite star — inline to save space */}
   328→          <div className="mb-4">
   329→            <div className="mb-1 flex items-center justify-between">
   330→              <div className="text-[10px] font-medium uppercase tracking-wider text-muted-foreground/70">
   331→                Name
   332→              </div>
   333→              <button
   334→                type="button"
   335→                onClick={() => update({ favorite: !form.favorite })}
   336→                className="flex h-6 w-6 items-center justify-center rounded-md transition-colors hover:bg-muted"
   337→                aria-label={form.favorite ? "Remove favorite" : "Add favorite"}
   338→                title={form.favorite ? "Remove favorite" : "Add favorite"}
   339→              >
   340→                <Star className={cn("h-4 w-4 transition-colors", form.favorite ? "fill-amber-400 text-amber-400" : "text-muted-foreground")} />
   341→              </button>
   342→            </div>
   343→            <input
   344→              type="text"
   345→              value={form.name}
   346→              onChange={(e) => update({ name: e.target.value })}
   347→              placeholder="e.g. GitHub, Bank card, WiFi password"
   348→              autoFocus
   349→              className="w-full border-0 bg-transparent dark:bg-transparent px-0 py-0.5 text-lg font-medium text-foreground placeholder:text-muted-foreground/60 focus:outline-none"
   350→              aria-label="Item name"
   351→            />
   352→          </div>
   353→
   354→          {/* Type-specific */}
   355→          {form.type === "login" && (
   356→            <div className="space-y-4">
   357→              {/* Username + Password */}
   358→              <FieldCluster>
   359→                <FieldRowInput label={isEmail(form.details.username) ? "Email" : "Username"} first>
   360→                  <Input
   361→                    value={form.details.username}
   362→                    onChange={(e) => updateDetails({ username: e.target.value })}
   363→                    placeholder="you@example.com"
   364→                    autoComplete="off"
   365→                    className={flatInputCls}
   366→                  />
   367→                </FieldRowInput>
   368→                <FieldRowInput label="Password">
   369→                  <PasswordField
   370→                    value={form.details.password}
   371→                    onChange={(v) => updateDetails({ password: v })}
   372→                    placeholder="Enter password"
   373→                    showStrength
   374→                    showGenerate
   375→                    generatorOptions={settings.generator}
   376→                    inputClassName={flatPasswordInputCls}
   377→                  />
   378→                </FieldRowInput>
   379→              </FieldCluster>
   380→
   381→              {/* TOTP */}
   382→              <FieldCluster>
   383→                <FieldRowInput label="Verification code (TOTP)" first>
   384→                  <Input
   385→                    value={form.details.totp}
   386→                    onChange={(e) => updateDetails({ totp: e.target.value })}
   387→                    placeholder="Base32 secret or otpauth:// URI"
   388→                    className={cn(flatInputCls, "font-secret")}
   389→                    autoComplete="off"
   390→                  />
   391→                </FieldRowInput>
   392→              </FieldCluster>
   393→
   394→              {/* URLs */}
   395→              <FieldCluster>
   396→                <FieldRowInput label="URLs" first>
   397→                  <div className="space-y-2">
   398→                    {urls.map((url, idx) => (
   399→                      <div key={idx} className="flex items-center gap-2">
   400→                        <Link2 className="h-4 w-4 shrink-0 text-muted-foreground" />
   401→                        <Input
   402→                          value={url}
   403→                          onChange={(e) => setUrl(idx, e.target.value)}
   404→                          placeholder="https://example.com"
   405→                          className={cn(flatInputCls, "font-secret")}
   406→                          autoComplete="off"
   407→                        />
   408→                        {urls.length > 1 && (
   409→                          <Button
   410→                            type="button"
   411→                            size="icon"
   412→                            variant="ghost"
   413→                            className="h-8 w-8 shrink-0 text-muted-foreground hover:text-red-400"
   414→                            onClick={() => removeUrl(idx)}
   415→                          >
   416→                            <Trash2 className="h-3.5 w-3.5" />
   417→                          </Button>
   418→                        )}
   419→                      </div>
   420→                    ))}
   421→                    <Button type="button" variant="outline" size="sm" onClick={addUrl}>
   422→                      <Plus className="mr-1.5 h-3.5 w-3.5" />
   423→                      Add URL
   424→                    </Button>
   425→                  </div>
   426→                </FieldRowInput>
   427→              </FieldCluster>
   428→
   429→              {/* Notes */}
   430→              <FieldCluster>
   431→                <FieldRowInput label="Notes" first>
   432→                  <Textarea
   433→                    value={form.details.notes}
   434→                    onChange={(e) => updateDetails({ notes: e.target.value })}
   435→                    placeholder="Optional secure notes…"
   436→                    rows={3}
   437→                    className={flatTextareaCls}
   438→                  />
   439→                </FieldRowInput>
   440→              </FieldCluster>
   441→            </div>
   442→          )}
   443→
   444→          {form.type === "note" && (
   445→            <FieldCluster>
   446→              <FieldRowInput label="Content" first>
   447→                <Textarea
   448→                  value={form.details.content}
   449→                  onChange={(e) => updateDetails({ content: e.target.value })}
   450→                  placeholder="Write your secure note here…"
   451→                  rows={10}
   452→                  className={flatTextareaCls}
   453→                />
   454→              </FieldRowInput>
   455→            </FieldCluster>
   456→          )}
   457→
   458→          {form.type === "card" && (
   459→            <div className="space-y-4">
   460→              {/* Cardholder + Card number + CVV + Expiry + PIN */}
   461→              <FieldCluster>
   462→                <FieldRowInput label="Cardholder name" first>
   463→                  <Input
   464→                    value={form.details.cardholder}
   465→                    onChange={(e) => updateDetails({ cardholder: e.target.value })}
   466→                    placeholder="Name on card"
   467→                    autoComplete="off"
   468→                    className={flatInputCls}
   469→                  />
   470→                </FieldRowInput>
   471→                <FieldRowInput label="Card number">
   472→                  <PasswordField
   473→                    value={form.details.number}
   474→                    onChange={(v) => {
   475→                      const brand = detectCardBrand(v);
   476→                      updateDetails({ number: v, brand: brand || form.details.brand });
   477→                    }}
   478→                    placeholder="0000 0000 0000 0000"
   479→                    showGenerate={false}
   480→                    inputClassName={flatPasswordInputCls}
   481→                  />
   482→                  {form.details.brand && (
   483→                    <p className="mt-1 text-xs text-muted-foreground">Detected: {form.details.brand}</p>
   484→                  )}
   485→                </FieldRowInput>
   486→                <div className="grid grid-cols-2 divide-x divide-border/50">
   487→                  <FieldRowInput label="CVV" first>
   488→                    <PasswordField
   489→                      value={form.details.cvv}
   490→                      onChange={(v) => updateDetails({ cvv: v })}
   491→                      placeholder="123"
   492→                      showGenerate={false}
   493→                      inputClassName={flatPasswordInputCls}
   494→                    />
   495→                  </FieldRowInput>
   496→                  <FieldRowInput label="Expiry (MM/YY)" first>
   497→                    <Input
   498→                      value={form.details.expiry}
   499→                      onChange={(e) => updateDetails({ expiry: e.target.value })}
   500→                      placeholder="08/27"
   501→                      className={cn(flatInputCls, "font-secret")}
   502→                      autoComplete="off"
   503→                    />
   504→                  </FieldRowInput>
   505→                </div>
   506→                <FieldRowInput label="PIN">
   507→                  <PasswordField
   508→                    value={form.details.pin}
   509→                    onChange={(v) => updateDetails({ pin: v })}
   510→                    placeholder="••••"
   511→                    showGenerate={false}
   512→                    inputClassName={flatPasswordInputCls}
   513→                  />
   514→                </FieldRowInput>
   515→              </FieldCluster>
   516→
   517→              {/* Notes */}
   518→              <FieldCluster>
   519→                <FieldRowInput label="Notes" first>
   520→                  <Textarea
   521→                    value={form.details.notes}
   522→                    onChange={(e) => updateDetails({ notes: e.target.value })}
   523→                    placeholder="Optional notes…"
   524→                    rows={3}

   538→                    <Input

   545→                  <FieldRowInput label="Last name" first>
