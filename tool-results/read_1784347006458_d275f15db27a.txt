    74→function subtitle(item: VaultItem): string {
    75→  switch (item.type) {
    76→    case "login":
    77→      return item.details.username || item.details.urls[0] || "—";
    78→    case "note":
    79→      return item.details.content.slice(0, 60) || "Empty note";
    80→    case "card":
    81→      return (
    82→        item.details.brand ||
    83→        item.details.number.slice(-4).padStart(item.details.number.length, "•") ||
    84→        "—"
    85→      );
    86→    case "identity":
    87→      return [item.details.firstName, item.details.lastName].filter(Boolean).join(" ") || item.details.email || "—";
    88→  }

    91→/* ----------------------- ActiveRowHighlight (rAF spring) --------------- */
    92→
    93→/**
    94→ * Single persistent highlight element rendered ONCE inside the <ul>.
    95→ *
    96→ * Slides toward the active item's bounding box via a requestAnimationFrame
    97→ * spring (exponential lerp, factor 0.22). Uses the `animateRef` pattern —
    98→ * the latest step function is stashed in a ref so external listeners
    99→ * (scroll, MutationObserver) can always kick the loop without re-binding.
   100→ *
   101→ * On `activeId` change: measure + start the animation. If the item isn't
   102→ * found in the DOM, `setVisible(false)` and bail. A MutationObserver re-
   103→ * checks when the list DOM changes; a scroll listener re-measures on
   104→ * scroll so the highlight stays glued to the active row.
   105→ */
   106→function ActiveRowHighlight({
   107→  listRef,
   108→  activeId,
   109→}: {
   110→  listRef: React.RefObject<HTMLUListElement | null>;
   111→  activeId: string | null;
   112→}) {
   113→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   114→  const rafRef = React.useRef<number | null>(null);
   115→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   116→  const wasVisibleRef = React.useRef(false);
   117→  const [visible, setVisible] = React.useState(false);
   118→  // animateRef holds the latest step function so external listeners can
   119→  // kick it without re-binding their own dependencies.
   120→  const animateRef = React.useRef<(() => void) | null>(null);
   121→
   122→  // Measure the active item's rect relative to the <ul>. If not found,
   123→  // hide the highlight.
   124→  const measure = React.useCallback(() => {
   125→    const list = listRef.current;
   126→    if (!activeId || !list) {
   127→      targetRef.current = null;
   128→      setVisible(false);
   129→      return;
   130→    }
   131→    const el = list.querySelector<HTMLElement>(
   132→      `[data-item-id="${CSS.escape(activeId)}"]`,
   133→    );
   134→    if (!el) {
   135→      targetRef.current = null;
   136→      setVisible(false);
   137→      return;
   138→    }
   139→    const r = el.getBoundingClientRect();
   140→    const lr = list.getBoundingClientRect();
   141→    const next = {
   142→      x: r.left - lr.left,
   143→      y: r.top - lr.top,
   144→      w: r.width,
   145→      h: r.height,
   146→    };
   147→    // First activation (or returning from invisible): snap to target so
   148→    // we don't slide in from (0, 0).
   149→    if (!wasVisibleRef.current) {
   150→      const hl = highlightRef.current;
   151→      if (hl) {
   152→        hl.style.transform = `translate(${next.x}px, ${next.y}px)`;
   153→        hl.style.width = `${next.w}px`;
   154→        hl.style.height = `${next.h}px`;
   155→      }
   156→    }
   157→    targetRef.current = next;
   158→    setVisible(true);
   159→  }, [activeId, listRef]);
   160→
   161→  // (Re)define the animation step on every render so it captures the
   162→  // latest refs. Store it in animateRef for external listeners.
   163→  React.useEffect(() => {
   164→    const step = () => {
   165→      const hl = highlightRef.current;
   166→      const target = targetRef.current;
   167→      const list = listRef.current;
   168→      if (!hl || !target || !list) {
   169→        rafRef.current = null;
   170→        return;
   171→      }
   172→      const cur = hl.getBoundingClientRect();
   173→      const lr = list.getBoundingClientRect();
   174→      const cx = cur.left - lr.left;
   175→      const cy = cur.top - lr.top;
   176→      const FACTOR = 0.22;
   177→      const nx = cx + (target.x - cx) * FACTOR;
   178→      const ny = cy + (target.y - cy) * FACTOR;
   179→      const nw = cur.width + (target.w - cur.width) * FACTOR;
   180→      const nh = cur.height + (target.h - cur.height) * FACTOR;
   181→      const settled =
   182→        Math.abs(target.x - nx) < 0.5 &&
   183→        Math.abs(target.y - ny) < 0.5 &&
   184→        Math.abs(target.w - nw) < 0.5 &&
   185→        Math.abs(target.h - nh) < 0.5;
   186→      const fx = settled ? target.x : nx;
   187→      const fy = settled ? target.y : ny;
   188→      const fw = settled ? target.w : nw;
   189→      const fh = settled ? target.h : nh;
   190→      hl.style.transform = `translate(${fx}px, ${fy}px)`;
   191→      hl.style.width = `${fw}px`;
   192→      hl.style.height = `${fh}px`;
   193→      if (settled) {
   194→        rafRef.current = null;
   195→        return;
   196→      }
   197→      rafRef.current = requestAnimationFrame(step);
   198→    };
   199→    animateRef.current = step;
   200→    return () => {
   201→      animateRef.current = null;
   202→    };
   203→  }, [listRef]);
   204→
   205→  // Track visibility transitions for the snap-on-first-show behavior.
   206→  React.useEffect(() => {
   207→    wasVisibleRef.current = visible;
   208→  }, [visible]);
   209→
   210→  // Re-measure + kick the animation if it isn't already running.
   211→  const kick = React.useCallback(() => {
   212→    measure();
   213→    if (!rafRef.current && animateRef.current) {
   214→      rafRef.current = requestAnimationFrame(animateRef.current);
   215→    }
   216→  }, [measure]);
   217→
   218→  // On activeId change: measure + start the animation. Cancel any in-
   219→  // flight loop on cleanup.
   220→  React.useEffect(() => {
   221→    kick();
   222→    return () => {
   223→      if (rafRef.current) {
   224→        cancelAnimationFrame(rafRef.current);
   225→        rafRef.current = null;
   226→      }
   227→    };
   228→  }, [activeId, kick]);
   229→
   230→  // MutationObserver — re-check when the list DOM changes (items added,
   231→  // removed, reordered by sort, attributes toggled).
   232→  React.useEffect(() => {
   233→    const list = listRef.current;
   234→    if (!list) return;
   235→    const observer = new MutationObserver(() => kick());
   236→    observer.observe(list, {
   237→      childList: true,
   238→      subtree: true,
   239→      attributes: true,
   240→      attributeFilter: ["data-item-id"],
   241→    });
   242→    return () => observer.disconnect();
   243→  }, [listRef, kick]);
   244→
   245→  // Scroll listener — re-measure + re-animate on scroll so the highlight
   246→  // stays glued to the active row (both inside the scroll viewport and
   247→  // when the page itself scrolls).
   248→  React.useEffect(() => {
   249→    const list = listRef.current;
   250→    if (!list) return;
   251→    const container =
   252→      list.closest('[data-radix-scroll-area-viewport]') ??
   253→      list.parentElement ??
   254→      null;
   255→    const handler = () => kick();
   256→    container?.addEventListener("scroll", handler, { passive: true });
   257→    window.addEventListener("scroll", handler, { passive: true });
   258→    return () => {
   259→      container?.removeEventListener("scroll", handler);
   260→      window.removeEventListener("scroll", handler);
   261→    };
   262→  }, [listRef, kick]);
   263→
   264→  if (!visible) return null;
   265→  return (
   266→    <div
   267→      ref={highlightRef}
   268→      aria-hidden
   269→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   270→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   271→    />
   272→  );
   273→}

   275→/* ------------------------------- ItemList ------------------------------ */
   276→
   277→export function ItemList({
   278→  filter,
   279→  setFilter,
   280→  activeVault: activeVaultProp,
   281→  onMobileBack,
   282→}: {
   283→  /** Type filter (All / Login / Note / Card / Identity). */
   284→  filter: FilterType;
   285→  setFilter: (f: FilterType) => void;
   286→  /** Active vault filter from the store — passed explicitly so the list re-
   287→   *  renders when the sidebar switches between All/Favorites/Trash/vault. */
   288→  activeVault?: string;
   289→  /** Mobile-only callback when the user taps Back from a Trash detail. */
   290→  onMobileBack?: () => void;
   291→}) {
   292→  // Store hooks — we read activeVault directly here as well so the list
   293→  // always re-renders even if a parent forgets to thread the prop through.
   294→  const items = useVault((s) => s.items);
   295→  const vaults = useVault((s) => s.vaults);
   296→  const selectedId = useVault((s) => s.selectedId);
   297→  const setSelected = useVault((s) => s.setSelected);
   298→  const searchQuery = useVault((s) => s.searchQuery);
   299→  const setEditorOpen = useVault((s) => s.setEditorOpen);
   300→  const trashItem = useVault((s) => s.trashItem);
   301→  const restoreItem = useVault((s) => s.restoreItem);
   302→  const permanentlyDeleteItem = useVault((s) => s.permanentlyDeleteItem);
   303→  const moveItemToVault = useVault((s) => s.moveItemToVault);
   304→  const toggleFavorite = useVault((s) => s.toggleFavorite);
   305→  const sortFavoritesFirst = useVault((s) => s.settings.sortFavoritesFirst);
   306→
   307→  const activeVault = activeVaultProp ?? useVault.getState().activeVault;
   308→  const isTrashView = activeVault === "trash";
   309→
   310→  // Sort persistence — read the initial value from localStorage so the
   311→  // user's last sort preference survives reloads.
   312→  const [sort, setSortState] = React.useState<SortKey>(() => {
   313→    if (typeof window === "undefined") return "newest";
   314→    const stored = window.localStorage.getItem("lcked-sort") as SortKey | null;
   315→    return stored === "newest" || stored === "oldest" || stored === "alphabetical"
   316→      ? stored
   317→      : "newest";
   318→  });
   319→  const setSort = React.useCallback((next: SortKey) => {
   320→    setSortState(next);
   321→    try {
   322→      window.localStorage.setItem("lcked-sort", next);
   323→    } catch {
   324→      // localStorage may be denied (private mode, etc.) — best-effort.
   325→    }
   326→  }, []);
   327→
   328→  const [multiSelect, setMultiSelect] = React.useState(false);
   329→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(new Set());
   330→  // Reset multi-select whenever we leave multi-select mode.
   331→  React.useEffect(() => {
   332→    if (!multiSelect) setSelectedIds(new Set());
   333→  }, [multiSelect]);
   334→
   335→  // Deferred search query — keeps typing responsive on large vaults by
   336→  // letting the filter/sort work happen at lower priority.
   337→  const deferredSearch = React.useDeferredValue(searchQuery);
   338→
   339→  // List ref for the ActiveRowHighlight.
   340→  const listRef = React.useRef<HTMLUListElement | null>(null);
   341→
   342→  /* --------------------------- filtering + sorting --------------------------- */
   343→  const filtered = React.useMemo(() => {
   344→    let list = items;
   345→    // Vault-level filter.
   346→    if (activeVault === "trash") list = list.filter((i) => i.trashed);
   347→    else if (activeVault === "favorites") list = list.filter((i) => !i.trashed && i.favorite);
   348→    else if (activeVault && activeVault !== "all")
   349→      list = list.filter((i) => !i.trashed && i.vaultId === activeVault);
   350→    else list = list.filter((i) => !i.trashed);
   351→    // Type filter (secondary, list-header dropdown).
   352→    if (filter !== "all" && typeof filter === "string") {
   353→      list = list.filter((i) => i.type === filter);
   354→    }
   355→    // Fuzzy search.
   356→    list = searchItems(list, deferredSearch);
   357→    // Sort.
   358→    const sorted = [...list];
   359→    sorted.sort((a, b) => {
   360→      // Pin ALWAYS pins to top (unconditional). The schema doesn't have a
   361→      // dedicated `pinned` flag yet — this branch is a placeholder so the
   362→      // behavior is correct the moment one is added.
   363→      const aPin = false;
   364→      const bPin = false;
   365→      // Favorite only when sortFavoritesFirst is on.
   366→      const aFav = sortFavoritesFirst && a.favorite;
   367→      const bFav = sortFavoritesFirst && b.favorite;
   368→      // Favorite takes priority over pin when both active.
   369→      if (aFav !== bFav) return aFav ? -1 : 1;
   370→      if (aPin !== bPin) return aPin ? -1 : 1;
   371→      // Primary sort.
   372→      if (sort === "newest") return b.updatedAt - a.updatedAt;
   373→      if (sort === "oldest") return a.updatedAt - b.updatedAt;
   374→      return a.name.localeCompare(b.name);
   375→    });
   376→    return sorted;
   377→  }, [items, activeVault, filter, deferredSearch, sort, sortFavoritesFirst]);
   378→
   379→  // Scroll-into-view when selection changes via keyboard.
   380→  React.useEffect(() => {
   381→    if (!selectedId) return;
   382→    const el = document.querySelector(`[data-item-id="${CSS.escape(selectedId)}"]`);
   383→    el?.scrollIntoView({ block: "nearest", behavior: "smooth" });
   384→  }, [selectedId]);
   385→
   386→  /* --------------------------- multi-select actions --------------------------- */
   387→  const toggleSelected = (id: string) => {
   388→    setSelectedIds((prev) => {
   389→      const next = new Set(prev);
   390→      if (next.has(id)) next.delete(id);
   391→      else next.add(id);
   392→      return next;
   393→    });
   394→  };
   395→  const selectAll = () => setSelectedIds(new Set(filtered.map((i) => i.id)));
   396→  const deselectAll = () => setSelectedIds(new Set());
   397→
   398→  const handleMultiTrash = async () => {
   399→    if (selectedIds.size === 0) return;
   400→    const ids = Array.from(selectedIds);
   401→    try {
   402→      await Promise.all(ids.map((id) => trashItem(id)));
   403→      toast.success(`Moved ${ids.length} to Trash`);
   404→      setMultiSelect(false);
   405→    } catch {
   406→      toast.error("Could not move items");
   407→    }
   408→  };
   409→  const handleMultiMove = async (vaultId: string | null) => {
   410→    if (selectedIds.size === 0) return;
   411→    const ids = Array.from(selectedIds);
   412→    try {
   413→      await Promise.all(ids.map((id) => moveItemToVault(id, vaultId)));
   414→      const label =
   415→        vaultId === null
   416→          ? "All Items"
   417→          : vaults.find((v) => v.id === vaultId)?.name ?? "vault";
   418→      toast.success(`Moved ${ids.length} to ${label}`);
   419→      setMultiSelect(false);
   420→    } catch {
   421→      toast.error("Could not move items");
   422→    }
   423→  };
   424→
   425→  /* ------------------------------- item helpers ------------------------------- */
   426→  // Helper used by the empty-area context menu (and any other caller that
   427→  // wants to spawn the editor pre-seeded with a type).
   428→  const createItem = React.useCallback((type: ItemType) => {
   429→    stashNewItemType(type);
   430→    setEditorOpen(true);
   431→  }, [setEditorOpen]);
   432→
   433→  const copyField = React.useCallback(
   434→    async (value: string | undefined, label: string) => {
   435→      if (!value) return;
   436→      try {
   437→        await copyWithAutoClear(value, label);
   438→        toast.success(`${label} copied`, { description: "Auto-clears in 30s" });
   439→      } catch {
   440→        toast.error("Clipboard access denied");
   441→      }
   442→    },
   443→    [],
   444→  );
   445→
   446→  /* ------------------------------- rendering ------------------------------- */
   447→  return (
   448→    <div className="flex h-full flex-col">
   449→      {/* Filter bar — type + sort + multi-select menu */}
   450→      <div className="border-b border-border bg-background px-3 py-2">
   451→        <div className="flex items-center gap-2">
   452→          <Select
   453→            value={typeof filter === "string" ? filter : "all"}
   454→            onValueChange={(v) => setFilter(v as FilterType)}
   455→          >
   456→            <SelectTrigger size="sm" className="h-8 w-[110px] shrink-0 border-border bg-muted/40 dark:bg-secondary/20">
   457→              <LayoutGrid className="h-3.5 w-3.5 text-muted-foreground" />
   458→              <SelectValue placeholder="Type" />
   459→            </SelectTrigger>
   460→            <SelectContent>
   461→              {TYPE_OPTIONS.map((opt) => (
   462→                <SelectItem key={opt.value} value={opt.value}>
   463→                  {opt.label}
   464→                </SelectItem>
   465→              ))}
   466→            </SelectContent>
   467→          </Select>
   468→          {/* Sort dropdown — outline + muted/40 + border-border to match
   469→              the type Select (fixes the visual mismatch from the old
   470→              `variant="secondary"` button). */}
   471→          <DropdownMenu>
   472→            <DropdownMenuTrigger asChild>
   473→              <Button
   474→                variant="outline"
   475→                size="sm"
   476→                className="h-8 shrink-0 gap-1.5 border-border bg-muted/40 px-2.5 hover:bg-muted/60 dark:bg-secondary/20"
   477→              >
   478→                <ArrowUpDown className="h-3.5 w-3.5 text-muted-foreground" />
   479→                <span className="text-xs">{SORT_OPTIONS.find((o) => o.value === sort)?.label ?? "Sort"}</span>
   480→                <ChevronDown className="h-3 w-3 opacity-50" />
   481→              </Button>
   482→            </DropdownMenuTrigger>
   483→            <DropdownMenuContent align="start" className="w-40">
   484→              {SORT_OPTIONS.map((opt) => (
   485→                <DropdownMenuItem key={opt.value} onClick={() => setSort(opt.value)} className="text-xs">
   486→                  {opt.label}
   487→                </DropdownMenuItem>
   488→              ))}
   489→            </DropdownMenuContent>
   490→          </DropdownMenu>
   491→
   492→          {/* 3-dots multi-select + select-all menu */}
   493→          <DropdownMenu>
   494→            <DropdownMenuTrigger asChild>
   495→              <Button
   496→                variant="ghost"
   497→                size="icon"
   498→                className="ml-auto h-8 w-8 shrink-0 text-muted-foreground hover:text-foreground"
   499→                aria-label="List actions"
   500→                aria-pressed={multiSelect}
   501→              >
   502→                <MoreVertical className="h-4 w-4" />
   503→              </Button>
   504→            </DropdownMenuTrigger>
   505→            <DropdownMenuContent align="end" className="w-44">
   506→              <DropdownMenuItem onClick={() => setMultiSelect((m) => !m)}>
   507→                {multiSelect ? "Exit multi-select" : "Multi-select"}
   508→              </DropdownMenuItem>
   509→              <DropdownMenuSeparator />
   510→              <DropdownMenuItem onClick={selectAll} disabled={filtered.length === 0}>
   511→                Select all
   512→              </DropdownMenuItem>
   513→              <DropdownMenuItem onClick={deselectAll} disabled={selectedIds.size === 0}>
   514→                Deselect all
   515→              </DropdownMenuItem>
   516→            </DropdownMenuContent>
   517→          </DropdownMenu>
   518→        </div>
   519→
   520→        {/* Active vault label */}
   521→        <div className="mt-2 flex items-center gap-1.5 text-xs text-muted-foreground">
   522→          {isTrashView && (
   523→            <span className="inline-flex items-center gap-1 rounded-md bg-amber-500/15 px-1.5 py-0.5 text-amber-400">
   524→              <Trash2 className="h-3 w-3" />
   525→              Trash · {filtered.length} items
   526→            </span>
   527→          )}
   528→          {!isTrashView && activeVault === "favorites" && (
   529→            <span className="inline-flex items-center gap-1">
   530→              <Star className="h-3 w-3 fill-amber-400 text-amber-400" />
   531→              Favorites · {filtered.length}
   532→            </span>
   533→          )}
   534→          {!isTrashView && activeVault !== "favorites" && activeVault !== "all" && (
   535→            <span className="inline-flex items-center gap-1">
   536→              {vaults.find((v) => v.id === activeVault)?.name ?? "Vault"} · {filtered.length}
   537→            </span>
   538→          )}
   539→          {!isTrashView && activeVault === "all" && (
   540→            <span>All Items · {filtered.length}</span>
   541→          )}
   542→        </div>
   543→      </div>
   544→
   545→      {/* Multi-select action bar */}
   546→      <AnimatePresence>
   547→        {multiSelect && (
   548→          <motion.div
   549→            initial={{ height: 0, opacity: 0 }}
   550→            animate={{ height: "auto", opacity: 1 }}
   551→            exit={{ height: 0, opacity: 0 }}
   552→            className="overflow-hidden border-b border-border bg-muted/40"
   553→          >
   554→            <div className="flex items-center gap-2 px-3 py-2">
   555→              <Button size="sm" variant="ghost" className="h-7 gap-1.5 px-2" onClick={() => setMultiSelect(false)}>
   556→                <X className="h-3.5 w-3.5" />
   557→                Cancel
   558→              </Button>
   559→              {/* "N Item(s) selected" — fixed minWidth prevents layout
   560→                  shift as the count changes between 0/1/many. */}
   561→              <span
   562→                className="text-xs text-muted-foreground"
   563→                style={{ minWidth: 130, display: "inline-block" }}
   564→              >
   565→                {selectedIds.size} Item{selectedIds.size === 1 ? "" : "s"} selected
   566→              </span>
   567→              <div className="ml-auto flex items-center gap-1.5">
   568→                {/* Move to vault dropdown */}
   569→                <DropdownMenu>
   570→                  <DropdownMenuTrigger asChild>
   571→                    <Button size="sm" variant="outline" className="h-7 gap-1.5 px-2" disabled={selectedIds.size === 0}>
   572→                      <FolderInput className="h-3.5 w-3.5" />
   573→                      Move
   574→                      <ChevronDown className="h-3 w-3 opacity-90" />
   575→                    </Button>
   576→                  </DropdownMenuTrigger>
   577→                  <DropdownMenuContent align="end" className="w-48">
   578→                    <DropdownMenuItem onSelect={() => handleMultiMove(null)}>
   579→                      All Items
   580→                    </DropdownMenuItem>
   581→                    {vaults.map((v) => (
   582→                      <DropdownMenuItem key={v.id} onSelect={() => handleMultiMove(v.id)}>
   583→                        {v.name}
   584→                      </DropdownMenuItem>
   585→                    ))}
   586→                  </DropdownMenuContent>
   587→                </DropdownMenu>
   588→                <Button
   589→                  size="sm"
   590→                  variant="outline"
   591→                  className="h-7 gap-1.5 border-destructive/40 px-2 text-destructive hover:bg-destructive/10 hover:text-destructive"
   592→                  disabled={selectedIds.size === 0}
   593→                  onClick={handleMultiTrash}
   594→                >
   595→                  <Trash2 className="h-3.5 w-3.5" />
   596→                  Trash
   597→                </Button>
   598→              </div>
   599→            </div>
   600→          </motion.div>
   601→        )}
   602→      </AnimatePresence>
   603→
   604→      {/* List — wrapped in a ContextMenu so right-clicking empty space
   605→          offers quick "New Login/Note/Card/Identity" actions. */}
   606→      <ContextMenu>
   607→        <ContextMenuTrigger asChild>
   608→          <ScrollArea className="lcked-scroll flex-1">
   609→            <div className="p-2">
   610→              {filtered.length === 0 ? (
   611→                <EmptyList
   612→                  hasItems={items.length > 0}
   613→                  isTrash={isTrashView}
   614→                  onCreate={() => createItem("login")}
   615→                />
   616→              ) : (
   617→                <LayoutGroup>
   618→                  <ul
   619→                    ref={listRef}
   620→                    className="relative space-y-0.5"
   621→                    role="listbox"
   622→                    aria-label="Vault items"
   623→                  >
   624→                    {/* Single persistent highlight — slides between active
   625→                        items via rAF spring (see ActiveRowHighlight). */}
   626→                    <ActiveRowHighlight
   627→                      listRef={listRef}
   628→                      activeId={multiSelect ? null : selectedId}
   629→                    />
   630→                    <AnimatePresence initial={false}>
   631→                      {filtered.map((item) => {
   632→                        const active = item.id === selectedId && !multiSelect;
   633→                        const checked = selectedIds.has(item.id);
   634→                        return (
   635→                          <motion.li
   636→                            key={item.id}
   637→                            layout
   638→                            initial={{ opacity: 0 }}
   639→                            animate={{ opacity: 1 }}
   640→                            exit={{ opacity: 0 }}
   641→                            transition={{ duration: 0.12 }}
   642→                            data-item-id={item.id}
   643→                            className="relative"
   644→                          >
   645→                            <ItemRow
   646→                              item={item}
   647→                              active={active}
   648→                              checked={checked}
   649→                              multiSelect={multiSelect}
   650→                              isTrashView={isTrashView}
   651→                              onPick={() =>
   652→                                multiSelect ? toggleSelected(item.id) : setSelected(item.id)
   653→                              }
   654→                              onToggleSelected={() => toggleSelected(item.id)}
   655→                              onRestore={() =>
   656→                                restoreItem(item.id).then(() =>
   657→                                  toast.success("Item restored"),
   658→                                )
   659→                              }
   660→                              onPermanentDelete={() =>
   661→                                permanentlyDeleteItem(item.id).then(() =>
   662→                                  toast.success("Permanently deleted"),
   663→                                )
   664→                              }
   665→                              onCopyField={copyField}
   666→                              onToggleFavorite={() => toggleFavorite(item.id)}
   667→                              onEdit={() => setEditorOpen(true, item.id)}
   668→                              onTrash={() =>
   669→                                trashItem(item.id).then(() =>
   670→                                  toast.success("Moved to Trash"),
   671→                                )
   672→                              }
   673→                            />
   674→                          </motion.li>
   675→                        );
   676→                      })}
   677→                    </AnimatePresence>
   678→                  </ul>
   679→                </LayoutGroup>
   680→              )}
   681→            </div>
   682→          </ScrollArea>
   683→        </ContextMenuTrigger>
   684→        <ContextMenuContent>
   685→          <ContextMenuItem onSelect={() => createItem("login")}>
   686→            <KeyRound className="h-4 w-4 text-violet-400" />
   687→            New Login
   688→          </ContextMenuItem>
   689→          <ContextMenuItem onSelect={() => createItem("note")}>
   690→            <StickyNote className="h-4 w-4 text-amber-400" />
   691→            New Note
   692→          </ContextMenuItem>
   693→          <ContextMenuItem onSelect={() => createItem("card")}>
   694→            <CreditCard className="h-4 w-4 text-emerald-400" />
   695→            New Card
   696→          </ContextMenuItem>
   697→          <ContextMenuItem onSelect={() => createItem("identity")}>
   698→            <UserRound className="h-4 w-4 text-sky-400" />
   699→            New Identity
   700→          </ContextMenuItem>
   701→        </ContextMenuContent>
   702→      </ContextMenu>
   703→
   704→      {/* Mobile back hint (only in trash view, for screen-reader friendliness) */}
   705→      {isTrashView && onMobileBack && (
   706→        <button
   707→          onClick={onMobileBack}
   708→          className="sr-only focus:not-sr-only"
   709→        >
   710→          Back to All Items
   711→        </button>
   712→      )}
   713→    </div>
   714→  );
   715→}

   719→/**
   720→ * The per-item row, extracted so it can carry its own ContextMenu without
   721→ * forcing the parent map to re-render every row when one is right-clicked.
   722→ */
   723→function ItemRow({
   724→  item,
   725→  active,
   726→  checked,
   727→  multiSelect,
   728→  isTrashView,
   729→  onPick,
   730→  onToggleSelected,
   731→  onRestore,
   732→  onPermanentDelete,
   733→  onCopyField,
   734→  onToggleFavorite,
   735→  onEdit,
   736→  onTrash,
   737→}: {
   738→  item: VaultItem;
   739→  active: boolean;
   740→  checked: boolean;
   741→  multiSelect: boolean;
   742→  isTrashView: boolean;
   743→  onPick: () => void;
   744→  onToggleSelected: () => void;
   745→  onRestore: () => void;
   746→  onPermanentDelete: () => void;
   747→  onCopyField: (value: string | undefined, label: string) => void;
   748→  onToggleFavorite: () => void;
   749→  onEdit: () => void;
   750→  onTrash: () => void;
   751→}) {
   752→  // Per-item context menu — wraps the button so left-click still picks the
   753→  // item, and right-click opens the context menu with copy/edit/etc.
   754→  return (
   755→    <ContextMenu>
   756→      <ContextMenuTrigger asChild>
   757→        <button
   758→          onClick={onPick}
   759→          role="option"
   760→          aria-selected={active || checked}
   761→          draggable={!multiSelect}
   762→          onDragStart={(e) => {
   763→            e.dataTransfer.setData("text/lcked-item", item.id);
   764→            e.dataTransfer.effectAllowed = "move";
   765→          }}
   766→          className={cn(
   767→            "group relative flex w-full items-center gap-3 rounded-lg px-2.5 py-2 text-left transition-colors",
   768→            active ? "text-accent-foreground" : "hover:bg-muted/60",
   769→            checked && "bg-accent/60",
   770→          )}
   771→        >
   772→          {/* Multi-select checkbox OR favicon/type-icon */}
   773→          {multiSelect ? (
   774→            <span className="relative z-10 flex h-7 w-7 shrink-0 items-center justify-center">
   775→              {checked ? (
   776→                <CheckSquare className="h-5 w-5 text-primary" />
   777→              ) : (
   778→                <Square className="h-5 w-5 text-muted-foreground" />
   779→              )}
   780→            </span>
   781→          ) : item.type === "login" && item.details.urls[0] ? (
   782→            <FaviconIcon
   783→              url={item.details.urls[0]}
   784→              size={28}
   785→              className="relative z-10"
   786→            />
   787→          ) : (
   788→            <ItemTypeIcon type={item.type} size="sm" className="relative z-10" />
   789→          )}
   790→
   791→          <div className="relative z-10 min-w-0 flex-1">
   792→            <div className="flex items-center gap-1.5">
   793→              <span className={cn("truncate text-sm font-medium", item.trashed && "line-through opacity-60")}>
   794→                {item.name}
   795→              </span>
   796→              {item.favorite && !item.trashed && (
   797→                <Star className="h-3 w-3 shrink-0 fill-amber-400 text-amber-400" />
   798→              )}
   799→            </div>
   800→            <div className="truncate text-xs text-muted-foreground">
   801→              {subtitle(item)}
   802→            </div>
   803→          </div>
   804→
   805→          {/* Type badge OR trash hover actions */}
   806→          {isTrashView ? (
   807→            <div className="relative z-10 flex shrink-0 items-center gap-0.5 opacity-0 transition-opacity group-hover:opacity-100 focus-within:opacity-100">
   808→              <Button
   809→                size="icon"
   810→                variant="ghost"
   811→                className="h-7 w-7 text-muted-foreground hover:text-foreground"
   812→                onClick={(e) => {
   813→                  e.stopPropagation();
   814→                  onRestore();
   815→                }}
   816→                aria-label="Restore"
   817→                title="Restore"
   818→              >
   819→                <RotateCcw className="h-3.5 w-3.5" />
   820→              </Button>
   821→              <Button
   822→                size="icon"
   823→                variant="ghost"
   824→                className="h-7 w-7 text-muted-foreground hover:text-red-400"
   825→                onClick={(e) => {
   826→                  e.stopPropagation();
   827→                  onPermanentDelete();
   828→                }}
   829→                aria-label="Delete permanently"
   830→                title="Delete permanently"
   831→              >
   832→                <Trash2 className="h-3.5 w-3.5" />
   833→              </Button>
   834→            </div>
   835→          ) : null}
   836→        </button>
   837→      </ContextMenuTrigger>
   838→      <ContextMenuContent>
   839→        {item.type === "login" && (
   840→          <>
   841→            <ContextMenuItem
   842→              onSelect={() => onCopyField(item.details.username, isEmail(item.details.username) ? "Email" : "Username")}
   843→              disabled={!item.details.username}
   844→            >
   845→              {isEmail(item.details.username) ? "Copy email" : "Copy username"}
   846→            </ContextMenuItem>
   847→            {item.details.password && (
   848→              <ContextMenuItem onSelect={() => onCopyField(item.details.password, "Password")}>
   849→                Copy password
   850→              </ContextMenuItem>
   851→            )}
   852→            {item.details.urls[0] && (
   853→              <ContextMenuItem onSelect={() => onCopyField(item.details.urls[0], "URL")}>
   854→                Copy URL
   855→              </ContextMenuItem>
   856→            )}
   857→            <ContextMenuSeparator />
   858→          </>
   859→        )}
   860→        {!isTrashView && (
   861→          <ContextMenuItem onSelect={onToggleFavorite}>
   862→            {item.favorite ? "Unfavorite" : "Favorite"}
   863→          </ContextMenuItem>
   864→        )}
   865→        <ContextMenuItem onSelect={onEdit}>Edit</ContextMenuItem>
   866→        {!isTrashView && (
   867→          <>
   868→            <ContextMenuSeparator />
   869→            <ContextMenuItem variant="destructive" onSelect={onTrash}>
   870→              Move to trash
   871→            </ContextMenuItem>
   872→          </>
   873→        )}
   874→      </ContextMenuContent>
   875→    </ContextMenu>
   876→  );
   877→}

   881→function EmptyList({
   882→  hasItems,
   883→  isTrash,
   884→  onCreate,
   885→}: {
   886→  hasItems: boolean;
   887→  isTrash: boolean;
   888→  onCreate: () => void;

   890→  const setActiveVault = useVault((s) => s.setActiveVault);
   891→  const setImportExportOpen = useVault((s) => s.setImportExportOpen);
   892→  return (
   893→    <div className="lcked-grid flex flex-col items-center justify-center px-6 py-20 text-center">
   894→      <DiamondMark size={56} glow className="mb-4" />
   895→      <h3 className="text-sm font-semibold">
   896→        {isTrash
   897→          ? "Trash is empty"
   898→          : hasItems
   899→            ? "No matches"
   900→            : "Your vault is empty"}
   901→      </h3>
   902→      <p className="mt-1.5 max-w-[16rem] text-xs leading-relaxed text-muted-foreground">
   903→        {isTrash
   904→          ? "Deleted items land here. They auto-purge after 30 days."
   905→          : hasItems
   906→            ? "Try a different search or filter."
   907→            : "Your secrets, encrypted on this device. Nothing leaves. Ever."}
   908→      </p>

   910→        <div className="mt-5 flex flex-col gap-2">
   911→          <Button size="sm" onClick={onCreate}>
   912→            Add your first item
   913→          </Button>

   915→            Import from Bitwarden / 1Password / Proton Pass
   916→          </Button>
   917→        </div>
   918→      )}
   919→      {isTrash && hasItems && (
   920→        <Button
   921→          size="sm"
   922→          variant="ghost"
   923→          className="mt-5"
   924→          onClick={() => setActiveVault("all")}
   925→        >
   926→          Back to All Items
   927→        </Button>
   928→      )}
   929→    </div>
   930→  );
   931→}
