     1→/**
     2→ * LCKED — Vault store (Zustand)
     3→ * ---------------------------------------------------------------------------
     4→ * Owns the auth state machine + every vault mutation. CryptoKey objects live
     5→ * only in memory here — they are NEVER persisted. All async actions are
     6→ * fault-tolerant: a failed IndexedDB write reverts the optimistic UI change.
     7→ */
     8→
     9→import { create } from "zustand";
    10→import { persist } from "zustand/middleware";
    11→import {
    12→  buildVerifier,
    13→  bytesToBase64,
    14→  checkVerifier,
    15→  decryptJson,
    16→  deriveMasterKey,
    17→  encryptJson,
    18→  generateVaultKey,
    19→  PBKDF2_ITERATIONS,
    20→  randomBytes,
    21→  randomId,
    22→  unwrapVaultKey,
    23→  wrapVaultKey,
    24→} from "@/lib/crypto";
    25→import {
    26→  deleteStoredItem,
    27→  loadAllStoredItems,
    28→  loadVaultMeta,
    29→  putStoredItem,
    30→  saveVaultMeta,
    31→  vaultExists,
    32→  wipeVault,
    33→} from "@/lib/vault-db";
    34→import {
    35→  DEFAULT_VAULT_SETTINGS,
    36→  type GeneratorOptions,
    37→  type NewItemInput,
    38→  type VaultDef,
    39→  type VaultItem,
    40→  type VaultMeta,
    41→  type VaultSettings,
    42→  type StoredItem,
    43→} from "@/lib/types";
    44→import {
    45→  exportToCsv,
    46→  importFromText,
    47→  type ImportResult,
    48→  type LckedExport,
    49→} from "@/lib/import-export";
    50→import { getSeedItems, getSeedVaults } from "@/lib/seed-data";
    51→
    52→export type VaultStatus = "loading" | "setup" | "locked" | "unlocked";
    53→
    54→interface VaultState {
    55→  status: VaultStatus;
    56→  items: VaultItem[];
    57→  /** User-defined vaults (colored containers, Proton Pass–style). */
    58→  vaults: VaultDef[];
    59→  /** Active vault filter: "all" | "trash" | vault id. */
    60→  activeVault: string;
    61→  /** Vault editor (create/rename/delete vault) dialog state. */
    62→  vaultEditorOpen: boolean;
    63→  editingVaultId: string | null;
    64→  createVaultDialogOpen: boolean;
    65→  settings: VaultSettings;
    66→  /** UI-only selection / filter state (not persisted). */
    67→  selectedId: string | null;
    68→  searchQuery: string;
    69→  /** CryptoKey handles — held in memory only while unlocked. */
    70→  masterKey: CryptoKey | null;
    71→  vaultKey: CryptoKey | null;
    72→  /** Whether the active editor is open (and which item, if editing). */
    73→  editorOpen: boolean;
    74→  editorItemId: string | null;
    75→  generatorOpen: boolean;
    76→  importExportOpen: boolean;
    77→  settingsOpen: boolean;
    78→  commandOpen: boolean;
    79→
    80→  // lifecycle
    81→  init: () => Promise<void>;
    82→  setupVault: (masterPassword: string) => Promise<void>;
    83→  unlock: (masterPassword: string) => Promise<boolean>;
    84→  lock: () => void;
    85→  resetVault: () => Promise<void>;
    86→
    87→  // item CRUD
    88→  saveItem: (input: NewItemInput, existingId?: string) => Promise<VaultItem>;
    89→  /** Soft-delete — moves the item to Trash with a 30-day TTL. */
    90→  trashItem: (id: string) => Promise<void>;
    91→  restoreItem: (id: string) => Promise<void>;
    92→  permanentlyDeleteItem: (id: string) => Promise<void>;
    93→  emptyTrash: () => Promise<void>;
    94→  moveItemToVault: (itemId: string, vaultId: string | null) => Promise<void>;
    95→  toggleFavorite: (id: string) => Promise<void>;
    96→  duplicateItem: (id: string) => Promise<void>;
    97→  importItems: (filename: string, text: string) => Promise<ImportResult>;
    98→
    99→  // vault (custom containers) CRUD
   100→  createVault: (name: string, color: string, icon: string) => Promise<VaultDef>;
   101→  deleteVault: (id: string) => Promise<void>;
   102→  renameVault: (id: string, name: string) => Promise<void>;
   103→  updateVault: (id: string, patch: Partial<Omit<VaultDef, "id" | "createdAt">>) => Promise<void>;
   104→  setActiveVault: (v: string) => void;
   105→  setVaultEditorOpen: (open: boolean, vaultId?: string | null) => void;
   106→  setCreateVaultDialogOpen: (open: boolean) => void;
   107→
   108→  // settings
   109→  updateSettings: (patch: Partial<VaultSettings>) => Promise<void>;
   110→  updateGenerator: (patch: Partial<GeneratorOptions>) => void;
   111→  changeMasterPassword: (current: string, next: string) => Promise<boolean>;
   112→
   113→  // export
   114→  exportEncrypted: (password: string) => Promise<string>;
   115→  exportCsv: () => string;
   116→
   117→  // UI helpers
   118→  setSelected: (id: string | null) => void;
   119→  setSearch: (q: string) => void;
   120→  setEditorOpen: (open: boolean, itemId?: string | null) => void;
   121→  setGeneratorOpen: (open: boolean) => void;
   122→  setImportExportOpen: (open: boolean) => void;
   123→  setSettingsOpen: (open: boolean) => void;
   124→  setCommandOpen: (open: boolean) => void;
   125→}
   126→
   127→export const useVault = create<VaultState>()(
   128→  persist(
   129→    (set, get) => ({
   130→      status: "loading",
   131→      items: [],
   132→      vaults: [],
   133→      activeVault: "all",
   134→      vaultEditorOpen: false,
   135→      editingVaultId: null,
   136→      createVaultDialogOpen: false,
   137→      settings: DEFAULT_VAULT_SETTINGS,
   138→      selectedId: null,
   139→      searchQuery: "",
   140→      masterKey: null,
   141→      vaultKey: null,
   142→      editorOpen: false,
   143→      editorItemId: null,
   144→      generatorOpen: false,
   145→      importExportOpen: false,
   146→      settingsOpen: false,
   147→      commandOpen: false,
   148→
   149→      /* ------------------------------ lifecycle ------------------------------ */
   150→
   151→      init: async () => {
   152→        try {
   153→          if (typeof window === "undefined") return;
   154→          const exists = await vaultExists();
   155→          set({ status: exists ? "locked" : "setup" });
   156→        } catch (err) {
   157→          console.error("init failed", err);
   158→          set({ status: "setup" });
   159→        }
   160→      },
   161→
   162→      setupVault: async (masterPassword) => {
   163→        const salt = bytesToBase64(randomBytes(16));
   164→        const masterKey = await deriveMasterKey(masterPassword, salt, PBKDF2_ITERATIONS);
   165→        const vaultKey = await generateVaultKey();
   166→        const { ciphertext, iv } = await wrapVaultKey(vaultKey, masterKey);
   167→        const verifier = await buildVerifier(masterKey);
   168→
   169→        // Seed demo vaults + items for easier debugging of edge cases.
   170→        const now = Date.now();
   171→        const seedVaults = getSeedVaults(now);
   172→        const seedInputs = getSeedItems(now);
   173→
   174→        // Encrypt + persist each seed item.
   175→        const storedItems: StoredItem[] = [];
   176→        for (const input of seedInputs) {
   177→          const item: VaultItem = {
   178→            ...input,
   179→            id: randomId(),
   180→            createdAt: now,
   181→            updatedAt: now,
   182→          } as VaultItem;
   183→          const enc = await encryptJson(item, vaultKey);
   184→          const stored: StoredItem = {
   185→            id: item.id,
   186→            type: item.type,
   187→            ciphertext: enc.ciphertext,
   188→            iv: enc.iv,
   189→            createdAt: item.createdAt,
   190→            updatedAt: item.updatedAt,
   191→          };
   192→          storedItems.push(stored);
   193→          await putStoredItem(stored);
   194→        }
   195→
   196→        const meta: VaultMeta = {
   197→          id: "singleton",
   198→          salt,
   199→          iterations: PBKDF2_ITERATIONS,
   200→          encryptedVaultKey: ciphertext,
   201→          vaultKeyIv: iv,
   202→          verifier: verifier.verifier,
   203→          verifierIv: verifier.verifierIv,
   204→          verifierToken: verifier.verifierToken,
   205→          createdAt: now,
   206→          settings: DEFAULT_VAULT_SETTINGS,
   207→          vaults: seedVaults,
   208→        };
   209→        await saveVaultMeta(meta);
   210→
   211→        // Hydrate the decrypted items into state.
   212→        const decryptedItems: VaultItem[] = [];
   213→        for (const stored of storedItems) {
   214→          try {
   215→            const item = await decryptJson<VaultItem>(stored.ciphertext, stored.iv, vaultKey);
   216→            decryptedItems.push(item);
   217→          } catch { /* skip undecryptable */ }
   218→        }
   219→
   220→        set({
   221→          status: "unlocked",
   222→          masterKey,
   223→          vaultKey,
   224→          settings: DEFAULT_VAULT_SETTINGS,
   225→          items: decryptedItems,
   226→          vaults: seedVaults,
   227→          activeVault: "all",
   228→          selectedId: null,
   229→        });
   230→      },
   231→
   232→      unlock: async (masterPassword) => {
   233→        const meta = await loadVaultMeta();
   234→        if (!meta) {
   235→          set({ status: "setup" });
   236→          return false;
   237→        }
   238→        const masterKey = await deriveMasterKey(
   239→          masterPassword,
   240→          meta.salt,
   241→          meta.iterations,
   242→        );
   243→        const ok = await checkVerifier(
   244→          masterKey,
   245→          meta.verifier,
   246→          meta.verifierIv,
   247→          meta.verifierToken,
   248→        );
   249→        if (!ok) return false;
   250→
   251→        const vaultKey = await unwrapVaultKey(
   252→          meta.encryptedVaultKey,
   253→          meta.vaultKeyIv,
   254→          masterKey,
   255→        );
   256→
   257→        // Decrypt every stored item.
   258→        const stored = await loadAllStoredItems();
   259→        const items: VaultItem[] = [];
   260→        const toReencrypt: VaultItem[] = [];
   261→        let needsMetaSave = false;
   262→        const TRASH_TTL_MS = 30 * 24 * 60 * 60 * 1000; // 30 days
   263→        const now = Date.now();
   264→        for (const s of stored) {
   265→          try {
   266→            const item = await decryptJson<VaultItem>(s.ciphertext, s.iv, vaultKey);
   267→            // Migration: ensure the new soft-delete + vaultId fields exist on
   268→            // items written by older LCKED versions. Mutated records are queued
   269→            // for a transparent re-encrypt + persist.
   270→            let migrated: VaultItem | null = null;
   271→            if (item.vaultId === undefined) {
   272→              item.vaultId = null;
   273→              migrated = item;
   274→            }
   275→            if (item.trashed === undefined) {
   276→              item.trashed = false;
   277→              migrated = item;
   278→            }
   279→            if (item.trashedAt === undefined) {
   280→              item.trashedAt = null;
   281→              migrated = item;
   282→            }
   283→            // 30-day auto-delete: drop trashed items whose TTL has expired.
   284→            if (item.trashed && item.trashedAt && now - item.trashedAt > TRASH_TTL_MS) {
   285→              try {
   286→                await deleteStoredItem(item.id);
   287→              } catch {
   288→                // best-effort; keep going
   289→              }
   290→              continue;
   291→            }
   292→            if (migrated) toReencrypt.push(migrated);
   293→            items.push(item);
   294→          } catch {
   295→            // skip corrupt record but keep the vault usable
   296→            console.warn("Failed to decrypt item", s.id);
   297→          }
   298→        }
   299→        // Persist any migrated records so we never re-migrate them again.
   300→        if (toReencrypt.length > 0) {
   301→          await Promise.all(
   302→            toReencrypt.map(async (it) => {
   303→              const { ciphertext, iv } = await encryptJson(it, vaultKey);
   304→              await putStoredItem({
   305→                id: it.id,
   306→                type: it.type,
   307→                ciphertext,
   308→                iv,
   309→                createdAt: it.createdAt,
   310→                updatedAt: it.updatedAt,
   311→              });
   312→            }),
   313→          );
   314→        }
   315→        items.sort((a, b) => b.updatedAt - a.updatedAt);
   316→
   317→        // Hydrate vaults (custom containers) — migrate older meta that lacks
   318→        // the field by persisting an empty array.
   319→        const vaults: VaultDef[] = Array.isArray(meta.vaults) ? meta.vaults : [];
   320→        if (!Array.isArray(meta.vaults)) {
   321→          needsMetaSave = true;
   322→        }
   323→        if (needsMetaSave) {
   324→          await saveVaultMeta({ ...meta, vaults });
   325→        }
   326→
   327→        set({
   328→          status: "unlocked",
   329→          masterKey,
   330→          vaultKey,
   331→          items,
   332→          vaults,
   333→          activeVault: "all",
   334→          // Merge with defaults so vaults persisted before the
   335→          // showFavicons / sortFavoritesFirst settings existed still get
   336→          // sane values instead of `undefined`.
   337→          settings: { ...DEFAULT_VAULT_SETTINGS, ...meta.settings },
   338→        });
   339→        return true;
   340→      },
   341→
   342→      lock: () => {
   343→        set({
   344→          status: "locked",
   345→          masterKey: null,
   346→          vaultKey: null,
   347→          items: [],
   348→          vaults: [],
   349→          activeVault: "all",
   350→          selectedId: null,
   351→          searchQuery: "",
   352→          editorOpen: false,
   353→          generatorOpen: false,
   354→          settingsOpen: false,
   355→          commandOpen: false,
   356→          vaultEditorOpen: false,
   357→          editingVaultId: null,
   358→          createVaultDialogOpen: false,
   359→        });
   360→      },
   361→
   362→      resetVault: async () => {
   363→        await wipeVault();
   364→        set({
   365→          status: "setup",
   366→          items: [],
   367→          vaults: [],
   368→          activeVault: "all",
   369→          masterKey: null,
   370→          vaultKey: null,
   371→          selectedId: null,
   372→          settings: DEFAULT_VAULT_SETTINGS,
   373→          vaultEditorOpen: false,
   374→          editingVaultId: null,
   375→          createVaultDialogOpen: false,
   376→        });
   377→      },
   378→
   379→      /* ------------------------------- CRUD -------------------------------- */
   380→
   381→      saveItem: async (input, existingId) => {
   382→        const { vaultKey } = get();
   383→        if (!vaultKey) throw new Error("Vault is locked");
   384→
   385→        const now = Date.now();
   386→        const id = existingId ?? randomId();
   387→        const existing = existingId ? get().items.find((i) => i.id === existingId) : undefined;
   388→        // Never inherit trashed state from a duplicated item — duplicates land
   389→        // in the active view, not in Trash.
   390→        const baseTrashed = existing ? existing.trashed : false;
   391→        const baseTrashedAt = existing ? existing.trashedAt : null;
   392→        const baseVaultId = existing ? existing.vaultId : (input as VaultItem).vaultId ?? null;
   393→        const item: VaultItem = {
   394→          ...(input as VaultItem),
   395→          id,
   396→          createdAt: existing?.createdAt ?? now,
   397→          updatedAt: now,
   398→          vaultId: baseVaultId,
   399→          trashed: baseTrashed,
   400→          trashedAt: baseTrashedAt,
   401→        } as VaultItem;
   402→
   403→        const { ciphertext, iv } = await encryptJson(item, vaultKey);
   404→        await putStoredItem({ id, type: item.type, ciphertext, iv, createdAt: item.createdAt, updatedAt: now });
   405→
   406→        set((state) => {
   407→          const others = state.items.filter((i) => i.id !== id);
   408→          const next = [item, ...others].sort((a, b) => b.updatedAt - a.updatedAt);
   409→          return { items: next, selectedId: id };
   410→        });
   411→        return item;
   412→      },
   413→
   414→      /**
   415→       * Soft-delete: marks the item as trashed and stamps trashedAt. The record
   416→       * stays in IndexedDB (still encrypted) so it can be restored. Auto-purged
   417→       * 30 days later on the next unlock.
   418→       */
   419→      trashItem: async (id) => {
   420→        await updateItemFlags(id, { trashed: true, trashedAt: Date.now() }, get, set);
   421→      },
   422→
   423→      restoreItem: async (id) => {
   424→        await updateItemFlags(id, { trashed: false, trashedAt: null }, get, set);
   425→      },
   426→
   427→      permanentlyDeleteItem: async (id) => {
   428→        const prev = get().items;
   429→        set((state) => ({
   430→          items: state.items.filter((i) => i.id !== id),
   431→          selectedId: state.selectedId === id ? null : state.selectedId,
   432→        }));
   433→        try {
   434→          await deleteStoredItem(id);
   435→        } catch (err) {
   436→          console.error("permanent delete failed", err);
   437→          set({ items: prev });
   438→          throw err;
   439→        }
   440→      },
   441→
   442→      emptyTrash: async () => {
   443→        const trashed = get().items.filter((i) => i.trashed);
   444→        const prev = get().items;
   445→        // Optimistic: drop all trashed items from UI.
   446→        set((state) => ({
   447→          items: state.items.filter((i) => !i.trashed),
   448→          selectedId:
   449→            state.selectedId && trashed.some((t) => t.id === state.selectedId)
   450→              ? null
   451→              : state.selectedId,
   452→        }));
   453→        try {
   454→          await Promise.all(trashed.map((t) => deleteStoredItem(t.id)));
   455→        } catch (err) {
   456→          console.error("emptyTrash failed", err);
   457→          set({ items: prev });
   458→          throw err;
   459→        }
   460→      },
   461→
   462→      moveItemToVault: async (itemId, vaultId) => {
   463→        await updateItemFlags(itemId, { vaultId, updatedAt: Date.now() }, get, set);
   464→      },
   465→
   466→      toggleFavorite: async (id) => {
   467→        const item = get().items.find((i) => i.id === id);
   468→        if (!item) return;
   469→        const updated: VaultItem = {
   470→          ...item,
   471→          favorite: !item.favorite,
   472→          updatedAt: Date.now(),
   473→        } as VaultItem;
   474→        const { vaultKey } = get();
   475→        if (vaultKey) {
   476→          const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   477→          await putStoredItem({ id, type: updated.type, ciphertext, iv, createdAt: updated.createdAt, updatedAt: updated.updatedAt });
   478→        }
   479→        set((state) => ({
   480→          items: state.items
   481→            .map((i) => (i.id === id ? updated : i))
   482→            .sort((a, b) => b.updatedAt - a.updatedAt),
   483→        }));
   484→      },
   485→
   486→      duplicateItem: async (id) => {
   487→        const item = get().items.find((i) => i.id === id);
   488→        if (!item) return;
   489→        const { id: _id, createdAt: _c, updatedAt: _u, trashed: _t, trashedAt: _ta, ...rest } = item;
   490→        await get().saveItem(rest as NewItemInput);
   491→      },
   492→
   493→      importItems: async (filename, text) => {
   494→        const { vaultKey } = get();
   495→        if (!vaultKey) throw new Error("Vault is locked");
   496→        const { result, items } = importFromText(filename, text);
   497→
   498→        // Encrypted LCKED imports require the export's password — handled by a
   499→        // dedicated UI flow. Here we surface a clear hint instead of failing.
   500→        if ((result as any).__raw) {
   501→          return {
   502→            imported: 0,
   503→            skipped: 0,
   504→            warnings: ["Encrypted LCKED files need their own password — use the dedicated import path."],
   505→          };
   506→        }
   507→
   508→        let imported = 0;
   509→        for (const input of items) {
   510→          try {
   511→            await get().saveItem(input);
   512→            imported++;
   513→          } catch {
   514→            result.skipped++;
   515→          }
   516→        }
   517→        result.imported = imported;
   518→        return result;
   519→      },
   520→
   521→      /* --------------------------- vaults (containers) ---------------------- */
   522→
   523→      createVault: async (name, color, icon) => {
   524→        const meta = await loadVaultMeta();
   525→        if (!meta) throw new Error("Vault meta missing");
   526→        const vault: VaultDef = {
   527→          id: randomId(),
   528→          name: name.trim() || "Untitled vault",
   529→          color,
   530→          icon,
   531→          createdAt: Date.now(),
   532→        };
   533→        const vaults = [...(meta.vaults ?? []), vault];
   534→        await saveVaultMeta({ ...meta, vaults });
   535→        set({ vaults, createVaultDialogOpen: false, vaultEditorOpen: false, editingVaultId: null });
   536→        return vault;
   537→      },
   538→
   539→      deleteVault: async (id) => {
   540→        const meta = await loadVaultMeta();
   541→        if (!meta) return;
   542→        const vaults = (meta.vaults ?? []).filter((v) => v.id !== id);
   543→        await saveVaultMeta({ ...meta, vaults });
   544→        // Orphan any items that lived in this vault — they fall back to the
   545→        // default vault (vaultId = null). Re-encrypt + persist each.
   546→        const { vaultKey, items } = get();
   547→        const orphaned = items.filter((i) => i.vaultId === id);
   548→        if (vaultKey && orphaned.length > 0) {
   549→          await Promise.all(
   550→            orphaned.map(async (it) => {
   551→              const next: VaultItem = { ...it, vaultId: null, updatedAt: Date.now() } as VaultItem;
   552→              const { ciphertext, iv } = await encryptJson(next, vaultKey);
   553→              await putStoredItem({
   554→                id: next.id,
   555→                type: next.type,
   556→                ciphertext,
   557→                iv,
   558→                createdAt: next.createdAt,
   559→                updatedAt: next.updatedAt,
   560→              });
   561→            }),
   562→          );
   563→          set((state) => ({
   564→            items: state.items.map((i) =>
   565→              i.vaultId === id ? ({ ...i, vaultId: null, updatedAt: Date.now() } as VaultItem) : i,
   566→            ),
   567→          }));
   568→        }
   569→        set((state) => ({
   570→          vaults,
   571→          activeVault: state.activeVault === id ? "all" : state.activeVault,
   572→          vaultEditorOpen: state.editingVaultId === id ? false : state.vaultEditorOpen,
   573→          editingVaultId: state.editingVaultId === id ? null : state.editingVaultId,
   574→        }));
   575→      },
   576→
   577→      renameVault: async (id, name) => {
   578→        await get().updateVault(id, { name });
   579→      },
   580→
   581→      updateVault: async (id, patch) => {
   582→        const meta = await loadVaultMeta();
   583→        if (!meta) return;
   584→        const vaults = (meta.vaults ?? []).map((v) =>
   585→          v.id === id ? { ...v, ...patch, name: patch.name ?? v.name } : v,
   586→        );
   587→        await saveVaultMeta({ ...meta, vaults });
   588→        set({ vaults, vaultEditorOpen: false, editingVaultId: null });
   589→      },
   590→
   591→      setActiveVault: (v) => set({ activeVault: v, settingsOpen: false }),

   715→/**
   716→ * Patch one or more flags on an item (vaultId / trashed / trashedAt / favorite)
   717→ * and re-encrypt + persist it. Used by trashItem / restoreItem / moveItemToVault
   718→ * so we have a single fault-tolerant code path.
   719→ */
   720→async function updateItemFlags(
   721→  id: string,
   722→  patch: Partial<Pick<VaultItem, "vaultId" | "trashed" | "trashedAt" | "updatedAt">>,
   723→  get: () => VaultState,
   724→  set: (partial: Partial<VaultState>) => void,
   725→): Promise<void> {
   726→  const item = get().items.find((i) => i.id === id);
   727→  if (!item) return;
   728→  const now = patch.updatedAt ?? Date.now();
   729→  const updated: VaultItem = { ...item, ...patch, updatedAt: now } as VaultItem;
   730→  const { vaultKey } = get();
   731→  if (vaultKey) {
   732→    const { ciphertext, iv } = await encryptJson(updated, vaultKey);
   733→    await putStoredItem({
   734→      id,
   735→      type: updated.type,
   736→      ciphertext,
   737→      iv,
   738→      createdAt: updated.createdAt,
   739→      updatedAt: now,
   740→    });
   741→  }
   742→  set({
   743→    items: get()
   744→      .items.map((i) => (i.id === id ? updated : i))
   745→      .sort((a, b) => b.updatedAt - a.updatedAt),
   746→  });
   747→}

   749→/**
   750→ * Copy-to-clipboard with an automatic 30-second clear. Returns the timeout id
   751→ * so callers can cancel early if needed. Sensitive values never linger.
   752→ */
   753→const clipboardTimers = new Map<string, ReturnType<typeof setTimeout>>();
   754→
   755→export async function copyWithAutoClear(
   756→  value: string,
   757→  key = "default",
   758→  clearMs = 30_000,
   759→): Promise<void> {
   760→  if (typeof navigator === "undefined" || !navigator.clipboard?.writeText) {
   761→    throw new Error("Clipboard API unavailable");
   762→  }
   763→  await navigator.clipboard.writeText(value);
   764→  // Clear any prior timer for this key.
   765→  const prior = clipboardTimers.get(key);
   766→  if (prior) clearTimeout(prior);
   767→  const timer = setTimeout(async () => {
   768→    try {
   769→      const current = await navigator.clipboard.readText().catch(() => "");
   770→      if (current === value) {
   771→        await navigator.clipboard.writeText("");
   772→      }
   773→    } catch {
   774→      // readText may be denied — best-effort clear.
   775→    }
   776→    clipboardTimers.delete(key);
   777→  }, clearMs);
   778→  clipboardTimers.set(key, timer);
   779→}
   780→
   781→export function cancelClipboardClear(key = "default"): void {
   782→  const t = clipboardTimers.get(key);
   783→  if (t) {
   784→    clearTimeout(t);
   785→    clipboardTimers.delete(key);
   786→  }
   787→}
