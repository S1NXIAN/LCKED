     6→     6→ * Proton Pass–style vault selector. Renders the fixed rows (All / Favorites /
     7→     7→ * Trash) plus every user-defined vault. Each row is a drag-and-drop drop
     8→     8→ * target so an item dragged from the list can be moved between vaults.

    11→    11→ *   • Rows that host a ⋮ dropdown are rendered as <div role="button"> (NOT

    17→    17→ *     "Delete" calls deleteVault (which orphan-rescues any items inside).
    18→    18→ *   • Each row wrapper carries a `data-vault-key` attribute consumed by the
    19→    19→ *     sliding VaultActiveHighlight (rAF spring, same pattern as item-list's
    20→    20→ *     ActiveRowHighlight). The highlight lives in the OUTER container so it
    21→    21→ *     can reach Trash (which sits outside the scroll area).
    22→    22→ *   • CustomVaultRow is wrapped in <ContextMenu> with the same items as the

   151→   151→export interface VaultIconProps {
   152→   152→  /** Vault icon id (see VAULT_ICONS). */
   153→   153→  icon: string;
   154→   154→  /** Vault color id (see VAULT_COLORS). */
   155→   155→  color: string;
   156→   156→  /** Pixel size of the square swatch. Default 28. */
   157→   157→  size?: number;
   158→   158→  className?: string;
   159→   159→}
   160→   160→
   161→   161→/**
   162→   162→ * Render a vault's icon inside a colored rounded swatch. The fill is the
   163→   163→ * vault color at 18% opacity; the icon glyph uses the full hex for contrast.
   164→   164→ * Used by VaultsSidebar rows AND the create-vault-dialog picker preview.
   165→   165→ */
   166→   166→export function VaultIcon({ icon, color, size = 28, className }: VaultIconProps) {
   167→   167→  const hex = vaultColorHex(color);
   168→   168→  const Resolved = LUCIDE_BY_ID[icon] ?? Home;
   169→   169→  const px = `${size}px`;
   170→   170→  const glyph = Math.round(size * 0.55);
   171→   171→  return (
   172→   172→    <span
   173→   173→      className={cn("inline-flex shrink-0 items-center justify-center rounded-lg", className)}
   174→   174→      style={{
   175→   175→        width: px,
   176→   176→        height: px,
   177→   177→        backgroundColor: `${hex}29`, // ~16% opacity
   178→   178→        color: hex,
   179→   179→      }}
   180→   180→      aria-hidden="true"
   181→   181→    >
   182→   182→      <Resolved size={glyph} strokeWidth={2} />
   183→   183→    </span>
   184→   184→  );
   185→   185→}
   186→   186→

   191→   191→
   192→   192→interface VaultRowProps {
   193→   193→  /** Glyph rendered in the swatch. Pass a Lucide component for fixed rows. */
   194→   194→  icon: React.ReactNode;
   195→   195→  /** Row label. */
   196→   196→  label: string;
   197→   197→  /** Item count shown on the right. */
   198→   198→  count: number;
   199→   199→  /** Whether this row is the active filter. */
   200→   200→  active: boolean;
   201→   201→  /** Click handler — sets the active vault filter. */
   202→   202→  onSelect: () => void;
   203→   203→  /** Optional drag-and-drop target vault id (null = no drop). */
   204→   204→  dropVaultId?: string | null;
   205→   205→  /** Optional amber-tinted style (used by Trash when non-empty). */
   206→   206→  warn?: boolean;
   207→   207→  /** Optional ⋮ menu node (rendered in the reserved slot). */
   208→   208→  menu?: React.ReactNode;
   209→   209→  /** Whether the row is being dragged over. */
   210→   210→  dragOver?: boolean;
   211→   211→}
   212→   212→
   213→   213→/**
   214→   214→ * A single sidebar row. When `menu` is provided we render a div instead of a
   215→   215→ * button to avoid invalid nested-button HTML. The reserved `menuSlot` keeps
   216→   216→ * every row's right edge aligned whether or not a menu is present.
   217→   217→ *
   218→   218→ * The active state keeps `text-accent-foreground` but NO `bg-accent` — the
   219→   219→ * sliding VaultActiveHighlight provides the background, so a static one
   220→   220→ * would visually fight the spring.
   221→   221→ */
   222→   222→function VaultRow({
   223→   223→  icon,
   224→   224→  label,
   225→   225→  count,
   226→   226→  active,
   227→   227→  onSelect,
   228→   228→  dropVaultId,
   229→   229→  warn,
   230→   230→  menu,
   231→   231→  dragOver,
   232→   232→}: VaultRowProps) {
   233→   233→  const Tag = menu ? "div" : "button";
   234→   234→  const tagProps = menu
   235→   235→    ? { role: "button", tabIndex: 0 }
   236→   236→    : { type: "button" as const };
   237→   237→
   238→   238→  const handleDrop = (e: React.DragEvent) => {
   239→   239→    if (dropVaultId === undefined) return;
   240→   240→    e.preventDefault();
   241→   241→    const id = e.dataTransfer.getData("text/lcked-item");
   242→   242→    if (!id) return;
   243→   243→    useVault.getState().moveItemToVault(id, dropVaultId ?? null);
   244→   244→    toast.success(`Moved to ${label}`);
   245→   245→  };
   246→   246→  const handleDragOver = (e: React.DragEvent) => {
   247→   247→    if (dropVaultId === undefined) return;
   248→   248→    e.preventDefault();
   249→   249→    e.dataTransfer.dropEffect = "move";
   250→   250→  };
   251→   251→  const handleKeyDown = (e: React.KeyboardEvent) => {
   252→   252→    if (!menu) return;
   253→   253→    if (e.key === "Enter" || e.key === " ") {
   254→   254→      e.preventDefault();
   255→   255→      onSelect();
   256→   256→    }
   257→   257→  };
   258→   258→
   259→   259→  return (
   260→   260→    <Tag
   261→   261→      {...tagProps}
   262→   262→      onClick={(e: React.MouseEvent) => {
   263→   263→        const target = e.target as HTMLElement;
   264→   264→        if (target.closest("[data-menu-slot]")) return;
   265→   265→        onSelect();
   266→   266→      }}
   267→   267→      onKeyDown={menu ? handleKeyDown : undefined}
   268→   268→      onDrop={handleDrop}
   269→   269→      onDragOver={handleDragOver}
   270→   270→      aria-current={active ? "true" : undefined}
   271→   271→      className={cn(
   272→   272→        "group relative flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition-colors",
   273→   273→        "focus:outline-none focus-visible:ring-2 focus-visible:ring-ring/60",
   274→   274→        active
   275→   275→          ? "text-accent-foreground"
   276→   276→          : warn
   277→   277→            ? "text-foreground hover:bg-amber-500/10"
   278→   278→            : "text-foreground/80 hover:bg-muted/60 hover:text-foreground",
   279→   279→        dragOver && "ring-2 ring-primary/60 ring-offset-1 ring-offset-background",
   280→   280→      )}
   281→   281→    >
   282→   282→      <span className="flex h-7 w-7 shrink-0 items-center justify-center">{icon}</span>
   283→   283→      <span className="min-w-0 flex-1">
   284→   284→        <span className="block truncate text-sm font-medium leading-tight">{label}</span>
   285→   285→        <span
   286→   286→          className={cn(
   287→   287→            "block text-xs leading-tight tabular-nums",
   288→   288→            warn && !active ? "text-amber-400/80" : active ? "text-accent-foreground/70" : "text-muted-foreground",
   289→   289→          )}
   290→   290→        >
   291→   291→          {count} {count === 1 ? "item" : "items"}
   292→   292→        </span>
   293→   293→      </span>
   294→   294→      <span
   295→   295→        data-menu-slot
   296→   296→        className="flex h-7 w-7 shrink-0 items-center justify-center"
   297→   297→      >
   298→   298→        {menu}
   299→   299→      </span>
   300→   300→    </Tag>
   301→   301→  );
   302→   302→}

   321→   321→/**
   322→   322→ * Build the menu contents shared by the 3-dots dropdown AND the right-click

   325→   325→function VaultMenuItems({
   326→   326→  vault,
   327→   327→  onRename,
   328→   328→  onDelete,
   329→   329→  onHide,
   330→   330→  onMoveAll,
   331→   331→}: {
   332→   332→  vault: VaultDef;
   333→   333→  onRename: () => void;

   340→   340→

   342→   342→    <>
   343→   343→      <ContextMenuItem onSelect={onRename}>
   344→   344→        <Pencil className="h-3.5 w-3.5" />

   389→   389→function CustomVaultRow({

   445→   445→                    <Pencil className="h-3.5 w-3.5" />
   446→   446→                    Edit

   499→   499→          />

   502→   502→
   503→   503→      <AlertDialog open={confirmOpen} onOpenChange={setConfirmOpen}>

   530→   530→/* --------------------- VaultActiveHighlight (rAF spring) --------------------- */
   531→   531→
   532→   532→/**
   533→   533→ * Single persistent highlight rendered ONCE inside the sidebar outer
   534→   534→ * container. Slides toward the active row's bounding box via a
   535→   535→ * requestAnimationFrame spring (exponential lerp, factor 0.22). Uses the
   536→   536→ * same `animateRef` pattern as item-list's ActiveRowHighlight.
   537→   537→ *
   538→   538→ * The highlight lives in the OUTER container (NOT the scroll area) so it can
   539→   539→ * reach Trash, which sits below the scrollable vault list with `mt-auto`.
   540→   540→ */
   541→   541→function VaultActiveHighlight({
   542→   542→  containerRef,
   543→   543→  activeKey,
   544→   544→}: {
   545→   545→  containerRef: React.RefObject<HTMLDivElement | null>;
   546→   546→  activeKey: string;
   547→   547→}) {
   548→   548→  const highlightRef = React.useRef<HTMLDivElement | null>(null);
   549→   549→  const rafRef = React.useRef<number | null>(null);
   550→   550→  const targetRef = React.useRef<{ x: number; y: number; w: number; h: number } | null>(null);
   551→   551→  const wasVisibleRef = React.useRef(false);
   552→   552→  const [visible, setVisible] = React.useState(false);
   553→   553→  const animateRef = React.useRef<(() => void) | null>(null);
   554→   554→
   555→   555→  const measure = React.useCallback(() => {
   556→   556→    const container = containerRef.current;
   557→   557→    if (!activeKey || !container) {
   558→   558→      targetRef.current = null;
   559→   559→      setVisible(false);
   560→   560→      return;
   561→   561→    }
   562→   562→    const el = container.querySelector<HTMLElement>(
   563→   563→      `[data-vault-key="${CSS.escape(activeKey)}"]`,
   564→   564→    );
   565→   565→    if (!el) {
   566→   566→      targetRef.current = null;
   567→   567→      setVisible(false);
   568→   568→      return;
   569→   569→    }
   570→   570→    const r = el.getBoundingClientRect();
   571→   571→    const cr = container.getBoundingClientRect();
   572→   572→    const next = {
   573→   573→      x: r.left - cr.left,
   574→   574→      y: r.top - cr.top,
   575→   575→      w: r.width,
   576→   576→      h: r.height,
   577→   577→    };
   578→   578→    if (!wasVisibleRef.current) {
   579→   579→      const hl = highlightRef.current;
   580→   580→      if (hl) {
   581→   581→        hl.style.transform = `translate(${next.x}px, ${next.y}px)`;
   582→   582→        hl.style.width = `${next.w}px`;
   583→   583→        hl.style.height = `${next.h}px`;
   584→   584→      }
   585→   585→    }
   586→   586→    targetRef.current = next;
   587→   587→    setVisible(true);
   588→   588→  }, [activeKey, containerRef]);
   589→   589→
   590→   590→  React.useEffect(() => {
   591→   591→    const step = () => {
   592→   592→      const hl = highlightRef.current;
   593→   593→      const target = targetRef.current;
   594→   594→      const container = containerRef.current;
   595→   595→      if (!hl || !target || !container) {
   596→   596→        rafRef.current = null;
   597→   597→        return;
   598→   598→      }
   599→   599→      const cur = hl.getBoundingClientRect();
   600→   600→      const cr = container.getBoundingClientRect();
   601→   601→      const cx = cur.left - cr.left;
   602→   602→      const cy = cur.top - cr.top;
   603→   603→      const FACTOR = 0.22;
   604→   604→      const nx = cx + (target.x - cx) * FACTOR;
   605→   605→      const ny = cy + (target.y - cy) * FACTOR;
   606→   606→      const nw = cur.width + (target.w - cur.width) * FACTOR;
   607→   607→      const nh = cur.height + (target.h - cur.height) * FACTOR;
   608→   608→      const settled =
   609→   609→        Math.abs(target.x - nx) < 0.5 &&
   610→   610→        Math.abs(target.y - ny) < 0.5 &&
   611→   611→        Math.abs(target.w - nw) < 0.5 &&
   612→   612→        Math.abs(target.h - nh) < 0.5;
   613→   613→      const fx = settled ? target.x : nx;
   614→   614→      const fy = settled ? target.y : ny;
   615→   615→      const fw = settled ? target.w : nw;
   616→   616→      const fh = settled ? target.h : nh;
   617→   617→      hl.style.transform = `translate(${fx}px, ${fy}px)`;
   618→   618→      hl.style.width = `${fw}px`;
   619→   619→      hl.style.height = `${fh}px`;
   620→   620→      if (settled) {
   621→   621→        rafRef.current = null;
   622→   622→        return;
   623→   623→      }
   624→   624→      rafRef.current = requestAnimationFrame(step);
   625→   625→    };
   626→   626→    animateRef.current = step;
   627→   627→    return () => {
   628→   628→      animateRef.current = null;
   629→   629→    };
   630→   630→  }, [containerRef]);
   631→   631→
   632→   632→  React.useEffect(() => {
   633→   633→    wasVisibleRef.current = visible;
   634→   634→  }, [visible]);
   635→   635→
   636→   636→  const kick = React.useCallback(() => {
   637→   637→    measure();
   638→   638→    if (!rafRef.current && animateRef.current) {
   639→   639→      rafRef.current = requestAnimationFrame(animateRef.current);
   640→   640→    }
   641→   641→  }, [measure]);
   642→   642→
   643→   643→  React.useEffect(() => {
   644→   644→    kick();
   645→   645→    return () => {
   646→   646→      if (rafRef.current) {
   647→   647→        cancelAnimationFrame(rafRef.current);
   648→   648→        rafRef.current = null;
   649→   649→      }
   650→   650→    };
   651→   651→  }, [activeKey, kick]);
   652→   652→
   653→   653→  React.useEffect(() => {
   654→   654→    const container = containerRef.current;
   655→   655→    if (!container) return;
   656→   656→    const observer = new MutationObserver(() => kick());
   657→   657→    observer.observe(container, {
   658→   658→      childList: true,
   659→   659→      subtree: true,
   660→   660→      attributes: true,
   661→   661→      attributeFilter: ["data-vault-key"],
   662→   662→    });
   663→   663→    return () => observer.disconnect();
   664→   664→  }, [containerRef, kick]);
   665→   665→
   666→   666→  React.useEffect(() => {
   667→   667→    const container = containerRef.current;
   668→   668→    if (!container) return;
   669→   669→    const handler = () => kick();
   670→   670→    container.addEventListener("scroll", handler, { passive: true });
   671→   671→    window.addEventListener("scroll", handler, { passive: true });
   672→   672→    window.addEventListener("resize", handler, { passive: true });
   673→   673→    return () => {
   674→   674→      container.removeEventListener("scroll", handler);
   675→   675→      window.removeEventListener("scroll", handler);
   676→   676→      window.removeEventListener("resize", handler);
   677→   677→    };
   678→   678→  }, [containerRef, kick]);
   679→   679→
   680→   680→  if (!visible) return null;
   681→   681→  return (
   682→   682→    <div
   683→   683→      ref={highlightRef}
   684→   684→      aria-hidden
   685→   685→      className="lcked-active-glow pointer-events-none absolute left-0 top-0 rounded-lg"
   686→   686→      style={{ width: 0, height: 0, transform: "translate(0px, 0px)" }}
   687→   687→    />
   688→   688→  );
   689→   689→}
